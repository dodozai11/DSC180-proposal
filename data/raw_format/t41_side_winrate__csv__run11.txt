def q1(text):
    """Returns the win rate of blue-side teams."""
    lines = text.strip().split('\n')[1:]  # Skip header
    blue_wins = 0
    blue_total = 0
    
    for line in lines:
        parts = line.split(',')
        side = parts[2]
        result = parts[3]
        
        if side == 'blue':
            blue_total += 1
            if result == 'win':
                blue_wins += 1
    
    return blue_wins / blue_total if blue_total > 0 else 0


def q2(text):
    """Returns the average length in minutes of games won by red-side teams."""
    lines = text.strip().split('\n')[1:]  # Skip header
    red_wins_minutes = []
    
    for line in lines:
        parts = line.split(',')
        side = parts[2]
        result = parts[3]
        game_minutes = float(parts[4])
        
        if side == 'red' and result == 'win':
            red_wins_minutes.append(game_minutes)
    
    return sum(red_wins_minutes) / len(red_wins_minutes) if red_wins_minutes else 0


def q3(text):
    """Returns the name of the team with the most wins."""
    lines = text.strip().split('\n')[1:]  # Skip header
    team_wins = {}
    
    for line in lines:
        parts = line.split(',')
        team = parts[1]
        result = parts[3]
        
        if team not in team_wins:
            team_wins[team] = 0
        
        if result == 'win':
            team_wins[team] += 1
    
    return max(team_wins, key=team_wins.get) if team_wins else None

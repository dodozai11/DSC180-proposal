```python
import re

def parse_data(text):
    """Parse the match data from text format."""
    pattern = r"(\w+(?:\s+\w+)*?)\s+played\s+(\w+)\s+in\s+the\s+jungle\s+and\s+(won|lost);\s+at\s+10\s+minutes\s+they\s+had\s+a\s+CS\s+difference\s+of\s+([-\d]+),\s+(\d+)\s+kills\s+and\s+(\d+)\s+assists\."
    matches = re.findall(pattern, text)
    
    data = []
    for match in matches:
        player_name, champion, result, cs_diff, kills, assists = match
        data.append({
            'player': player_name,
            'champion': champion,
            'result': result,
            'cs_diff': int(cs_diff),
            'kills': int(kills),
            'assists': int(assists)
        })
    
    return data


def q1(text):
    """Return the win rate of players with a positive CS difference at 10 minutes."""
    data = parse_data(text)
    
    positive_cs_players = [row for row in data if row['cs_diff'] > 0]
    
    if not positive_cs_players:
        return 0
    
    wins = sum(1 for row in positive_cs_players if row['result'] == 'won')
    win_rate = wins / len(positive_cs_players)
    
    return win_rate


def q2(text):
    """Return the total number of kills at 10 minutes across all rows."""
    data = parse_data(text)
    
    total_kills = sum(row['kills'] for row in data)
    
    return total_kills


def q3(text):
    """Return the name of the champion with the most assists at 10 minutes in total."""
    data = parse_data(text)
    
    champion_assists = {}
    for row in data:
        champion = row['champion']
        if champion not in champion_assists:
            champion_assists[champion] = 0
        champion_assists[champion] += row['assists']
    
    champion_with_most_assists = max(champion_assists, key=champion_assists.get)
    
    return champion_with_most_assists
```

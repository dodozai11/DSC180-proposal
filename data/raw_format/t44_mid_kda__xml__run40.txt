```python
import re
from collections import Counter

def parse_data(text):
    """Parse the XML-like text data into a list of dictionaries."""
    rows = []
    pattern = r'<row><player>(.+?)</player><team>(.+?)</team><champion>(.+?)</champion><kills>(\d+)</kills><deaths>(\d+)</deaths><assists>(\d+)</assists><result>(\w+)</result></row>'
    
    matches = re.findall(pattern, text)
    for match in matches:
        row = {
            'player': match[0],
            'team': match[1],
            'champion': match[2],
            'kills': int(match[3]),
            'deaths': int(match[4]),
            'assists': int(match[5]),
            'result': match[6]
        }
        rows.append(row)
    
    return rows


def q1(text):
    """Return the name of the player with the highest KDA."""
    rows = parse_data(text)
    
    best_player = None
    best_kda = -1
    
    for row in rows:
        kda = (row['kills'] + row['assists']) / max(row['deaths'], 1)
        if kda > best_kda:
            best_kda = kda
            best_player = row['player']
    
    return best_player


def q2(text):
    """Return the average number of kills in games that were won."""
    rows = parse_data(text)
    
    wins = [row['kills'] for row in rows if row['result'] == 'win']
    
    if not wins:
        return 0
    
    return sum(wins) / len(wins)


def q3(text):
    """Return the name of the most-picked champion."""
    rows = parse_data(text)
    
    champions = [row['champion'] for row in rows]
    champion_counts = Counter(champions)
    
    most_picked = champion_counts.most_common(1)[0][0]
    
    return most_picked
```

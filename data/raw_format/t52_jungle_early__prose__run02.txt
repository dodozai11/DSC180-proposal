import re

def q1(text):
    """Return the win rate of players with a positive CS difference at 10 minutes."""
    pattern = r'(\w+(?:\s+\w+)*?)\s+played\s+(\w+)\s+in\s+the\s+jungle\s+and\s+(won|lost);.*?CS difference of\s+([-\d]+),.*?kills'
    matches = re.findall(pattern, text)
    
    positive_cs_wins = 0
    positive_cs_total = 0
    
    for player, champion, result, cs_diff in matches:
        cs_diff = int(cs_diff)
        if cs_diff > 0:
            positive_cs_total += 1
            if result == 'won':
                positive_cs_wins += 1
    
    if positive_cs_total == 0:
        return 0
    
    return positive_cs_wins / positive_cs_total


def q2(text):
    """Return the total number of kills at 10 minutes across all rows."""
    pattern = r'(\w+(?:\s+\w+)*?)\s+played\s+(\w+)\s+in\s+the\s+jungle\s+and\s+(won|lost);.*?CS difference of\s+([-\d]+),\s+(\d+)\s+kills'
    matches = re.findall(pattern, text)
    
    total_kills = 0
    for player, champion, result, cs_diff, kills in matches:
        total_kills += int(kills)
    
    return total_kills


def q3(text):
    """Return the name of the champion with the most assists at 10 minutes in total."""
    pattern = r'(\w+(?:\s+\w+)*?)\s+played\s+(\w+)\s+in\s+the\s+jungle\s+and\s+(won|lost);.*?CS difference of\s+([-\d]+),\s+(\d+)\s+kills\s+and\s+(\d+)\s+assists'
    matches = re.findall(pattern, text)
    
    champion_assists = {}
    for player, champion, result, cs_diff, kills, assists in matches:
        assists = int(assists)
        if champion not in champion_assists:
            champion_assists[champion] = 0
        champion_assists[champion] += assists
    
    if not champion_assists:
        return None
    
    return max(champion_assists, key=champion_assists.get)

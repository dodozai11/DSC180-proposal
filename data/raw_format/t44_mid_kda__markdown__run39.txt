def q1(text):
    """Returns the name of the player with the highest KDA."""
    lines = text.strip().split('\n')
    # Skip header row
    players = []
    for line in lines[1:]:
        parts = line.split('|')
        # Remove empty strings from split (due to leading/trailing |)
        parts = [p.strip() for p in parts if p.strip()]
        player = parts[0]
        kills = int(parts[3])
        deaths = int(parts[4])
        assists = int(parts[5])
        kda = (kills + assists) / max(deaths, 1)
        players.append((player, kda))
    
    # Return player with highest KDA
    return max(players, key=lambda x: x[1])[0]


def q2(text):
    """Returns the average number of kills in games that were won."""
    lines = text.strip().split('\n')
    # Skip header row
    wins = []
    for line in lines[1:]:
        parts = line.split('|')
        # Remove empty strings from split (due to leading/trailing |)
        parts = [p.strip() for p in parts if p.strip()]
        kills = int(parts[3])
        result = parts[6]
        if result == 'win':
            wins.append(kills)
    
    # Return average kills in wins
    return sum(wins) / len(wins) if wins else 0


def q3(text):
    """Returns the name of the most-picked champion."""
    lines = text.strip().split('\n')
    # Skip header row
    champion_counts = {}
    for line in lines[1:]:
        parts = line.split('|')
        # Remove empty strings from split (due to leading/trailing |)
        parts = [p.strip() for p in parts if p.strip()]
        champion = parts[2]
        champion_counts[champion] = champion_counts.get(champion, 0) + 1
    
    # Return champion with highest count
    return max(champion_counts, key=champion_counts.get)

import xml.etree.ElementTree as ET

def q1(text):
    """Returns the win rate of players with a positive CS difference at 10 minutes."""
    root = ET.fromstring(f"<root>{text}</root>")
    rows = root.findall("row")
    
    positive_cs_diff = []
    for row in rows:
        cs_diff_10 = int(row.find("cs_diff_10").text)
        if cs_diff_10 > 0:
            result = row.find("result").text
            positive_cs_diff.append(result == "win")
    
    if len(positive_cs_diff) == 0:
        return 0
    
    wins = sum(positive_cs_diff)
    return wins / len(positive_cs_diff)

def q2(text):
    """Returns the total number of kills at 10 minutes across all rows."""
    root = ET.fromstring(f"<root>{text}</root>")
    rows = root.findall("row")
    
    total_kills = 0
    for row in rows:
        kills_at_10 = int(row.find("kills_at_10").text)
        total_kills += kills_at_10
    
    return total_kills

def q3(text):
    """Returns the name of the champion with the most assists at 10 minutes in total."""
    root = ET.fromstring(f"<root>{text}</root>")
    rows = root.findall("row")
    
    champion_assists = {}
    for row in rows:
        champion = row.find("champion").text
        assists_at_10 = int(row.find("assists_at_10").text)
        
        if champion not in champion_assists:
            champion_assists[champion] = 0
        champion_assists[champion] += assists_at_10
    
    return max(champion_assists, key=champion_assists.get)

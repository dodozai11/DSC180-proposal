```python
import json
import re

def q1(text):
    data = json.loads(text)
    
    winners = [player for player in data if player['result'] == 'win']
    losers = [player for player in data if player['result'] == 'loss']
    
    avg_vision_winners = sum(p['vision_score'] for p in winners) / len(winners)
    avg_vision_losers = sum(p['vision_score'] for p in losers) / len(losers)
    
    return avg_vision_winners - avg_vision_losers

def q2(text):
    data = json.loads(text)
    
    max_wards_player = max(data, key=lambda p: p['wards_placed'])
    return max_wards_player['player']

def q3(text):
    data = json.loads(text)
    
    count = sum(1 for player in data if player['control_wards'] >= 15)
    return count
```

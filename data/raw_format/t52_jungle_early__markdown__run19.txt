def q1(text):
    lines = text.strip().split('\n')
    header = lines[0].split('|')[1:-1]
    header = [h.strip() for h in header]
    
    rows = []
    for line in lines[2:]:
        if line.strip():
            cells = line.split('|')[1:-1]
            cells = [c.strip() for c in cells]
            rows.append(dict(zip(header, cells)))
    
    positive_cs_diff = [row for row in rows if int(row['cs_diff_10']) > 0]
    
    if not positive_cs_diff:
        return 0
    
    wins = sum(1 for row in positive_cs_diff if row['result'] == 'win')
    return wins / len(positive_cs_diff)


def q2(text):
    lines = text.strip().split('\n')
    header = lines[0].split('|')[1:-1]
    header = [h.strip() for h in header]
    
    rows = []
    for line in lines[2:]:
        if line.strip():
            cells = line.split('|')[1:-1]
            cells = [c.strip() for c in cells]
            rows.append(dict(zip(header, cells)))
    
    total_kills = sum(int(row['kills_at_10']) for row in rows)
    return total_kills


def q3(text):
    lines = text.strip().split('\n')
    header = lines[0].split('|')[1:-1]
    header = [h.strip() for h in header]
    
    rows = []
    for line in lines[2:]:
        if line.strip():
            cells = line.split('|')[1:-1]
            cells = [c.strip() for c in cells]
            rows.append(dict(zip(header, cells)))
    
    champion_assists = {}
    for row in rows:
        champion = row['champion']
        assists = int(row['assists_at_10'])
        champion_assists[champion] = champion_assists.get(champion, 0) + assists
    
    return max(champion_assists, key=champion_assists.get)

def q1(text):
    """Return the name of the player with the highest CS per minute."""
    lines = text.strip().split('\n')
    # Skip header line
    data = lines[2:]
    
    max_cs = -1
    max_player = None
    
    for line in data:
        if not line.strip():
            continue
        parts = [p.strip() for p in line.split('|')]
        # Filter out empty strings from split
        parts = [p for p in parts if p]
        
        if len(parts) >= 5:
            player = parts[0]
            cs_per_min = float(parts[4])
            
            if cs_per_min > max_cs:
                max_cs = cs_per_min
                max_player = player
    
    return max_player


def q2(text):
    """Return the average gold share of players who won."""
    lines = text.strip().split('\n')
    # Skip header line
    data = lines[2:]
    
    total_gold_share = 0
    win_count = 0
    
    for line in data:
        if not line.strip():
            continue
        parts = [p.strip() for p in line.split('|')]
        # Filter out empty strings from split
        parts = [p for p in parts if p]
        
        if len(parts) >= 5:
            result = parts[3]
            gold_share = float(parts[5])
            
            if result == 'win':
                total_gold_share += gold_share
                win_count += 1
    
    if win_count == 0:
        return 0
    
    return total_gold_share / win_count

import re
from xml.etree import ElementTree as ET

def parse_data(text):
    """Parse XML data and return list of dictionaries"""
    root = ET.fromstring(f"<root>{text}</root>")
    data = []
    for row in root.findall('row'):
        data.append({
            'game': row.find('game').text,
            'team': row.find('team').text,
            'side': row.find('side').text,
            'result': row.find('result').text,
            'gold_diff_15': int(row.find('gold_diff_15').text)
        })
    return data

def q1(text):
    """Return the average gold difference at 15 minutes of the winning teams"""
    data = parse_data(text)
    winning_gold_diffs = [row['gold_diff_15'] for row in data if row['result'] == 'win']
    return sum(winning_gold_diffs) / len(winning_gold_diffs)

def q2(text):
    """Return how many teams won despite a negative gold difference at 15 minutes"""
    data = parse_data(text)
    count = 0
    for row in data:
        if row['result'] == 'win' and row['gold_diff_15'] < 0:
            count += 1
    return count

def q3(text):
    """Return the win rate of teams that were ahead in gold at 15 minutes"""
    data = parse_data(text)
    ahead_in_gold = [row for row in data if row['gold_diff_15'] > 0]
    if len(ahead_in_gold) == 0:
        return 0
    wins = sum(1 for row in ahead_in_gold if row['result'] == 'win')
    return wins / len(ahead_in_gold)

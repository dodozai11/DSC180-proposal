```python
import re
from collections import defaultdict

def q1(text):
    """Returns the number of different champions played."""
    champions = set()
    # Pattern: "played <Champion> in the jungle"
    matches = re.findall(r'played (\w+(?:\'?\w+)?(?:\'?\w+)?) in the jungle', text)
    for match in matches:
        champions.add(match)
    return len(champions)


def q2(text):
    """Returns the champion with the best win rate among champions played at least twice."""
    champion_stats = defaultdict(lambda: {'wins': 0, 'total': 0})
    
    # Pattern: "<Player> played <Champion> in the jungle and <Result>"
    matches = re.findall(r'played (\w+(?:\'?\w+)?(?:\'?\w+)?) in the jungle and (won|lost)', text)
    
    for champion, result in matches:
        champion_stats[champion]['total'] += 1
        if result == 'won':
            champion_stats[champion]['wins'] += 1
    
    # Filter champions played at least twice
    eligible_champions = {champ: stats for champ, stats in champion_stats.items() if stats['total'] >= 2}
    
    if not eligible_champions:
        return None
    
    # Find champion with best win rate
    best_champion = max(eligible_champions.items(), 
                       key=lambda x: x[1]['wins'] / x[1]['total'])
    return best_champion[0]


def q3(text):
    """Returns the average number of deaths in games that were lost."""
    losses = []
    
    # Pattern: "<Player> played <Champion> in the jungle and lost, dying <Deaths> times"
    matches = re.findall(r'and lost, dying (\d+) times', text)
    
    for deaths in matches:
        losses.append(int(deaths))
    
    if not losses:
        return 0
    
    return sum(losses) / len(losses)
```

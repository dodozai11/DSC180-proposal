def q1(text):
    """Return the name of the player with the highest KDA."""
    lines = text.strip().split('\n')
    header = lines[0].split(',')
    
    max_kda = -1
    best_player = None
    
    for line in lines[1:]:
        values = line.split(',')
        player = values[0]
        kills = int(values[3])
        deaths = int(values[4])
        assists = int(values[5])
        
        kda = (kills + assists) / max(deaths, 1)
        
        if kda > max_kda:
            max_kda = kda
            best_player = player
    
    return best_player


def q2(text):
    """Return the average number of kills in games that were won."""
    lines = text.strip().split('\n')
    
    total_kills = 0
    win_count = 0
    
    for line in lines[1:]:
        values = line.split(',')
        kills = int(values[3])
        result = values[6]
        
        if result == 'win':
            total_kills += kills
            win_count += 1
    
    if win_count == 0:
        return 0
    
    return total_kills / win_count


def q3(text):
    """Return the name of the most-picked champion."""
    lines = text.strip().split('\n')
    
    champion_count = {}
    
    for line in lines[1:]:
        values = line.split(',')
        champion = values[2]
        
        champion_count[champion] = champion_count.get(champion, 0) + 1
    
    most_picked = max(champion_count, key=champion_count.get)
    
    return most_picked

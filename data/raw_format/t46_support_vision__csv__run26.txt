import anthropic
import csv
import io


def q1(text: str) -> float:
    """Return the average vision score of players who won minus the average vision score of players who lost."""
    lines = text.strip().split('\n')
    reader = csv.DictReader(io.StringIO('\n'.join(lines)))
    
    wins = []
    losses = []
    
    for row in reader:
        vision_score = float(row['vision_score'])
        if row['result'] == 'win':
            wins.append(vision_score)
        else:
            losses.append(vision_score)
    
    avg_wins = sum(wins) / len(wins) if wins else 0
    avg_losses = sum(losses) / len(losses) if losses else 0
    
    return avg_wins - avg_losses


def q2(text: str) -> str:
    """Return the name of the player who placed the most wards."""
    lines = text.strip().split('\n')
    reader = csv.DictReader(io.StringIO('\n'.join(lines)))
    
    max_wards = -1
    max_player = None
    
    for row in reader:
        wards_placed = int(row['wards_placed'])
        if wards_placed > max_wards:
            max_wards = wards_placed
            max_player = row['player']
    
    return max_player


def q3(text: str) -> int:
    """Return how many players bought at least 15 control wards."""
    lines = text.strip().split('\n')
    reader = csv.DictReader(io.StringIO('\n'.join(lines)))
    
    count = 0
    for row in reader:
        control_wards = int(row['control_wards'])
        if control_wards >= 15:
            count += 1
    
    return count

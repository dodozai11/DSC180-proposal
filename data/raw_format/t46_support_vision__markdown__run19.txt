def q1(text):
    lines = text.strip().split('\n')
    header = lines[0].split('|')[1:-1]
    header = [h.strip() for h in header]
    
    wins_vision = []
    losses_vision = []
    
    for line in lines[2:]:
        if line.strip() == '':
            continue
        parts = line.split('|')[1:-1]
        parts = [p.strip() for p in parts]
        
        player, team, result, vision_score, wards_placed, control_wards = parts
        vision_score = float(vision_score)
        
        if result == 'win':
            wins_vision.append(vision_score)
        else:
            losses_vision.append(vision_score)
    
    avg_wins = sum(wins_vision) / len(wins_vision) if wins_vision else 0
    avg_losses = sum(losses_vision) / len(losses_vision) if losses_vision else 0
    
    return avg_wins - avg_losses


def q2(text):
    lines = text.strip().split('\n')
    
    max_wards = -1
    max_player = None
    
    for line in lines[2:]:
        if line.strip() == '':
            continue
        parts = line.split('|')[1:-1]
        parts = [p.strip() for p in parts]
        
        player, team, result, vision_score, wards_placed, control_wards = parts
        wards_placed = int(wards_placed)
        
        if wards_placed > max_wards:
            max_wards = wards_placed
            max_player = player
    
    return max_player


def q3(text):
    lines = text.strip().split('\n')
    
    count = 0

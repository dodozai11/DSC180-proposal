import re

def q1(text):
    """Return the average damage per minute of players who lost."""
    pattern = r'(\w+(?:\s+\w+)*)\s+of\s+(.+?)\s+played\s+(.+?)\s+in\s+the\s+top\s+lane\s+and\s+(won|lost),\s+dealing\s+([\d.]+)\s+damage\s+per\s+minute\s+and\s+taking\s+([\d.]+)\s+damage\s+per\s+minute'
    
    matches = re.findall(pattern, text)
    
    lost_damages = []
    for match in matches:
        player_name, team, champion, result, dpm, damage_taken = match
        if result == "lost":
            lost_damages.append(float(dpm))
    
    if lost_damages:
        return sum(lost_damages) / len(lost_damages)
    return 0


def q2(text):
    """Return the name of the player who took the most damage per minute."""
    pattern = r'(\w+(?:\s+\w+)*)\s+of\s+(.+?)\s+played\s+(.+?)\s+in\s+the\s+top\s+lane\s+and\s+(won|lost),\s+dealing\s+([\d.]+)\s+damage\s+per\s+minute\s+and\s+taking\s+([\d.]+)\s+damage\s+per\s+minute'
    
    matches = re.findall(pattern, text)
    
    max_damage_taken = -1
    max_player = None
    
    for match in matches:
        player_name, team, champion, result, dpm, damage_taken = match
        damage_taken_float = float(damage_taken)
        if damage_taken_float > max_damage_taken:
            max_damage_taken = damage_taken_float
            max_player = player_name
    
    return max_player


def q3(text):
    """Return the number of different teams."""
    pattern = r'(\w+(?:\s+\w+)*)\s+of\s+(.+?)\s+played\s+(.+?)\s+in\s+the\s+top\s+lane\s+and\s+(won|lost),\s+dealing\s+([\d.]+)\s+damage\s+per\s+minute\s+and\s+taking\s+([\d.]+)\s+damage\s+per\s+minute'
    
    matches = re.findall(pattern, text)
    
    teams = set()
    for match in matches:
        player_name, team, champion, result, dpm, damage_taken = match
        teams.add(team)
    
    return len(teams)

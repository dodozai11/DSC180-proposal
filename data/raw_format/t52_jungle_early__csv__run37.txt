def q1(text):
    lines = text.strip().split('\n')
    header = lines[0].split(',')
    
    cs_diff_10_idx = header.index('cs_diff_10')
    result_idx = header.index('result')
    
    positive_cs_diff_count = 0
    positive_cs_diff_wins = 0
    
    for line in lines[1:]:
        values = line.split(',')
        cs_diff_10 = int(values[cs_diff_10_idx])
        result = values[result_idx]
        
        if cs_diff_10 > 0:
            positive_cs_diff_count += 1
            if result == 'win':
                positive_cs_diff_wins += 1
    
    if positive_cs_diff_count == 0:
        return 0
    
    return positive_cs_diff_wins / positive_cs_diff_count


def q2(text):
    lines = text.strip().split('\n')
    header = lines[0].split(',')
    
    kills_at_10_idx = header.index('kills_at_10')
    
    total_kills = 0
    for line in lines[1:]:
        values = line.split(',')
        kills_at_10 = int(values[kills_at_10_idx])
        total_kills += kills_at_10
    
    return total_kills


def q3(text):
    lines = text.strip().split('\n')
    header = lines[0].split(',')
    
    champion_idx = header.index('champion')
    assists_at_10_idx = header.index('assists_at_10')
    
    champion_assists = {}
    for line in lines[1:]:
        values = line.split(',')
        champion = values[champion_idx]
        assists_at_10 = int(values[assists_at_10_idx])
        
        if champion not in champion_assists:
            champion_assists[champion] = 0
        champion_assists[champion] += assists_at_10
    
    return max(champion_assists, key=champion_assists.get)

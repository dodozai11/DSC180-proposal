import xml.etree.ElementTree as ET
from collections import Counter

def q1(text):
    """Returns the name of the player with the highest KDA."""
    root = ET.fromstring(f"<root>{text}</root>")
    
    best_player = None
    best_kda = -1
    
    for row in root.findall('row'):
        player = row.find('player').text
        kills = int(row.find('kills').text)
        deaths = int(row.find('deaths').text)
        assists = int(row.find('assists').text)
        
        kda = (kills + assists) / max(deaths, 1)
        
        if kda > best_kda:
            best_kda = kda
            best_player = player
    
    return best_player


def q2(text):
    """Returns the average number of kills in games that were won."""
    root = ET.fromstring(f"<root>{text}</root>")
    
    total_kills = 0
    win_count = 0
    
    for row in root.findall('row'):
        result = row.find('result').text
        if result == 'win':
            kills = int(row.find('kills').text)
            total_kills += kills
            win_count += 1
    
    if win_count == 0:
        return 0
    
    return total_kills / win_count


def q3(text):
    """Returns the name of the most-picked champion."""
    root = ET.fromstring(f"<root>{text}</root>")
    
    champions = []
    
    for row in root.findall('row'):
        champion = row.find('champion').text
        champions.append(champion)
    
    champion_counts = Counter(champions)
    most_picked = champion_counts.most_common(1)[0][0]
    
    return most_picked

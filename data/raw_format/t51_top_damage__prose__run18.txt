import re

def q1(text):
    """Return the average damage per minute of players who lost."""
    pattern = r'(\w+(?:\s+\w+)*)\s+of\s+([^a-z]+?)\s+played\s+\S+\s+in\s+the\s+top\s+lane\s+and\s+(won|lost),\s+dealing\s+([\d.]+)\s+damage\s+per\s+minute'
    
    matches = re.findall(pattern, text)
    
    lost_damages = []
    for player_name, team, result, damage_per_min in matches:
        if result == "lost":
            lost_damages.append(float(damage_per_min))
    
    if not lost_damages:
        return 0
    
    return sum(lost_damages) / len(lost_damages)


def q2(text):
    """Return the name of the player who took the most damage per minute."""
    pattern = r'(\w+(?:\s+\w+)*)\s+of\s+([^a-z]+?)\s+played\s+\S+\s+in\s+the\s+top\s+lane\s+and\s+(won|lost),\s+dealing\s+([\d.]+)\s+damage\s+per\s+minute\s+and\s+taking\s+([\d.]+)\s+damage\s+per\s+minute'
    
    matches = re.findall(pattern, text)
    
    max_damage_taken = -1
    max_player = None
    
    for player_name, team, result, damage_per_min, damage_taken in matches:
        damage_taken_float = float(damage_taken)
        if damage_taken_float > max_damage_taken:
            max_damage_taken = damage_taken_float
            max_player = player_name
    
    return max_player


def q3(text):
    """Return the number of different teams."""
    pattern = r'(\w+(?:\s+\w+)*)\s+of\s+([^a-z]+?)\s+played'
    
    matches = re.findall(pattern, text)
    
    teams = set()
    for player_name, team in matches:
        teams.add(team.strip())
    
    return len(teams)

def q3(text):
    """Returns the game label of the longest game."""
    lines = text.strip().split('\n')
    
    longest_game = None
    longest_duration = -1
    
    for line in lines[2:]:
        if line.strip():
            cells = line.split('|')[1:-1]
            cells = [c.strip() for c in cells]
            
            game, team, result, game_minutes, inhibitors, tower_damage = cells
            
            duration = float(game_minutes)
            if duration > longest_duration:
                longest_duration = duration
                longest_game = game
    
    return longest_game

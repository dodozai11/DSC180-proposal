```python
import re

def q1(text):
    """Returns the average number of inhibitors destroyed by winning teams."""
    winning_inhibitors = []
    
    # Find all game records
    pattern = r'In game (\w+), (\w+(?:\s+\w+)*) (won|lost) after ([\d.]+) minutes, destroying (\d+) inhibitors'
    matches = re.findall(pattern, text)
    
    for game_id, team, result, duration, inhibitors in matches:
        if result == "won":
            winning_inhibitors.append(int(inhibitors))
    
    if not winning_inhibitors:
        return 0
    return sum(winning_inhibitors) / len(winning_inhibitors)


def q2(text):
    """Returns the average damage to towers of losing teams."""
    losing_damage = []
    
    # Find all game records
    pattern = r'In game (\w+), (\w+(?:\s+\w+)*) (won|lost) after ([\d.]+) minutes, destroying (\d+) inhibitors and dealing (\d+) damage to towers'
    matches = re.findall(pattern, text)
    
    for game_id, team, result, duration, inhibitors, damage in matches:
        if result == "lost":
            losing_damage.append(int(damage))
    
    if not losing_damage:
        return 0
    return sum(losing_damage) / len(losing_damage)


def q3(text):
    """Returns the game label of the longest game."""
    max_duration = -1
    longest_game = None
    
    # Find all game records
    pattern = r'In game (\w+), (\w+(?:\s+\w+)*) (won|lost) after ([\d.]+) minutes'
    matches = re.findall(pattern, text)
    
    for game_id, team, result, duration in matches:
        duration_float = float(duration)
        if duration_float > max_duration:
            max_duration = duration_float
            longest_game = game_id
    
    return longest_game
```

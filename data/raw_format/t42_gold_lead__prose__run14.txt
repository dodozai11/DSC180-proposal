```python
import re

def parse_data(text):
    """Parse the text data into a list of game records."""
    pattern = r"In game (\w+), (.+?) played on the (\w+) side and (\w+), with a gold difference of ([-\d]+) at 15 minutes\."
    matches = re.findall(pattern, text)
    
    records = []
    for match in matches:
        game_id, team, side, result, gold_diff = match
        won = result == "won"
        records.append({
            "game_id": game_id,
            "team": team,
            "side": side,
            "won": won,
            "gold_diff": int(gold_diff)
        })
    
    return records

def q1(text):
    """Return the average gold difference at 15 minutes of the winning teams."""
    records = parse_data(text)
    winning_records = [r for r in records if r["won"]]
    
    if not winning_records:
        return 0
    
    total_gold_diff = sum(r["gold_diff"] for r in winning_records)
    return total_gold_diff / len(winning_records)

def q2(text):
    """Return how many teams won despite a negative gold difference at 15 minutes."""
    records = parse_data(text)
    return sum(1 for r in records if r["won"] and r["gold_diff"] < 0)

def q3(text):
    """Return the win rate of teams that were ahead in gold at 15 minutes."""
    records = parse_data(text)
    ahead_records = [r for r in records if r["gold_diff"] > 0]
    
    if not ahead_records:
        return 0
    
    wins = sum(1 for r in ahead_records if r["won"])
    return wins / len(ahead_records)
```

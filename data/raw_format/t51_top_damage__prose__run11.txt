import re
from collections import defaultdict

def q1(text):
    """Returns the average damage per minute of players who lost."""
    lost_damages = []
    
    # Pattern to match player data
    pattern = r"(\w+(?:\s+\w+)*?)\s+of\s+(.+?)\s+played\s+(.+?)\s+in\s+the\s+top\s+lane\s+and\s+(won|lost),\s+dealing\s+([\d.]+)\s+damage\s+per\s+minute\s+and\s+taking\s+([\d.]+)\s+damage\s+per\s+minute"
    
    matches = re.finditer(pattern, text)
    for match in matches:
        result = match.group(4)
        damage_dealt = float(match.group(5))
        
        if result == "lost":
            lost_damages.append(damage_dealt)
    
    return sum(lost_damages) / len(lost_damages) if lost_damages else 0

def q2(text):
    """Returns the name of the player who took the most damage per minute."""
    max_damage_taken = -1
    max_player = None
    
    # Pattern to match player data
    pattern = r"(\w+(?:\s+\w+)*?)\s+of\s+(.+?)\s+played\s+(.+?)\s+in\s+the\s+top\s+lane\s+and\s+(won|lost),\s+dealing\s+([\d.]+)\s+damage\s+per\s+minute\s+and\s+taking\s+([\d.]+)\s+damage\s+per\s+minute"
    
    matches = re.finditer(pattern, text)
    for match in matches:
        player_name = match.group(1)
        damage_taken = float(match.group(6))
        
        if damage_taken > max_damage_taken:
            max_damage_taken = damage_taken
            max_player = player_name
    
    return max_player

def q3(text):
    """Returns the number of different teams."""
    teams = set()
    
    # Pattern to match player data
    pattern = r"(\w+(?:\s+\w+)*?)\s+of\s+(.+?)\s+played\s+(.+?)\s+in\s+the\s+top\s+lane\s+and\s+(won|lost),\s+dealing\s+([\d.]+)\s+damage\s+per\s+minute\s+and\s+taking\s+([\d.]+)\s+damage\s+per\s+minute"
    
    matches = re.finditer(pattern, text)
    for match in matches:
        team_name = match.group(2)
        teams.add(team_name)
    
    return len(teams)

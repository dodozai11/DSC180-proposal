import xml.etree.ElementTree as ET

def q1(text):
    """Return the win rate of players with a positive CS difference at 10 minutes."""
    root = ET.fromstring(f"<root>{text}</root>")
    
    positive_cs_diff_rows = []
    for row in root.findall("row"):
        cs_diff_10 = int(row.find("cs_diff_10").text)
        if cs_diff_10 > 0:
            positive_cs_diff_rows.append(row)
    
    if not positive_cs_diff_rows:
        return 0
    
    wins = sum(1 for row in positive_cs_diff_rows if row.find("result").text == "win")
    return wins / len(positive_cs_diff_rows)


def q2(text):
    """Return the total number of kills at 10 minutes across all rows."""
    root = ET.fromstring(f"<root>{text}</root>")
    
    total_kills = 0
    for row in root.findall("row"):
        kills_at_10 = int(row.find("kills_at_10").text)
        total_kills += kills_at_10
    
    return total_kills


def q3(text):
    """Return the name of the champion with the most assists at 10 minutes in total."""
    root = ET.fromstring(f"<root>{text}</root>")
    
    champion_assists = {}
    for row in root.findall("row"):
        champion = row.find("champion").text
        assists_at_10 = int(row.find("assists_at_10").text)
        
        if champion not in champion_assists:
            champion_assists[champion] = 0
        champion_assists[champion] += assists_at_10
    
    return max(champion_assists, key=champion_assists.get)

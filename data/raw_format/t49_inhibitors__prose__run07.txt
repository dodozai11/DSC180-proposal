import re

def q1(text):
    """Returns the average number of inhibitors destroyed by winning teams."""
    winning_inhibitors = []
    
    # Find all game entries
    pattern = r'In game (\w+), (\w+(?:\s+\w+)*) (won|lost) after ([\d.]+) minutes, destroying (\d+) inhibitors'
    matches = re.findall(pattern, text)
    
    for game, team, result, duration, inhibitors in matches:
        if result == 'won':
            winning_inhibitors.append(int(inhibitors))
    
    return sum(winning_inhibitors) / len(winning_inhibitors) if winning_inhibitors else 0

def q2(text):
    """Returns the average damage to towers of losing teams."""
    losing_damage = []
    
    # Find all game entries
    pattern = r'In game (\w+), (\w+(?:\s+\w+)*) (won|lost) after ([\d.]+) minutes, destroying (\d+) inhibitors and dealing (\d+) damage to towers'
    matches = re.findall(pattern, text)
    
    for game, team, result, duration, inhibitors, damage in matches:
        if result == 'lost':
            losing_damage.append(int(damage))
    
    return sum(losing_damage) / len(losing_damage) if losing_damage else 0

def q3(text):
    """Returns the game label of the longest game."""
    longest_game = None
    longest_duration = -1
    
    # Find all game entries
    pattern = r'In game (\w+), (\w+(?:\s+\w+)*) (won|lost) after ([\d.]+) minutes'
    matches = re.findall(pattern, text)
    
    for game, team, result, duration in matches:
        duration_float = float(duration)
        if duration_float > longest_duration:
            longest_duration = duration_float
            longest_game = game
    
    return longest_game

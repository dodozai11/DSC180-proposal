def q1(text):
    """Return the win rate of teams that took the first tower."""
    lines = text.strip().split('\n')
    
    # Skip header line
    data_lines = lines[1:]
    
    first_tower_wins = 0
    first_tower_total = 0
    
    for line in data_lines:
        parts = line.split('|')
        # Remove empty strings from split and strip whitespace
        parts = [p.strip() for p in parts if p.strip()]
        
        if len(parts) < 5:
            continue
            
        first_tower = parts[2]
        result = parts[3]
        
        if first_tower == 'yes':
            first_tower_total += 1
            if result == 'win':
                first_tower_wins += 1
    
    if first_tower_total == 0:
        return 0
    
    return first_tower_wins / first_tower_total


def q2(text):
    """Return the average number of void grubs taken by winning teams."""
    lines = text.strip().split('\n')
    
    # Skip header line
    data_lines = lines[1:]
    
    void_grubs_sum = 0
    win_count = 0
    
    for line in data_lines:
        parts = line.split('|')
        # Remove empty strings from split and strip whitespace
        parts = [p.strip() for p in parts if p.strip()]
        
        if len(parts) < 5:
            continue
            
        result = parts[3]
        void_grubs = int(parts[4])
        
        if result == 'win':
            void_grubs_sum += void_grubs
            win_count += 1
    
    if win_count == 0:
        return 0
    
    return void_grubs_sum / win_count

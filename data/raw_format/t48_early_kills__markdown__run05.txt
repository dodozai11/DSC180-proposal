def q1(text):
    """Returns the average number of kills at 15 minutes of red-side teams."""
    lines = text.strip().split('\n')
    red_kills = []
    
    for line in lines[1:]:  # Skip header
        parts = [p.strip() for p in line.split('|')]
        if len(parts) >= 5:
            side = parts[2]
            kills_at_15 = parts[4]
            if side == 'red' and kills_at_15:
                red_kills.append(int(kills_at_15))
    
    return sum(red_kills) / len(red_kills) if red_kills else 0


def q2(text):
    """Returns the name of the team with the largest experience difference at 15 minutes."""
    lines = text.strip().split('\n')
    max_xp_diff = float('-inf')
    team_with_max_xp = None
    
    for line in lines[1:]:  # Skip header
        parts = [p.strip() for p in line.split('|')]
        if len(parts) >= 5:
            team = parts[1]
            xp_diff_15 = parts[5]
            if xp_diff_15:
                xp_diff = int(xp_diff_15)
                if abs(xp_diff) > abs(max_xp_diff):
                    max_xp_diff = xp_diff
                    team_with_max_xp = team
    
    return team_with_max_xp


def q3(text):
    """Returns how many teams won despite a negative experience difference at 15 minutes."""
    lines = text.strip().split('\n')
    count = 0
    
    for line in lines[1:]:  # Skip header
        parts = [p.strip() for p in line.split('|')]
        if len(parts) >= 5:
            result = parts[3]
            xp_diff_15 = parts[5]
            if result and xp_diff_15:
                if result == 'win' and int(xp_diff_15) < 0:
                    count += 1
    
    return count

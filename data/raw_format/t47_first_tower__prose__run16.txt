def q2(text):
    """Return the average number of void grubs taken by winning teams"""
    winning_void_grubs = []
    
    sentences = re.split(r'In game', text)
    
    for sentence in sentences[1:]:
        match = re.search(r'(\w+), ([^;]+) (won|lost);[^,]+, (\d+) void grubs', sentence)
        if match:
            game_id = match.group(1)
            team = match.group(2)
            result = match.group(3)
            void_grubs = int(match.group(4))
            
            if result == "won":
                winning_void_grubs.append(void_grubs)
    
    if len(winning_void_grubs) == 0:
        return 0
    return sum(winning_void_grubs) / len(winning_void_grubs)

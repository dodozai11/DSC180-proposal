```python
import re

def parse_data(text):
    """Parse the text data into a list of game records."""
    games = []
    
    # Pattern to match each game entry
    pattern = r'In game (G\d+), (.+?) (won|lost); they (took|did not take) the first tower and finished with (\d+) void grubs and (\d+) heralds\.'
    
    matches = re.findall(pattern, text)
    
    for match in matches:
        game_id, team_name, result, tower_status, void_grubs, heralds = match
        
        games.append({
            'game_id': game_id,
            'team_name': team_name,
            'won': result == 'won',
            'first_tower': tower_status == 'took',
            'void_grubs': int(void_grubs),
            'heralds': int(heralds)
        })
    
    return games


def q1(text):
    """Return the win rate of teams that took the first tower."""
    games = parse_data(text)
    
    teams_with_first_tower = [game for game in games if game['first_tower']]
    
    if not teams_with_first_tower:
        return 0
    
    wins = sum(1 for game in teams_with_first_tower if game['won'])
    win_rate = wins / len(teams_with_first_tower)
    
    return win_rate


def q2(text):
    """Return the average number of void grubs taken by winning teams."""
    games = parse_data(text)
    
    winning_teams = [game for game in games if game['won']]
    
    if not winning_teams:
        return 0
    
    total_void_grubs = sum(game['void_grubs'] for game in winning_teams)
    average = total_void_grubs / len(winning_teams)
    
    return average


def q3(text):
    """Return how many teams took at least one herald and still lost."""
    games = parse_data(text)
    
    teams_with_herald_and_lost = [
        game for game in games 
        if game['heralds'] >= 1 and not game['won']
    ]
    
    return len(teams_with_herald_and_lost)
```

def q1(text):
    """Returns the average number of inhibitors destroyed by winning teams."""
    lines = text.strip().split('\n')
    winning_inhibitors = []
    
    for line in lines[1:]:  # Skip header
        parts = line.split(',')
        result = parts[2]
        inhibitors = int(parts[4])
        
        if result == 'win':
            winning_inhibitors.append(inhibitors)
    
    return sum(winning_inhibitors) / len(winning_inhibitors) if winning_inhibitors else 0


def q2(text):
    """Returns the average damage to towers of losing teams."""
    lines = text.strip().split('\n')
    losing_tower_damage = []
    
    for line in lines[1:]:  # Skip header
        parts = line.split(',')
        result = parts[2]
        tower_damage = int(parts[5])
        
        if result == 'loss':
            losing_tower_damage.append(tower_damage)
    
    return sum(losing_tower_damage) / len(losing_tower_damage) if losing_tower_damage else 0


def q3(text):
    """Returns the game label of the longest game."""
    lines = text.strip().split('\n')
    longest_game = None
    longest_duration = -1
    
    for line in lines[1:]:  # Skip header
        parts = line.split(',')
        game = parts[0]
        game_minutes = float(parts[3])
        
        if game_minutes > longest_duration:
            longest_duration = game_minutes
            longest_game = game
    
    return longest_game

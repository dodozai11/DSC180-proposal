import re
from collections import Counter

def q1(text):
    """Returns the name of the player with the highest KDA."""
    rows = re.findall(r'<row>.*?</row>', text)
    
    max_kda = -1
    best_player = None
    
    for row in rows:
        player = re.search(r'<player>(.*?)</player>', row).group(1)
        kills = int(re.search(r'<kills>(.*?)</kills>', row).group(1))
        deaths = int(re.search(r'<deaths>(.*?)</deaths>', row).group(1))
        assists = int(re.search(r'<assists>(.*?)</assists>', row).group(1))
        
        kda = (kills + assists) / max(deaths, 1)
        
        if kda > max_kda:
            max_kda = kda
            best_player = player
    
    return best_player


def q2(text):
    """Returns the average number of kills in games that were won."""
    rows = re.findall(r'<row>.*?</row>', text)
    
    total_kills = 0
    win_count = 0
    
    for row in rows:
        result = re.search(r'<result>(.*?)</result>', row).group(1)
        if result == 'win':
            kills = int(re.search(r'<kills>(.*?)</kills>', row).group(1))
            total_kills += kills
            win_count += 1
    
    if win_count == 0:
        return 0
    
    return total_kills / win_count


def q3(text):
    """Returns the name of the most-picked champion."""
    rows = re.findall(r'<row>.*?</row>', text)
    
    champions = []
    
    for row in rows:
        champion = re.search(r'<champion>(.*?)</champion>', row).group(1)
        champions.append(champion)
    
    champion_counts = Counter(champions)
    most_picked = champion_counts.most_common(1)[0][0]
    
    return most_picked

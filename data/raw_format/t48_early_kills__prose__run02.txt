```python
import re

def q1(text):
    """Return the average number of kills at 15 minutes of red-side teams."""
    red_kills = []
    
    # Find all matches for red side teams
    pattern = r'In game \w+, (.+?) played on the red side and \w+, with (\d+) kills'
    matches = re.findall(pattern, text)
    
    for match in matches:
        kills = int(match[1])
        red_kills.append(kills)
    
    if red_kills:
        return sum(red_kills) / len(red_kills)
    return 0

def q2(text):
    """Return the name of the team with the largest experience difference at 15 minutes."""
    teams_exp_diff = []
    
    # Find all team records with experience difference
    pattern = r'In game \w+, (.+?) played on the (?:blue|red) side and \w+, with \d+ kills and an experience difference of (-?\d+) at 15 minutes'
    matches = re.findall(pattern, text)
    
    for team_name, exp_diff in matches:
        teams_exp_diff.append((team_name, int(exp_diff)))
    
    if teams_exp_diff:
        return max(teams_exp_diff, key=lambda x: x[1])[0]
    return None

def q3(text):
    """Return how many teams won despite a negative experience difference at 15 minutes."""
    count = 0
    
    # Find all team records with outcome and experience difference
    pattern = r'In game \w+, (.+?) played on the (?:blue|red) side and (won|lost), with \d+ kills and an experience difference of (-?\d+) at 15 minutes'
    matches = re.findall(pattern, text)
    
    for team_name, outcome, exp_diff in matches:
        if outcome == 'won' and int(exp_diff) < 0:
            count += 1
    
    return count
```

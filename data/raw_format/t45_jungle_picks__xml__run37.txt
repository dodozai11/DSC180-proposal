import re
from collections import defaultdict

def q1(text):
    """Return the number of different champions played."""
    champions = set(re.findall(r'<champion>([^<]+)</champion>', text))
    return len(champions)

def q2(text):
    """Return the name of the champion with the best win rate among champions played at least twice."""
    # Extract all rows
    rows = re.findall(
        r'<player>([^<]+)</player><champion>([^<]+)</champion><result>([^<]+)</result><deaths>(\d+)</deaths>',
        text
    )
    
    # Count wins and total games for each champion
    champion_stats = defaultdict(lambda: {'wins': 0, 'total': 0})
    
    for player, champion, result, deaths in rows:
        champion_stats[champion]['total'] += 1
        if result == 'win':
            champion_stats[champion]['wins'] += 1
    
    # Filter champions played at least twice
    eligible_champions = {
        champ: stats for champ, stats in champion_stats.items()
        if stats['total'] >= 2
    }
    
    if not eligible_champions:
        return None
    
    # Find champion with best win rate
    best_champion = max(
        eligible_champions.items(),
        key=lambda x: x[1]['wins'] / x[1]['total']
    )
    
    return best_champion[0]

def q3(text):
    """Return the average number of deaths in games that were lost."""
    # Extract all rows
    rows = re.findall(
        r'<player>([^<]+)</player><champion>([^<]+)</champion><result>([^<]+)</result><deaths>(\d+)</deaths>',
        text
    )
    
    # Filter losses and collect death counts
    losses = [int(deaths) for player, champion, result, deaths in rows if result == 'loss']
    
    if not losses:
        return 0
    
    return sum(losses) / len(losses)

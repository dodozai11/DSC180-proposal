```python
import re
from io import StringIO
import pandas as pd

def parse_data(text):
    """Parse the markdown table format into a DataFrame"""
    # Remove leading/trailing whitespace and split by lines
    lines = text.strip().split('\n')
    
    # Filter out separator lines (lines with dashes)
    data_lines = [line for line in lines if not re.match(r'^\|?\s*-+', line)]
    
    # Parse header
    header_line = data_lines[0]
    headers = [h.strip() for h in header_line.split('|')[1:-1]]
    
    # Parse data rows
    rows = []
    for line in data_lines[1:]:
        values = [v.strip() for v in line.split('|')[1:-1]]
        if len(values) == len(headers):
            row = {
                'game': values[0],
                'team': values[1],
                'side': values[2],
                'result': values[3],
                'gold_diff_15': int(values[4])
            }
            rows.append(row)
    
    return pd.DataFrame(rows)

def q1(text):
    """Return the average gold difference at 15 minutes of the winning teams."""
    df = parse_data(text)
    winning_teams = df[df['result'] == 'win']
    return winning_teams['gold_diff_15'].mean()

def q2(text):
    """Return how many teams won despite a negative gold difference at 15 minutes."""
    df = parse_data(text)
    won_with_negative_gold = df[(df['result'] == 'win') & (df['gold_diff_15'] < 0)]
    return len(won_with_negative_gold)

def q3(text):
    """Return the win rate of teams that were ahead in gold at 15 minutes."""
    df = parse_data(text)
    ahead_in_gold = df[df['gold_diff_15'] > 0]
    if len(ahead_in_gold) == 0:
        return 0.0
    wins_while_ahead = len(ahead_in_gold[ahead_in_gold['result'] == 'win'])
    return wins_while_ahead / len(ahead_in_gold)
```

```python
import re

def q1(text):
    """Returns the name of the player with the highest CS per minute."""
    pattern = r'(\w+(?:\s+\w+)*) of .+ played .+ in the bot lane and (?:won|lost), with ([\d.]+) CS per minute'
    matches = re.findall(pattern, text)
    
    if not matches:
        return None
    
    max_player = max(matches, key=lambda x: float(x[1]))
    return max_player[0]


def q2(text):
    """Returns the average gold share of players who won."""
    pattern = r'(\w+(?:\s+\w+)*) of .+ played .+ in the bot lane and (won|lost), with [\d.]+ CS per minute and a gold share of ([\d.]+)'
    matches = re.findall(pattern, text)
    
    if not matches:
        return None
    
    winning_gold_shares = [float(gold_share) for _, result, gold_share in matches if result == 'won']
    
    if not winning_gold_shares:
        return None
    
    return sum(winning_gold_shares) / len(winning_gold_shares)


def q3(text):
    """Returns how many players had more than 9 CS per minute."""
    pattern = r'(\w+(?:\s+\w+)*) of .+ played .+ in the bot lane and (?:won|lost), with ([\d.]+) CS per minute'
    matches = re.findall(pattern, text)
    
    if not matches:
        return 0
    
    count = sum(1 for _, cs_per_min in matches if float(cs_per_min) > 9)
    return count
```

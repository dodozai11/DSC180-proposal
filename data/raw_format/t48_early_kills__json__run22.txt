import json
import re

def q1(text):
    """Return the average number of kills at 15 minutes of red-side teams."""
    data = json.loads(text)
    red_side_kills = [row["kills_at_15"] for row in data if row["side"] == "red"]
    return sum(red_side_kills) / len(red_side_kills)

def q2(text):
    """Return the name of the team with the largest experience difference at 15 minutes."""
    data = json.loads(text)
    max_xp_diff = float('-inf')
    team_with_max_xp = None
    
    for row in data:
        if abs(row["xp_diff_15"]) > abs(max_xp_diff):
            max_xp_diff = row["xp_diff_15"]
            team_with_max_xp = row["team"]
    
    return team_with_max_xp

def q3(text):
    """Return how many teams won despite a negative experience difference at 15 minutes."""
    data = json.loads(text)
    count = 0
    
    for row in data:
        if row["result"] == "win" and row["xp_diff_15"] < 0:
            count += 1
    
    return count

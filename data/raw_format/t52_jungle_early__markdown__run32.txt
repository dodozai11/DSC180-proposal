            result = parts[2]
            cs_diff_10 = int(parts[3])

def q2(text):
    """Return the total number of kills at 10 minutes across all rows."""
    lines = text.strip().split('\n')
    
    # Skip header and separator lines
    data_lines = [line for line in lines[2:] if line.strip() and '|' in line]
    
    total_kills = 0
    
    for line in data_lines:
        parts = [p.strip() for p in line.split('|')[1:-1]]  # Remove empty first/last elements
        
        if len(parts) >= 5:
            kills_at_10 = int(parts[4])
            total_kills += kills_at_10
    
    return total_kills


def q3(text):
    """Return the name of the champion with the most assists at 10 minutes in total."""
    lines = text.strip().split('\n')
    
    # Skip header and separator lines
    data_lines = [line for line in lines[2:] if line.strip() and '|' in line]
    
    champion_assists = {}
    
    for line in data_lines:
        parts = [p.strip() for p in line.split('|')[1:-1]]  # Remove empty first/last elements
        
        if len(parts) >= 6:
            champion = parts[1]
            assists_at_10 = int(parts[5])
            
            if champion not in champion_assists:
                champion_assists[champion] = 0
            champion_assists[champion] += assists_at_10
    
    if not champion_assists:
        return None
    
    return max(champion_assists, key=champion_assists.get)

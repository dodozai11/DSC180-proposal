def q1(text):
    lines = text.strip().split('\n')
    header = lines[0].split('|')[1:-1]
    header = [h.strip() for h in header]
    
    max_cs = -1
    max_player = None
    
    for line in lines[2:]:
        if line.strip():
            parts = line.split('|')[1:-1]
            parts = [p.strip() for p in parts]
            
            player = parts[0]
            cs_per_min = float(parts[4])
            
            if cs_per_min > max_cs:
                max_cs = cs_per_min
                max_player = player
    
    return max_player


def q2(text):
    lines = text.strip().split('\n')
    
    total_gold = 0
    win_count = 0
    
    for line in lines[2:]:
        if line.strip():
            parts = line.split('|')[1:-1]
            parts = [p.strip() for p in parts]
            
            result = parts[3]
            gold_share = float(parts[5])
            
            if result == 'win':
                total_gold += gold_share
                win_count += 1
    
    return total_gold / win_count if win_count > 0 else 0


def q3(text):
    lines = text.strip().split('\n')
    
    count = 0
    
    for line in lines[2:]:
        if line.strip():
            parts = line.split('|')[1:-1]
            parts = [p.strip() for p in parts]
            
            cs_per_min = float(parts[4])
            
            if cs_per_min > 9:
                count += 1
    
    return count

def q1(text):
    """Return the name of the player with the highest KDA."""
    lines = text.strip().split('\n')
    header = lines[0].split(',')
    
    # Find column indices
    player_idx = header.index('player')
    kills_idx = header.index('kills')
    deaths_idx = header.index('deaths')
    assists_idx = header.index('assists')
    
    max_kda = -1
    best_player = None
    
    for line in lines[1:]:
        parts = line.split(',')
        player = parts[player_idx]
        kills = int(parts[kills_idx])
        deaths = int(parts[deaths_idx])
        assists = int(parts[assists_idx])
        
        kda = (kills + assists) / max(deaths, 1)
        
        if kda > max_kda:
            max_kda = kda
            best_player = player
    
    return best_player


def q2(text):
    """Return the average number of kills in games that were won."""
    lines = text.strip().split('\n')
    header = lines[0].split(',')
    
    # Find column indices
    kills_idx = header.index('kills')
    result_idx = header.index('result')
    
    total_kills = 0
    win_count = 0
    
    for line in lines[1:]:
        parts = line.split(',')
        result = parts[result_idx]
        
        if result == 'win':
            kills = int(parts[kills_idx])
            total_kills += kills
            win_count += 1
    
    if win_count == 0:
        return 0
    
    return total_kills / win_count


def q3(text):
    """Return the name of the most-picked champion."""
    lines = text.strip().split('\n')
    header = lines[0].split(',')
    
    # Find column index
    champion_idx = header.index('champion')
    
    champion_counts = {}
    
    for line in lines[1:]:
        parts = line.split(',')
        champion = parts[champion_idx]
        
        if champion in champion_counts:
            champion_counts[champion] += 1
        else:
            champion_counts[champion] = 1
    
    most_picked = max(champion_counts, key=champion_counts.get)
    
    return most_picked

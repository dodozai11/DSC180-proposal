import re

def parse_data(text):
    """Parse the text data into a list of dictionaries."""
    pattern = r"In game (G\d+), (.+?) played on the (blue|red) side and (won|lost), with a gold difference of ([-\d]+) at 15 minutes\."
    matches = re.findall(pattern, text)
    
    data = []
    for match in matches:
        game_id, team, side, result, gold_diff = match
        data.append({
            'game_id': game_id,
            'team': team,
            'side': side,
            'result': result,
            'gold_diff': int(gold_diff)
        })
    
    return data

def q1(text):
    """Return the average gold difference at 15 minutes of the winning teams."""
    data = parse_data(text)
    winning_teams = [entry['gold_diff'] for entry in data if entry['result'] == 'won']
    
    if not winning_teams:
        return 0
    
    return sum(winning_teams) / len(winning_teams)

def q2(text):
    """Return how many teams won despite a negative gold difference at 15 minutes."""
    data = parse_data(text)
    count = sum(1 for entry in data if entry['result'] == 'won' and entry['gold_diff'] < 0)
    
    return count

def q3(text):
    """Return the win rate of teams that were ahead in gold at 15 minutes."""
    data = parse_data(text)
    
    teams_ahead = [entry for entry in data if entry['gold_diff'] > 0]
    
    if not teams_ahead:
        return 0
    
    wins = sum(1 for entry in teams_ahead if entry['result'] == 'won')
    
    return wins / len(teams_ahead)

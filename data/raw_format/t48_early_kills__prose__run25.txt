```python
import re

def q1(text):
    """Returns the average number of kills at 15 minutes of red-side teams."""
    red_kills = []
    
    # Find all lines with "played on the red side"
    pattern = r'played on the red side.*?with (\d+) kills'
    matches = re.findall(pattern, text)
    
    for match in matches:
        red_kills.append(int(match))
    
    if len(red_kills) == 0:
        return 0
    
    return sum(red_kills) / len(red_kills)


def q2(text):
    """Returns the name of the team with the largest experience difference at 15 minutes."""
    max_exp_diff = float('-inf')
    team_with_max = None
    
    # Find all team entries with their experience difference
    pattern = r'(\w+(?:\s+\w+)*) played on the (?:blue|red) side.*?and (?:won|lost), with \d+ kills and an experience difference of (-?\d+)'
    matches = re.findall(pattern, text)
    
    for team, exp_diff in matches:
        exp_diff = int(exp_diff)
        if exp_diff > max_exp_diff:
            max_exp_diff = exp_diff
            team_with_max = team
    
    return team_with_max


def q3(text):
    """Returns how many teams won despite a negative experience difference at 15 minutes."""
    count = 0
    
    # Find all teams that won with negative experience difference
    pattern = r'.*?and won, with \d+ kills and an experience difference of (-?\d+)'
    matches = re.findall(pattern, text)
    
    for exp_diff in matches:
        exp_diff = int(exp_diff)
        if exp_diff < 0:
            count += 1
    
    return count
```

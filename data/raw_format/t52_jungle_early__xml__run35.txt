```python
import re
from collections import defaultdict

def q1(text):
    """Return the win rate of players with a positive CS difference at 10 minutes."""
    rows = re.findall(r'<row>.*?</row>', text)
    
    positive_cs_wins = 0
    positive_cs_total = 0
    
    for row in rows:
        cs_diff = int(re.search(r'<cs_diff_10>(-?\d+)</cs_diff_10>', row).group(1))
        result = re.search(r'<result>(\w+)</result>', row).group(1)
        
        if cs_diff > 0:
            positive_cs_total += 1
            if result == 'win':
                positive_cs_wins += 1
    
    if positive_cs_total == 0:
        return 0
    
    return positive_cs_wins / positive_cs_total


def q2(text):
    """Return the total number of kills at 10 minutes across all rows."""
    rows = re.findall(r'<row>.*?</row>', text)
    
    total_kills = 0
    
    for row in rows:
        kills = int(re.search(r'<kills_at_10>(\d+)</kills_at_10>', row).group(1))
        total_kills += kills
    
    return total_kills


def q3(text):
    """Return the name of the champion with the most assists at 10 minutes in total."""
    rows = re.findall(r'<row>.*?</row>', text)
    
    champion_assists = defaultdict(int)
    
    for row in rows:
        champion = re.search(r'<champion>([^<]+)</champion>', row).group(1)
        assists = int(re.search(r'<assists_at_10>(\d+)</assists_at_10>', row).group(1))
        champion_assists[champion] += assists
    
    return max(champion_assists, key=champion_assists.get)
```

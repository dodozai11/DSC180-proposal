import re

def q1(text):
    """Return the win rate of teams that took the first tower."""
    first_tower_teams = []
    
    # Parse all games
    game_pattern = r"In game (\w+), (.+?) (won|lost); they (did not take|took) the first tower"
    matches = re.findall(game_pattern, text)
    
    for game, team, result, tower_status in matches:
        took_first_tower = tower_status == "took"
        won = result == "won"
        
        if took_first_tower:
            first_tower_teams.append(won)
    
    if len(first_tower_teams) == 0:
        return 0
    
    win_rate = sum(first_tower_teams) / len(first_tower_teams)
    return win_rate


def q2(text):
    """Return the average number of void grubs taken by winning teams."""
    winning_void_grubs = []
    
    # Parse all games with void grubs and heralds
    game_pattern = r"In game (\w+), (.+?) (won|lost); they (did not take|took) the first tower and finished with (\d+) void grubs and (\d+) heralds"
    matches = re.findall(game_pattern, text)
    
    for game, team, result, tower_status, void_grubs, heralds in matches:
        won = result == "won"
        void_grubs_count = int(void_grubs)
        
        if won:
            winning_void_grubs.append(void_grubs_count)
    
    if len(winning_void_grubs) == 0:
        return 0
    
    avg_void_grubs = sum(winning_void_grubs) / len(winning_void_grubs)
    return avg_void_grubs

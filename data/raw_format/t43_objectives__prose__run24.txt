import re

def q1(text):
    """Return the win rate of teams that took the first dragon."""
    first_dragon_games = []
    
    # Parse all games
    game_pattern = r'In game (G\d+), (.+?) (won|lost); they (took|did not take) the first dragon and finished with (\d+) dragons, (\d+) barons and (\d+) towers\.'
    matches = re.findall(game_pattern, text)
    
    for match in matches:
        game_id, team_name, result, first_dragon, dragons, barons, towers = match
        took_first_dragon = (first_dragon == "took")
        won = (result == "won")
        
        if took_first_dragon:
            first_dragon_games.append(won)
    
    if len(first_dragon_games) == 0:
        return 0
    
    win_rate = sum(first_dragon_games) / len(first_dragon_games)
    return win_rate


def q2(text):
    """Return the average number of barons taken by winning teams."""
    winning_barons = []

    matches = re.findall(game_pattern, text)
    
    for match in matches:
        game_id, team_name, result, first_dragon, dragons, barons, towers = match
        won = (result == "won")
        
        if won:
            winning_barons.append(int(barons))
    
    if len(winning_barons) == 0:
        return 0
    
    average_barons = sum(winning_barons) / len(winning_barons)
    return average_barons

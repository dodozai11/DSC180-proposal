def q1(text):
    """Return the name of the player with the highest CS per minute."""
    lines = text.strip().split('\n')
    # Skip header line
    max_cs = -1
    max_player = None
    
    for line in lines[1:]:
        if not line.strip():
            continue
        parts = line.split('|')
        parts = [p.strip() for p in parts]
        # parts[0] is empty, parts[1] is player, parts[5] is cs_per_min
        player = parts[1]
        cs_per_min = float(parts[5])
        
        if cs_per_min > max_cs:
            max_cs = cs_per_min
            max_player = player
    
    return max_player


def q2(text):
    """Return the average gold share of players who won."""
    lines = text.strip().split('\n')
    # Skip header line
    total_gold_share = 0
    win_count = 0
    
    for line in lines[1:]:
        if not line.strip():
            continue
        parts = line.split('|')
        parts = [p.strip() for p in parts]
        # parts[0] is empty, parts[4] is result, parts[6] is gold_share
        result = parts[4]
        gold_share = float(parts[6])
        
        if result == 'win':
            total_gold_share += gold_share
            win_count += 1
    
    return total_gold_share / win_count if win_count > 0 else 0


def q3(text):
    """Return how many players had more than 9 CS per minute."""
    lines = text.strip().split('\n')
    # Skip header line
    count = 0
    
    for line in lines[1:]:
        if not line.strip():
            continue
        parts = line.split('|')
        parts = [p.strip() for p in parts]
        # parts[0] is empty, parts[5] is cs_per_min
        cs_per_min = float(parts[5])
        
        if cs_per_min > 9:
            count += 1
    
    return count

def q1(text):
    lines = text.strip().split('\n')
    headers = lines[0].split(',')
    
    vision_score_idx = headers.index('vision_score')
    result_idx = headers.index('result')
    
    win_scores = []
    loss_scores = []
    
    for line in lines[1:]:
        parts = line.split(',')
        vision_score = int(parts[vision_score_idx])
        result = parts[result_idx]
        
        if result == 'win':
            win_scores.append(vision_score)
        else:
            loss_scores.append(vision_score)
    
    avg_win = sum(win_scores) / len(win_scores)
    avg_loss = sum(loss_scores) / len(loss_scores)
    
    return avg_win - avg_loss


def q2(text):
    lines = text.strip().split('\n')
    headers = lines[0].split(',')
    
    player_idx = headers.index('player')
    wards_placed_idx = headers.index('wards_placed')
    
    max_wards = -1
    max_player = None
    
    for line in lines[1:]:
        parts = line.split(',')
        player = parts[player_idx]
        wards_placed = int(parts[wards_placed_idx])
        
        if wards_placed > max_wards:
            max_wards = wards_placed
            max_player = player
    
    return max_player


def q3(text):
    lines = text.strip().split('\n')
    headers = lines[0].split(',')
    
    control_wards_idx = headers.index('control_wards')
    
    count = 0
    
    for line in lines[1:]:
        parts = line.split(',')
        control_wards = int(parts[control_wards_idx])
        
        if control_wards >= 15:
            count += 1
    
    return count

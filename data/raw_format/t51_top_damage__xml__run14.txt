import xml.etree.ElementTree as ET

def q1(text):
    """Return the average damage per minute of players who lost."""
    root = ET.fromstring(f"<root>{text}</root>")
    losses = []
    for row in root.findall('row'):
        if row.find('result').text == 'loss':
            dpm = float(row.find('damage_per_min').text)
            losses.append(dpm)
    return sum(losses) / len(losses) if losses else 0

def q2(text):
    """Return the name of the player who took the most damage per minute."""
    root = ET.fromstring(f"<root>{text}</root>")
    max_player = None
    max_damage = -1
    for row in root.findall('row'):
        dtpm = float(row.find('damage_taken_per_min').text)
        if dtpm > max_damage:
            max_damage = dtpm
            max_player = row.find('player').text
    return max_player

def q3(text):
    """Return the number of different teams."""
    root = ET.fromstring(f"<root>{text}</root>")
    teams = set()
    for row in root.findall('row'):
        team = row.find('team').text
        teams.add(team)
    return len(teams)

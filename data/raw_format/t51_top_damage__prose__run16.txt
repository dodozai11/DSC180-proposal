```python
import re

def q1(text):
    """Return the average damage per minute of players who lost."""
    matches = re.findall(
        r'(\w+(?:\s+\w+)*)\s+of\s+(.+?)\s+played\s+.+?\s+and\s+(won|lost),\s+dealing\s+([\d.]+)\s+damage per minute',
        text
    )
    
    lost_damages = [float(damage) for player, team, result, damage in matches if result == 'lost']
    
    if not lost_damages:
        return 0
    
    return sum(lost_damages) / len(lost_damages)


def q2(text):
    """Return the name of the player who took the most damage per minute."""
    matches = re.findall(
        r'(\w+(?:\s+\w+)*)\s+of\s+(.+?)\s+played\s+.+?\s+and\s+(?:won|lost),\s+dealing\s+[\d.]+\s+damage per minute\s+and\s+taking\s+([\d.]+)\s+damage per minute',
        text
    )
    
    if not matches:
        return None
    
    max_player = max(matches, key=lambda x: float(x[1]))
    return max_player[0]


def q3(text):
    """Return the number of different teams."""
    matches = re.findall(
        r'\w+(?:\s+\w+)*\s+of\s+(.+?)\s+played',
        text
    )
    
    teams = set(matches)
    return len(teams)
```

import json
import re

def q1(text):
    """Return the win rate of blue-side teams."""
    data = json.loads(text)
    blue_games = [game for game in data if game["side"] == "blue"]
    blue_wins = [game for game in blue_games if game["result"] == "win"]
    if len(blue_games) == 0:
        return 0
    return len(blue_wins) / len(blue_games)

def q2(text):
    """Return the average length in minutes of games won by red-side team."""
    data = json.loads(text)
    red_wins = [game for game in data if game["side"] == "red" and game["result"] == "win"]
    if len(red_wins) == 0:
        return 0
    return sum(game["game_minutes"] for game in red_wins) / len(red_wins)

def q3(text):
    """Return the name of the team with the most wins."""
    data = json.loads(text)
    wins_by_team = {}
    for game in data:
        if game["result"] == "win":
            team = game["team"]
            wins_by_team[team] = wins_by_team.get(team, 0) + 1
    if not wins_by_team:
        return None
    return max(wins_by_team, key=wins_by_team.get)

def q1(text):
    lines = text.strip().split('\n')
    header = lines[0].split('|')[1:-1]
    header = [h.strip() for h in header]
    
    max_kda = -1
    best_player = None
    
    for line in lines[2:]:
        if line.strip() == '':
            continue
        cells = line.split('|')[1:-1]
        cells = [c.strip() for c in cells]
        
        player = cells[0]
        kills = int(cells[3])
        deaths = int(cells[4])
        assists = int(cells[5])
        
        kda = (kills + assists) / max(deaths, 1)
        
        if kda > max_kda:
            max_kda = kda
            best_player = player
    
    return best_player


def q2(text):
    lines = text.strip().split('\n')
    
    total_kills = 0
    win_count = 0
    
    for line in lines[2:]:
        if line.strip() == '':
            continue
        cells = line.split('|')[1:-1]
        cells = [c.strip() for c in cells]
        
        kills = int(cells[3])
        result = cells[6]
        
        if result == 'win':
            total_kills += kills
            win_count += 1
    
    if win_count == 0:
        return 0
    
    return total_kills / win_count


def q3(text):
    lines = text.strip().split('\n')
    
    champion_counts = {}
    
    for line in lines[2:]:
        if line.strip() == '':
            continue
        cells = line.split('|')[1:-1]
        cells = [c.strip() for c in cells]
        
        champion = cells[2]
        
        if champion not in champion_counts:
            champion_counts[champion] = 0
        champion_counts[champion] += 1
    
    most_picked = max(champion_counts, key=champion_counts.get)
    return most_picked

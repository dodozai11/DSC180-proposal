```python
import re

def q1(text):
    """Return the average number of inhibitors destroyed by winning teams."""
    winning_inhibitors = []
    
    # Find all game entries
    pattern = r'In game (G\d+), (\w+(?:\s+\w+)*) (won|lost) after ([\d.]+) minutes, destroying (\d+) inhibitors'
    matches = re.findall(pattern, text)
    
    for game_id, team, result, duration, inhibitors in matches:
        if result == 'won':
            winning_inhibitors.append(int(inhibitors))
    
    return sum(winning_inhibitors) / len(winning_inhibitors) if winning_inhibitors else 0

def q2(text):
    """Return the average damage to towers of losing teams."""
    losing_damages = []
    
    # Find all game entries
    pattern = r'In game (G\d+), (\w+(?:\s+\w+)*) (won|lost) after ([\d.]+) minutes, destroying (\d+) inhibitors and dealing (\d+) damage to towers'
    matches = re.findall(pattern, text)
    
    for game_id, team, result, duration, inhibitors, damage in matches:
        if result == 'lost':
            losing_damages.append(int(damage))
    
    return sum(losing_damages) / len(losing_damages) if losing_damages else 0

def q3(text):
    """Return the game label of the longest game."""
    game_durations = {}
    
    # Find all game entries
    pattern = r'In game (G\d+), (\w+(?:\s+\w+)*) (won|lost) after ([\d.]+) minutes'
    matches = re.findall(pattern, text)
    
    for game_id, team, result, duration in matches:
        duration_float = float(duration)
        if game_id not in game_durations:
            game_durations[game_id] = duration_float
    
    longest_game = max(game_durations, key=game_durations.get)
    return longest_game
```

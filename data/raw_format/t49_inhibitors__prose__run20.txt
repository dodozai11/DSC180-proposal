```python
import re

def q1(text):
    """Return the average number of inhibitors destroyed by winning teams."""
    winning_inhibitors = []
    
    # Find all matches where a team won
    pattern = r'(\w+[\s\w]*?)\s+won\s+after\s+([\d.]+)\s+minutes,\s+destroying\s+(\d+)\s+inhibitors'
    matches = re.findall(pattern, text)
    
    for match in matches:
        inhibitors = int(match[2])
        winning_inhibitors.append(inhibitors)
    
    if winning_inhibitors:
        return sum(winning_inhibitors) / len(winning_inhibitors)
    return 0

def q2(text):
    """Return the average damage to towers of losing teams."""
    losing_damage = []
    
    # Find all matches where a team lost
    pattern = r'(\w+[\s\w]*?)\s+lost\s+after\s+([\d.]+)\s+minutes,\s+destroying\s+(\d+)\s+inhibitors\s+and\s+dealing\s+(\d+)\s+damage'
    matches = re.findall(pattern, text)
    
    for match in matches:
        damage = int(match[3])
        losing_damage.append(damage)
    
    if losing_damage:
        return sum(losing_damage) / len(losing_damage)
    return 0

def q3(text):
    """Return the game label of the longest game."""
    longest_game = None
    longest_duration = -1
    
    # Find all game entries with their game labels and durations
    pattern = r'In game (G\d+),.*?after\s+([\d.]+)\s+minutes'
    matches = re.findall(pattern, text)
    
    for match in matches:
        game_label = match[0]
        duration = float(match[1])
        
        if duration > longest_duration:
            longest_duration = duration
            longest_game = game_label
    
    return longest_game
```

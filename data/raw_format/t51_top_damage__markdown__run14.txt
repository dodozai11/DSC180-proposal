def q1(text):
    lines = text.strip().split('\n')
    header = lines[0].split('|')
    header = [h.strip() for h in header if h.strip()]
    
    total_damage = 0
    count = 0
    
    for line in lines[2:]:
        if not line.strip():
            continue
        cells = line.split('|')
        cells = [c.strip() for c in cells if c.strip()]
        
        result = cells[3]
        damage_per_min = float(cells[4])
        
        if result == 'loss':
            total_damage += damage_per_min
            count += 1
    
    return total_damage / count if count > 0 else 0


def q2(text):
    lines = text.strip().split('\n')
    header = lines[0].split('|')
    header = [h.strip() for h in header if h.strip()]
    
    max_damage = -1
    max_player = None
    
    for line in lines[2:]:
        if not line.strip():
            continue
        cells = line.split('|')
        cells = [c.strip() for c in cells if c.strip()]
        
        player = cells[0]
        damage_taken_per_min = float(cells[5])
        
        if damage_taken_per_min > max_damage:
            max_damage = damage_taken_per_min
            max_player = player
    
    return max_player


def q3(text):
    lines = text.strip().split('\n')
    header = lines[0].split('|')
    header = [h.strip() for h in header if h.strip()]
    
    teams = set()
    
    for line in lines[2:]:
        if not line.strip():
            continue
        cells = line.split('|')
        cells = [c.strip() for c in cells if c.strip()]
        
        team = cells[1]
        teams.add(team)
    
    return len(teams)

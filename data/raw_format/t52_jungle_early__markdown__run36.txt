def q1(text):
    lines = text.strip().split('\n')
    header = lines[0].split('|')[1:-1]
    header = [h.strip() for h in header]
    
    rows = []
    for line in lines[2:]:
        if line.strip():
            cells = line.split('|')[1:-1]
            cells = [c.strip() for c.strip() in cells]
            rows.append(cells)
    
    cs_diff_10_idx = header.index('cs_diff_10')
    result_idx = header.index('result')
    
    positive_cs_diff_rows = [row for row in rows if int(row[cs_diff_10_idx]) > 0]
    
    if not positive_cs_diff_rows:
        return 0
    
    wins = sum(1 for row in positive_cs_diff_rows if row[result_idx] == 'win')
    return wins / len(positive_cs_diff_rows)


def q2(text):
    lines = text.strip().split('\n')
    header = lines[0].split('|')[1:-1]
    header = [h.strip() for h in header]
    
    rows = []
    for line in lines[2:]:
        if line.strip():
            cells = line.split('|')[1:-1]
            cells = [c.strip() for c in cells]
            rows.append(cells)
    
    kills_at_10_idx = header.index('kills_at_10')
    
    total_kills = sum(int(row[kills_at_10_idx]) for row in rows)
    return total_kills


def q3(text):
    lines = text.strip().split('\n')
    header = lines[0].split('|')[1:-1]
    header = [h.strip() for h in header]
    
    rows = []
    for line in lines[2:]:
        if line.strip():
            cells = line.split('|')[1:-1]
            cells = [c.strip() for c in cells]
            rows.append(cells)
    
    champion_idx = header.index('champion')
    assists_at_10_idx = header.index('assists_at_10')
    
    champion_assists = {}
    for row in rows:
        champion = row[champion_idx]
        assists = int(row[assists_at_10_idx])
        champion_assists[champion] = champion_assists.get(champion, 0) + assists
    
    return max(champion_assists, key=champion_assists.get)

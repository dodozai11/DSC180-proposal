def q1(text):
    """Return the win rate of blue-side teams."""
    lines = text.strip().split('\n')
    # Skip header row
    data_lines = lines[1:]
    
    blue_wins = 0
    blue_total = 0
    
    for line in data_lines:
        parts = [p.strip() for p in line.split('|')[1:-1]]  # Remove empty strings from split
        if len(parts) >= 3:
            side = parts[2]
            result = parts[3]
            
            if side == 'blue':
                blue_total += 1
                if result == 'win':
                    blue_wins += 1
    
    if blue_total == 0:
        return 0
    return blue_wins / blue_total


def q2(text):
    """Return the average length in minutes of games won by the red-side team."""
    lines = text.strip().split('\n')
    # Skip header row
    data_lines = lines[1:]
    
    red_win_times = []
    
    for line in data_lines:
        parts = [p.strip() for p in line.split('|')[1:-1]]  # Remove empty strings from split
        if len(parts) >= 5:
            side = parts[2]
            result = parts[3]
            game_minutes = float(parts[4])
            
            if side == 'red' and result == 'win':
                red_win_times.append(game_minutes)
    
    if len(red_win_times) == 0:
        return 0
    return sum(red_win_times) / len(red_win_times)


def q3(text):
    """Return the name of the team with the most wins."""
    lines = text.strip().split('\n')
    # Skip header row
    data_lines = lines[1:]
    
    team_wins = {}
    
    for line in data_lines:
        parts = [p.strip() for p in line.split('|')[1:-1]]  # Remove empty strings from split
        if len(parts) >= 4:
            team = parts[1]
            result = parts[3]
            
            if team not in team_wins:
                team_wins[team] = 0
            
            if result == 'win':
                team_wins[team] += 1
    
    if not team_wins:
        return None
    return max(team_wins, key=team_wins.get)

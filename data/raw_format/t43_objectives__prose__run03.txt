def q2(text):
    """Returns the average number of barons taken by winning teams."""
    winning_barons = []
    
    # Parse each game entry
    pattern = r'In game (G\d+), ([^;]+); they (?:took|did not take) the first dragon and finished with \d+ dragons, (\d+) barons and \d+ towers\.'
    
    matches = list(re.finditer(pattern, text))
    
    for i in range(0, len(matches), 2):
        if i + 1 < len(matches):
            match1 = matches[i]
            match2 = matches[i + 1]
            
            team1 = match1.group(2)
            barons1 = int(match1.group(3))
            team2 = match2.group(2)
            barons2 = int(match2.group(3))
            
            # Find the winner in the original text
            game_section = text[match1.start():match2.end()]
            if f"{team1} won" in game_section:
                winning_barons.append(barons1)
            else:
                winning_barons.append(barons2)
    
    if len(winning_barons) == 0:
        return 0
    return sum(winning_barons) / len(winning_barons)


def q3(text):
    """Returns the total number of towers destroyed across all rows."""
    pattern = r'finished with \d+ dragons, \d+ barons and (\d+) towers'
    matches = re.findall(pattern, text)
    total_towers = sum(int(m) for m in matches)
    return total_towers

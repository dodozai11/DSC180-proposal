def q1(text):
    """Return the win rate of teams that took the first tower."""
    lines = text.strip().split('\n')
    
    # Skip header line
    data_lines = lines[1:]
    
    first_tower_wins = 0
    first_tower_total = 0
    
    for line in data_lines:
        parts = [p.strip() for p in line.split('|')[1:-1]]  # Remove empty strings from split
        if len(parts) >= 3:
            first_tower = parts[2]
            result = parts[1]
            
            if first_tower == 'yes':
                first_tower_total += 1
                if result == 'win':
                    first_tower_wins += 1
    
    if first_tower_total == 0:
        return 0
    
    return first_tower_wins / first_tower_total


def q2(text):
    """Return the average number of void grubs taken by winning teams."""
    lines = text.strip().split('\n')
    
    # Skip header line
    data_lines = lines[1:]
    
    winning_void_grubs = []
    
    for line in data_lines:
        parts = [p.strip() for p in line.split('|')[1:-1]]  # Remove empty strings from split
        if len(parts) >= 4:
            result = parts[1]
            void_grubs = int(parts[3])
            
            if result == 'win':
                winning_void_grubs.append(void_grubs)
    
    if len(winning_void_grubs) == 0:
        return 0
    
    return sum(winning_void_grubs) / len(winning_void_grubs)


def q3(text):
    """Return how many teams took at least one herald and still lost."""
    lines = text.strip().split('\n')
    
    # Skip header line
    data_lines = lines[1:]
    
    count = 0
    
    for line in data_lines:
        parts = [p.strip() for p in line.split('|')[1:-1]]  # Remove empty strings from split
        if len(parts) >= 5:
            result = parts[1]
            heralds = int(parts[4])
            
            if heralds >= 1 and result == 'loss':
                count += 1
    
    return count

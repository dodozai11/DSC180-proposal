import anthropic
import json


def q1(text: str) -> float:
    """Returns the average vision score of players who won minus the average vision score of players who lost."""
    client = anthropic.Anthropic()
    
    message = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": f"""Parse this League of Legends esports data and calculate the average vision score of players who won minus the average vision score of players who lost.

Data:
{text}

Return ONLY a single number (can be decimal) representing the difference. No other text."""
            }
        ]
    )
    
    return float(message.content[0].text.strip())

```python
import re

def parse_data(text):
    """Parse the text data into a list of game records."""
    games = []
    # Pattern to match each game record
    pattern = r"In game (G\d+), (.+?) played on the (blue|red) side and (won|lost), with a gold difference of (-?\d+) at 15 minutes\."
    matches = re.findall(pattern, text)
    
    for match in matches:
        game_id, team, side, result, gold_diff = match
        games.append({
            'game_id': game_id,
            'team': team,
            'side': side,
            'result': result,
            'gold_diff': int(gold_diff),
            'won': result == 'won'
        })
    
    return games

def q1(text):
    """Return the average gold difference at 15 minutes of the winning teams."""
    games = parse_data(text)
    winning_games = [game for game in games if game['won']]
    
    if not winning_games:
        return 0
    
    total_gold_diff = sum(game['gold_diff'] for game in winning_games)
    return total_gold_diff / len(winning_games)

def q2(text):
    """Return how many teams won despite a negative gold difference at 15 minutes."""
    games = parse_data(text)
    won_with_negative = [game for game in games if game['won'] and game['gold_diff'] < 0]
    return len(won_with_negative)

def q3(text):
    """Return the win rate of teams that were ahead in gold at 15 minutes."""
    games = parse_data(text)
    ahead_in_gold = [game for game in games if game['gold_diff'] > 0]
    
    if not ahead_in_gold:
        return 0
    
    wins = sum(1 for game in ahead_in_gold if game['won'])
    return wins / len(ahead_in_gold)
```

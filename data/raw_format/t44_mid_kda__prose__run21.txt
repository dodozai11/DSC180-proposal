```python
import re
from collections import Counter

def parse_matches(text):
    """Parse match data from text format."""
    matches = []
    # Pattern to match: "Name of Team played Champion in role and result with K kills, D deaths and A assists."
    pattern = r"(\w+) of (.+?) played (.+?) in (\w+) and (\w+) with (\d+) kills, (\d+) deaths and (\d+) assists"
    
    for match in re.finditer(pattern, text):
        player_name = match.group(1)
        team = match.group(2)
        champion = match.group(3)
        role = match.group(4)
        result = match.group(5)
        kills = int(match.group(6))
        deaths = int(match.group(7))
        assists = int(match.group(8))
        
        matches.append({
            'player': player_name,
            'team': team,
            'champion': champion,
            'role': role,
            'result': result,
            'kills': kills,
            'deaths': deaths,
            'assists': assists
        })
    
    return matches

def q1(text):
    """Return the name of the player with the highest KDA."""
    matches = parse_matches(text)
    
    best_player = None
    best_kda = -1
    
    for match in matches:
        kda = (match['kills'] + match['assists']) / max(match['deaths'], 1)
        if kda > best_kda:
            best_kda = kda
            best_player = match['player']
    
    return best_player

def q2(text):
    """Return the average number of kills in games that were won."""
    matches = parse_matches(text)
    
    won_kills = []
    for match in matches:
        if match['result'] == 'won':
            won_kills.append(match['kills'])
    
    if not won_kills:
        return 0
    
    return sum(won_kills) / len(won_kills)

def q3(text):
    """Return the name of the most-picked champion."""
    matches = parse_matches(text)
    
    champions = [match['champion'] for match in matches]
    champion_counts = Counter(champions)
    
    most_common_champion = champion_counts.most_common(1)[0][0]
    
    return most_common_champion
```

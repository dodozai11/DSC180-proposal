```python
import re
from xml.etree import ElementTree as ET

def q1(text):
    """Return the name of the player with the highest CS per minute."""
    root = ET.fromstring(f"<root>{text}</root>")
    max_cs = -1
    max_player = None
    for row in root.findall('row'):
        cs = float(row.find('cs_per_min').text)
        if cs > max_cs:
            max_cs = cs
            max_player = row.find('player').text
    return max_player

def q2(text):
    """Return the average gold share of players who won."""
    root = ET.fromstring(f"<root>{text}</root>")
    total_gold_share = 0
    win_count = 0
    for row in root.findall('row'):
        if row.find('result').text == 'win':
            gold_share = float(row.find('gold_share').text)
            total_gold_share += gold_share
            win_count += 1
    return total_gold_share / win_count if win_count > 0 else 0

def q3(text):
    """Return how many players had more than 9 CS per minute."""
    root = ET.fromstring(f"<root>{text}</root>")
    count = 0
    for row in root.findall('row'):
        cs = float(row.find('cs_per_min').text)
        if cs > 9:
            count += 1
    return count
```

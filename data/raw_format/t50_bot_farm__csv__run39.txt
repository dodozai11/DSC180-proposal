def q1(text):
    lines = text.strip().split('\n')
    header = lines[0].split(',')
    
    max_cs_per_min = -1
    max_player = None
    
    for line in lines[1:]:
        values = line.split(',')
        player = values[0]
        cs_per_min = float(values[4])
        
        if cs_per_min > max_cs_per_min:
            max_cs_per_min = cs_per_min
            max_player = player

def q2(text):
    lines = text.strip().split('\n')
    
    total_gold_share = 0
    win_count = 0
    
    for line in lines[1:]:
        values = line.split(',')
        result = values[3]
        gold_share = float(values[5])
        
        if result == 'win':
            total_gold_share += gold_share
            win_count += 1
    
    return total_gold_share / win_count if win_count > 0 else 0


def q3(text):
    lines = text.strip().split('\n')
    
    count = 0

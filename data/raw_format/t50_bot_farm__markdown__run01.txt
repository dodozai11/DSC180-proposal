def q1(text):
    lines = text.strip().split('\n')
    header = lines[0].split('|')[1:-1]
    header = [h.strip() for h in header]
    
    cs_per_min_idx = header.index('cs_per_min')
    player_idx = header.index('player')
    
    max_cs = -1
    max_player = None
    
    for line in lines[2:]:
        if line.strip() == '':
            continue
        cells = line.split('|')[1:-1]
        cells = [c.strip() for c in cells]
        
        player = cells[player_idx]
        cs_per_min = float(cells[cs_per_min_idx])
        
        if cs_per_min > max_cs:
            max_cs = cs_per_min
            max_player = player
    
    return max_player


def q2(text):
    lines = text.strip().split('\n')
    header = lines[0].split('|')[1:-1]
    header = [h.strip() for h in header]
    
    gold_share_idx = header.index('gold_share')
    result_idx = header.index('result')
    
    total_gold = 0
    win_count = 0
    
    for line in lines[2:]:
        if line.strip() == '':
            continue
        cells = line.split('|')[1:-1]
        cells = [c.strip() for c in cells]
        
        result = cells[result_idx]
        gold_share = float(cells[gold_share_idx])
        
        if result == 'win':
            total_gold += gold_share
            win_count += 1
    
    return total_gold / win_count

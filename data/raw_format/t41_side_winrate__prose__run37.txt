```python
import re
from collections import defaultdict

def parse_data(text):
    """Parse the text data into a list of game records."""
    games = []
    pattern = r'In game (\w+), (.+?) played on the (blue|red) side and (won|lost); the game lasted ([\d.]+) minutes\.'
    matches = re.findall(pattern, text)
    
    for match in matches:
        game_id, team, side, result, duration = match
        games.append({
            'game_id': game_id,
            'team': team,
            'side': side,
            'result': result,
            'duration': float(duration)
        })
    
    return games

def q1(text):
    """Return the win rate of blue-side teams."""
    games = parse_data(text)
    
    blue_wins = 0
    blue_total = 0
    
    for game in games:
        if game['side'] == 'blue':
            blue_total += 1
            if game['result'] == 'won':
                blue_wins += 1
    
    if blue_total == 0:
        return 0
    
    return blue_wins / blue_total

def q2(text):
    """Return the average length in minutes of games won by the red-side team."""
    games = parse_data(text)
    
    red_wins = []
    
    for game in games:
        if game['side'] == 'red' and game['result'] == 'won':
            red_wins.append(game['duration'])
    
    if len(red_wins) == 0:
        return 0
    
    return sum(red_wins) / len(red_wins)

def q3(text):
    """Return the name of the team with the most wins."""
    games = parse_data(text)
    
    team_wins = defaultdict(int)
    
    for game in games:
        if game['result'] == 'won':
            team_wins[game['team']] += 1
    
    if not team_wins:
        return None
    
    return max(team_wins, key=team_wins.get)
```

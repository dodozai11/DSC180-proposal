import re

def q1(text):
    """Returns the average gold difference at 15 minutes of the winning teams."""
    winning_gold_diffs = []
    
    lines = text.split('. ')
    for line in lines:
        if not line.strip():
            continue
        
        # Parse: "In game G##, Team Name played on the color side and result, with a gold difference of value at 15 minutes"
        match = re.search(r'In game (\w+), (.+?) played on the (\w+) side and (\w+), with a gold difference of (-?\d+) at 15 minutes', line)
        if match:
            game_id, team_name, side, result, gold_diff = match.groups()
            gold_diff = int(gold_diff)
            
            if result == 'won':
                winning_gold_diffs.append(gold_diff)
    
    if winning_gold_diffs:
        return sum(winning_gold_diffs) / len(winning_gold_diffs)
    return 0

def q2(text):
    """Returns how many teams won despite a negative gold difference at 15 minutes."""
    count = 0
    
    lines = text.split('. ')
    for line in lines:
        if not line.strip():
            continue
        
        match = re.search(r'In game (\w+), (.+?) played on the (\w+) side and (\w+), with a gold difference of (-?\d+) at 15 minutes', line)
        if match:
            game_id, team_name, side, result, gold_diff = match.groups()
            gold_diff = int(gold_diff)
            
            if result == 'won' and gold_diff < 0:
                count += 1
    
    return count

def q3(text):
    """Returns the win rate of teams that were ahead in gold at 15 minutes."""
    ahead_and_won = 0
    ahead_total = 0
    
    lines = text.split('. ')
    for line in lines:
        if not line.strip():
            continue
        
        match = re.search(r'In game (\w+), (.+?) played on the (\w+) side and (\w+), with a gold difference of (-?\d+) at 15 minutes', line)
        if match:
            game_id, team_name, side, result, gold_diff = match.groups()
            gold_diff = int(gold_diff)
            
            if gold_diff > 0:
                ahead_total += 1
                if result == 'won':
                    ahead_and_won += 1
    
    if ahead_total == 0:
        return 0
    return ahead_and_won / ahead_total

def q1(text):
    lines = text.strip().split('\n')
    max_cs_player = None
    max_cs = -1
    
    for i in range(2, len(lines)):
        if not lines[i].strip():
            continue
        parts = [p.strip() for p in lines[i].split('|') if p.strip()]
        if len(parts) >= 5:
            player = parts[0]
            cs_per_min = float(parts[4])
            if cs_per_min > max_cs:
                max_cs = cs_per_min
                max_cs_player = player
    
    return max_cs_player


def q2(text):
    lines = text.strip().split('\n')
    total_gold_share = 0
    win_count = 0
    
    for i in range(2, len(lines)):
        if not lines[i].strip():
            continue
        parts = [p.strip() for p in lines[i].split('|') if p.strip()]
        if len(parts) >= 6:
            result = parts[3]
            gold_share = float(parts[5])
            if result.lower() == 'win':
                total_gold_share += gold_share
                win_count += 1
    
    if win_count == 0:
        return 0
    return total_gold_share / win_count


def q3(text):
    lines = text.strip().split('\n')
    count = 0
    
    for i in range(2, len(lines)):
        if not lines[i].strip():
            continue
        parts = [p.strip() for p in lines[i].split('|') if p.strip()]
        if len(parts) >= 5:
            cs_per_min = float(parts[4])
            if cs_per_min > 9:
                count += 1
    
    return count

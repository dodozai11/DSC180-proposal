import json
import re


def q1(text):
    """Returns the average vision score of players who won minus the average vision score of players who lost."""
    data = json.loads(text)
    
    winners = [player for player in data if player["result"] == "win"]
    losers = [player for player in data if player["result"] == "loss"]
    
    avg_winners = sum(p["vision_score"] for p in winners) / len(winners)
    avg_losers = sum(p["vision_score"] for p in losers) / len(losers)
    
    return avg_winners - avg_losers


def q2(text):
    """Returns the name of the player who placed the most wards."""
    data = json.loads(text)
    
    player_with_most_wards = max(data, key=lambda p: p["wards_placed"])
    
    return player_with_most_wards["player"]


def q3(text):
    """Returns how many players bought at least 15 control wards."""
    data = json.loads(text)
    
    count = sum(1 for player in data if player["control_wards"] >= 15)
    
    return count

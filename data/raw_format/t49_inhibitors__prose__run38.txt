```python
import re

def q1(text):
    """Return the average number of inhibitors destroyed by winning teams."""
    winning_inhibitors = []
    
    # Find all game entries with outcome and inhibitors
    pattern = r'In game (G\d+), (\w+(?:\s+\w+)*) won after ([\d.]+) minutes, destroying (\d+) inhibitors'
    matches = re.findall(pattern, text)
    
    for match in matches:
        game_id, team, duration, inhibitors = match
        winning_inhibitors.append(int(inhibitors))
    
    if winning_inhibitors:
        return sum(winning_inhibitors) / len(winning_inhibitors)
    return 0

def q2(text):
    """Return the average damage to towers of losing teams."""
    losing_damages = []
    
    # Find all game entries with outcome and tower damage
    pattern = r'In game (G\d+), (\w+(?:\s+\w+)*) lost after ([\d.]+) minutes, destroying (\d+) inhibitors and dealing (\d+) damage to towers'
    matches = re.findall(pattern, text)
    
    for match in matches:
        game_id, team, duration, inhibitors, damage = match
        losing_damages.append(int(damage))
    
    if losing_damages:
        return sum(losing_damages) / len(losing_damages)
    return 0

def q3(text):
    """Return the game label of the longest game."""
    max_duration = -1
    longest_game = None
    
    # Find all game entries with duration
    pattern = r'In game (G\d+), \w+(?:\s+\w+)* (?:won|lost) after ([\d.]+) minutes'
    matches = re.findall(pattern, text)
    
    # Use a set to track which games we've already processed
    seen_games = set()
    
    for match in matches:
        game_id, duration = match
        if game_id not in seen_games:
            seen_games.add(game_id)
            duration_float = float(duration)
            if duration_float > max_duration:
                max_duration = duration_float
                longest_game = game_id
    
    return longest_game
```

def q1(text):
    lines = text.strip().split('\n')
    header = lines[0].split('|')[1:-1]
    header = [h.strip() for h in header]
    
    damage_per_min_idx = header.index('damage_per_min')
    result_idx = header.index('result')
    
    total_damage = 0
    count = 0
    
    for line in lines[2:]:
        if line.strip():
            cells = line.split('|')[1:-1]
            cells = [c.strip() for c in cells]
            
            if cells[result_idx] == 'loss':
                total_damage += float(cells[damage_per_min_idx])
                count += 1
    
    return total_damage / count if count > 0 else 0


def q2(text):
    lines = text.strip().split('\n')
    header = lines[0].split('|')[1:-1]
    header = [h.strip() for h in header]
    
    player_idx = header.index('player')
    damage_taken_per_min_idx = header.index('damage_taken_per_min')
    
    max_damage = -1
    max_player = None
    
    for line in lines[2:]:
        if line.strip():
            cells = line.split('|')[1:-1]
            cells = [c.strip() for c in cells]
            
            damage_taken = float(cells[damage_taken_per_min_idx])
            if damage_taken > max_damage:
                max_damage = damage_taken
                max_player = cells[player_idx]
    
    return max_player


def q3(text):
    lines = text.strip().split('\n')
    header = lines[0].split('|')[1:-1]
    header = [h.strip() for h in header]
    
    team_idx = header.index('team')
    
    teams = set()
    
    for line in lines[2:]:
        if line.strip():
            cells = line.split('|')[1:-1]
            cells = [c.strip() for c in cells]
            teams.add(cells[team_idx])
    
    return len(teams)

def q1(text):
    """Returns the average number of inhibitors destroyed by winning teams."""
    lines = text.strip().split('\n')
    
    # Skip header row
    data_lines = lines[1:]
    
    winning_inhibitors = []
    
    for line in data_lines:
        parts = [p.strip() for p in line.split('|')]
        # parts[0] is empty, parts[1] is game, parts[2] is team, parts[3] is result, etc.
        result = parts[3]
        inhibitors = int(parts[5])
        
        if result == 'win':
            winning_inhibitors.append(inhibitors)
    
    return sum(winning_inhibitors) / len(winning_inhibitors)


def q2(text):
    """Returns the average damage to towers of losing teams."""
    lines = text.strip().split('\n')
    
    # Skip header row
    data_lines = lines[1:]
    
    losing_tower_damage = []
    
    for line in data_lines:
        parts = [p.strip() for p in line.split('|')]
        # parts[0] is empty, parts[1] is game, parts[2] is team, parts[3] is result, parts[6] is tower_damage
        result = parts[3]
        tower_damage = int(parts[6])
        
        if result == 'loss':
            losing_tower_damage.append(tower_damage)
    
    return sum(losing_tower_damage) / len(losing_tower_damage)


def q3(text):
    """Returns the game label of the longest game."""
    lines = text.strip().split('\n')
    
    # Skip header row
    data_lines = lines[1:]
    
    max_duration = -1
    longest_game = None
    
    for line in data_lines:
        parts = [p.strip() for p in line.split('|')]
        # parts[0] is empty, parts[1] is game, parts[4] is game_minutes
        game = parts[1]
        game_minutes = float(parts[4])
        
        if game_minutes > max_duration:
            max_duration = game_minutes
            longest_game = game
    
    return longest_game

```python
import re

def q1(text):
    """Return the win rate of teams that took the first tower."""
    first_tower_wins = 0
    first_tower_total = 0
    
    # Parse each game entry
    pattern = r"In game (G\d+), (.+?) (won|lost); they (took|did not take) the first tower"
    matches = re.findall(pattern, text)
    
    for match in matches:
        game_id, team_name, result, tower_status = match
        if tower_status == "took":
            first_tower_total += 1
            if result == "won":
                first_tower_wins += 1
    
    if first_tower_total == 0:
        return 0
    return first_tower_wins / first_tower_total


def q2(text):
    """Return the average number of void grubs taken by winning teams."""
    winning_void_grubs = []
    
    # Parse each game entry
    pattern = r"In game (G\d+), (.+?) (won|lost); they (?:took|did not take) the first tower and finished with (\d+) void grubs"
    matches = re.findall(pattern, text)
    
    for match in matches:
        game_id, team_name, result, void_grubs = match
        if result == "won":
            winning_void_grubs.append(int(void_grubs))
    
    if len(winning_void_grubs) == 0:
        return 0
    return sum(winning_void_grubs) / len(winning_void_grubs)


def q3(text):
    """Return how many teams took at least one herald and still lost."""
    lost_with_herald = 0
    
    # Parse each game entry
    pattern = r"In game (G\d+), (.+?) (won|lost); they (?:took|did not take) the first tower and finished with \d+ void grubs and (\d+) heralds"
    matches = re.findall(pattern, text)
    
    for match in matches:
        game_id, team_name, result, heralds = match
        if result == "lost" and int(heralds) >= 1:
            lost_with_herald += 1
    
    return lost_with_herald
```

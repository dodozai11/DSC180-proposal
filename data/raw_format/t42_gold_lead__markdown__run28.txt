import re
from collections import defaultdict

def parse_data(text):
    """Parse the data from the text format."""
    lines = text.strip().split('\n')
    # Skip header line
    data = []
    for line in lines[1:]:
        # Parse the line - handle team names with spaces
        parts = line.split('|')
        parts = [p.strip() for p in parts if p.strip()]
        if len(parts) >= 5:
            game = parts[0]
            team = parts[1]
            side = parts[2]
            result = parts[3]
            gold_diff_15 = int(parts[4])
            data.append({
                'game': game,
                'team': team,
                'side': side,
                'result': result,
                'gold_diff_15': gold_diff_15
            })
    return data


def q1(text):
    """Return the average gold difference at 15 minutes of the winning teams."""
    data = parse_data(text)
    winning_gold_diffs = [row['gold_diff_15'] for row in data if row['result'] == 'win']
    if len(winning_gold_diffs) == 0:
        return 0
    return sum(winning_gold_diffs) / len(winning_gold_diffs)


def q2(text):
    """Return how many teams won despite a negative gold difference at 15 minutes."""
    data = parse_data(text)
    count = 0
    for row in data:
        if row['result'] == 'win' and row['gold_diff_15'] < 0:
            count += 1
    return count


def q3(text):
    """Return the win rate of teams that were ahead in gold at 15 minutes."""
    data = parse_data(text)
    ahead_teams = [row for row in data if row['gold_diff_15'] > 0]
    if len(ahead_teams) == 0:
        return 0
    wins = sum(1 for row in ahead_teams if row['result'] == 'win')
    return wins / len(ahead_teams)

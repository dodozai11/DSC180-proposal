import re

def q1(text):
    """Return the win rate of teams that took the first tower."""
    matches = re.findall(
        r"In game (\w+), (.+?) (won|lost); they (took|did not take) the first tower",
        text
    )
    
    first_tower_games = []
    for game, team, result, tower_status in matches:
        took_first_tower = tower_status == "took"
        won = result == "won"
        first_tower_games.append((took_first_tower, won))
    
    teams_with_first_tower = [won for took, won in first_tower_games if took]
    
    if len(teams_with_first_tower) == 0:
        return 0
    
    win_rate = sum(teams_with_first_tower) / len(teams_with_first_tower)
    return win_rate


def q2(text):
    """Return the average number of void grubs taken by winning teams."""
    matches = re.findall(
        r"In game (\w+), (.+?) (won|lost); .+? finished with (\d+) void grubs",
        text
    )
    
    winning_void_grubs = []
    for game, team, result, void_grubs in matches:
        if result == "won":
            winning_void_grubs.append(int(void_grubs))
    
    if len(winning_void_grubs) == 0:
        return 0
    
    average = sum(winning_void_grubs) / len(winning_void_grubs)
    return average


def q3(text):
    """Return how many teams took at least one herald and still lost."""
    matches = re.findall(
        r"In game (\w+), (.+?) (won|lost); .+? finished with \d+ void grubs and (\d+) heralds",
        text
    )
    
    count = 0
    for game, team, result, heralds in matches:
        heralds_count = int(heralds)
        if heralds_count >= 1 and result == "lost":
            count += 1
    
    return count

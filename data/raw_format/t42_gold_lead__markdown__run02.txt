```python
import csv
from io import StringIO
from statistics import mean


def parse_data(text):
    """Parse the CSV data from text."""
    reader = csv.DictReader(StringIO(text.strip()), delimiter='|')
    rows = []
    for row in reader:
        # Clean up whitespace in keys and values
        cleaned_row = {k.strip(): v.strip() for k, v in row.items()}
        rows.append(cleaned_row)
    return rows


def q1(text):
    """Return the average gold difference at 15 minutes of the winning teams."""
    rows = parse_data(text)
    winning_gold_diffs = [
        int(row['gold_diff_15']) 
        for row in rows 
        if row['result'] == 'win'
    ]
    return mean(winning_gold_diffs)


def q2(text):
    """Return how many teams won despite a negative gold difference at 15 minutes."""
    rows = parse_data(text)
    count = sum(
        1 
        for row in rows 
        if row['result'] == 'win' and int(row['gold_diff_15']) < 0
    )
    return count


def q3(text):
    """Return the win rate of teams that were ahead in gold at 15 minutes."""
    rows = parse_data(text)
    ahead_in_gold = [row for row in rows if int(row['gold_diff_15']) > 0]
    
    if not ahead_in_gold:
        return 0
    
    wins = sum(1 for row in ahead_in_gold if row['result'] == 'win')
    return wins / len(ahead_in_gold)
```

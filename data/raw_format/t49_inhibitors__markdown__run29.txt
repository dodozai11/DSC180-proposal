def q1(text):
    """Return the average number of inhibitors destroyed by winning teams."""
    lines = text.strip().split('\n')
    header = lines[0].split('|')
    
    winning_inhibitors = []
    
    for line in lines[2:]:
        if line.strip():
            parts = [x.strip() for x in line.split('|')[1:-1]]
            if len(parts) >= 3:
                result = parts[2]
                inhibitors = float(parts[4])
                if result == 'win':
                    winning_inhibitors.append(inhibitors)
    
    return sum(winning_inhibitors) / len(winning_inhibitors) if winning_inhibitors else 0


def q2(text):
    """Return the average damage to towers of losing teams."""
    lines = text.strip().split('\n')
    
    losing_damage = []
    
    for line in lines[2:]:
        if line.strip():
            parts = [x.strip() for x in line.split('|')[1:-1]]
            if len(parts) >= 5:
                result = parts[2]
                tower_damage = float(parts[5])
                if result == 'loss':
                    losing_damage.append(tower_damage)
    
    return sum(losing_damage) / len(losing_damage) if losing_damage else 0


def q3(text):
    """Return the game label of the longest game."""
    lines = text.strip().split('\n')
    
    max_duration = -1
    longest_game = None
    
    for line in lines[2:]:
        if line.strip():
            parts = [x.strip() for x in line.split('|')[1:-1]]
            if len(parts) >= 4:
                game = parts[0]
                duration = float(parts[3])
                if duration > max_duration:
                    max_duration = duration
                    longest_game = game
    
    return longest_game

import re
from collections import defaultdict

def q1(text):
    """Returns the number of different champions played."""
    champions = set()
    # Pattern: "played <Champion> in the"
    matches = re.findall(r'played (\w+(?:\'?\w+)*) in the', text)
    champions.update(matches)
    return len(champions)

def q2(text):
    """Returns the champion with the best win rate among champions played at least twice."""
    champion_stats = defaultdict(lambda: {'wins': 0, 'total': 0})
    
    # Pattern: "played <Champion> in the jungle and <won/lost>"
    matches = re.findall(r'played (\w+(?:\'?\w+)*) in the jungle and (\w+)', text)
    
    for champion, result in matches:
        champion_stats[champion]['total'] += 1
        if result == 'won':
            champion_stats[champion]['wins'] += 1
    
    # Filter champions played at least twice
    qualified = {champ: stats for champ, stats in champion_stats.items() if stats['total'] >= 2}
    
    if not qualified:
        return None
    
    # Find champion with best win rate
    best_champion = max(qualified.items(), key=lambda x: x[1]['wins'] / x[1]['total'])
    return best_champion[0]

def q3(text):
    """Returns the average number of deaths in games that were lost."""
    deaths_in_losses = []
    
    # Pattern: "and <won/lost>, dying <number> times"
    matches = re.findall(r'and (\w+), dying (\d+) times', text)
    
    for result, deaths in matches:
        if result == 'lost':
            deaths_in_losses.append(int(deaths))
    
    if not deaths_in_losses:
        return 0
    
    return sum(deaths_in_losses) / len(deaths_in_losses)

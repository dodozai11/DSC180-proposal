import re
from collections import defaultdict

def q1(text):
    """Return the number of different champions played."""
    champions = set()
    # Pattern: "played <champion> in the jungle"
    matches = re.findall(r'played (\w+(?:\'?\w+)?)', text)
    champions.update(matches)
    return len(champions)


def q2(text):
    """Return the champion with the best win rate among champions played at least twice."""
    champion_stats = defaultdict(lambda: {'wins': 0, 'total': 0})
    
    # Pattern: "<player> played <champion> in the jungle and <result>, dying <deaths> times"
    pattern = r'played (\w+(?:\'?\w+)?)\s+in the jungle and (won|lost)'
    matches = re.findall(pattern, text)
    
    for champion, result in matches:
        champion_stats[champion]['total'] += 1
        if result == 'won':
            champion_stats[champion]['wins'] += 1
    
    # Filter champions played at least twice
    qualified_champions = {
        champ: stats for champ, stats in champion_stats.items()
        if stats['total'] >= 2
    }
    
    if not qualified_champions:
        return None
    
    # Find champion with best win rate
    best_champion = max(
        qualified_champions.items(),
        key=lambda x: x[1]['wins'] / x[1]['total']
    )[0]
    
    return best_champion


def q3(text):
    """Return the average number of deaths in games that were lost."""
    deaths_in_losses = []
    
    # Pattern: "and lost, dying <deaths> times"
    pattern = r'and lost, dying (\d+) times'
    matches = re.findall(pattern, text)
    
    deaths_in_losses = [int(deaths) for deaths in matches]
    
    if not deaths_in_losses:
        return 0
    
    return sum(deaths_in_losses) / len(deaths_in_losses)

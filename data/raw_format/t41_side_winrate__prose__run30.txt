```python
import re
from collections import defaultdict

def parse_data(text):
    """Parse the match data from text format."""
    matches = []
    pattern = r"In game (\w+), (.+?) played on the (\w+) side and (\w+); the game lasted ([\d.]+) minutes\."
    
    for match in re.finditer(pattern, text):
        game_id = match.group(1)
        team = match.group(2)
        side = match.group(3)
        result = match.group(4)
        duration = float(match.group(5))
        
        matches.append({
            'game_id': game_id,
            'team': team,
            'side': side,
            'result': result,
            'duration': duration,
            'won': result == 'won'
        })
    
    return matches

def q1(text):
    """Return the win rate of blue-side teams."""
    matches = parse_data(text)
    
    blue_matches = [m for m in matches if m['side'] == 'blue']
    blue_wins = sum(1 for m in blue_matches if m['won'])
    
    if len(blue_matches) == 0:
        return 0
    
    return blue_wins / len(blue_matches)

def q2(text):
    """Return the average length in minutes of games won by the red-side team."""
    matches = parse_data(text)
    
    red_wins = [m for m in matches if m['side'] == 'red' and m['won']]
    
    if len(red_wins) == 0:
        return 0
    
    total_duration = sum(m['duration'] for m in red_wins)
    return total_duration / len(red_wins)

def q3(text):
    """Return the name of the team with the most wins."""
    matches = parse_data(text)
    
    team_wins = defaultdict(int)
    for m in matches:
        if m['won']:
            team_wins[m['team']] += 1
    
    if not team_wins:
        return None
    
    return max(team_wins, key=team_wins.get)
```

import json
import re

def q1(text):
    """Returns the name of the player with the highest CS per minute."""
    data = json.loads(text)
    return max(data, key=lambda x: x['cs_per_min'])['player']

def q2(text):
    """Returns the average gold share of players who won."""
    data = json.loads(text)
    winners = [player['gold_share'] for player in data if player['result'] == 'win']
    return sum(winners) / len(winners) if winners else 0

def q3(text):
    """Returns how many players had more than 9 CS per minute."""
    data = json.loads(text)
    return sum(1 for player in data if player['cs_per_min'] > 9)

import json
import re

def q1(text):
    """Returns the number of different champions played."""
    data = json.loads(text)
    champions = set(entry["champion"] for entry in data)
    return len(champions)


def q2(text):
    """Returns the name of the champion with the best win rate among champions played at least twice."""
    data = json.loads(text)
    
    # Count wins and total games for each champion
    champion_stats = {}
    for entry in data:
        champion = entry["champion"]
        if champion not in champion_stats:
            champion_stats[champion] = {"wins": 0, "total": 0}
        
        champion_stats[champion]["total"] += 1
        if entry["result"] == "win":
            champion_stats[champion]["wins"] += 1
    
    # Filter champions played at least twice
    champions_twice_or_more = {
        champ: stats for champ, stats in champion_stats.items()
        if stats["total"] >= 2
    }
    
    if not champions_twice_or_more:
        return None
    
    # Find champion with best win rate
    best_champion = max(
        champions_twice_or_more.items(),
        key=lambda x: x[1]["wins"] / x[1]["total"]
    )
    
    return best_champion[0]


def q3(text):
    """Returns the average number of deaths in games that were lost."""
    data = json.loads(text)
    
    losses = [entry["deaths"] for entry in data if entry["result"] == "loss"]
    
    if not losses:
        return 0
    
    return sum(losses) / len(losses)

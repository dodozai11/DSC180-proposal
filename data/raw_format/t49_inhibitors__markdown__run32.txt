def q1(text):
    """Return the average number of inhibitors destroyed by winning teams."""
    lines = text.strip().split('\n')
    header = lines[0].split('|')[1:-1]
    header = [h.strip() for h in header]
    
    inhibitors_won = []
    for line in lines[2:]:
        if line.strip():
            parts = [p.strip() for p in line.split('|')[1:-1]]
            if len(parts) == len(header):
                row = dict(zip(header, parts))
                if row['result'] == 'win':
                    inhibitors_won.append(int(row['inhibitors']))
    
    return sum(inhibitors_won) / len(inhibitors_won) if inhibitors_won else 0

def q3(text):
    """Return the game label of the longest game."""
    lines = text.strip().split('\n')
    header = lines[0].split('|')[1:-1]
    header = [h.strip() for h in header]
    
    longest_game = None
    longest_duration = -1
    
    for line in lines[2:]:
        if line.strip():
            parts = [p.strip() for p in line.split('|')[1:-1]]
            if len(parts) == len(header):
                row = dict(zip(header, parts))
                duration = float(row['game_minutes'])
                if duration > longest_duration:
                    longest_duration = duration
                    longest_game = row['game']
    
    return longest_game

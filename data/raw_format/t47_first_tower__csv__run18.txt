def q1(text):
    lines = text.strip().split('\n')[1:]  # Skip header
    
    first_tower_games = []
    for line in lines:
        parts = line.split(',')
        first_tower = parts[3]
        result = parts[2]
        
        if first_tower == 'yes':
            first_tower_games.append(result)
    
    if len(first_tower_games) == 0:
        return 0
    
    wins = sum(1 for result in first_tower_games if result == 'win')
    return wins / len(first_tower_games)


def q2(text):
    lines = text.strip().split('\n')[1:]  # Skip header
    
    winning_void_grubs = []
    for line in lines:
        parts = line.split(',')
        result = parts[2]
        void_grubs = int(parts[4])
        
        if result == 'win':
            winning_void_grubs.append(void_grubs)
    
    if len(winning_void_grubs) == 0:
        return 0
    
    return sum(winning_void_grubs) / len(winning_void_grubs)


def q3(text):
    lines = text.strip().split('\n')[1:]  # Skip header
    
    count = 0
    for line in lines:
        parts = line.split(',')
        result = parts[2]
        heralds = int(parts[5])
        
        if heralds >= 1 and result == 'loss':
            count += 1
    
    return count

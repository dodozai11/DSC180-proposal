import re

def q1(text):
    """Return the win rate of players with a positive CS difference at 10 minutes."""
    matches = re.findall(
        r'(\w+)\s+played\s+(\w+)\s+in\s+the\s+jungle\s+and\s+(won|lost);\s+at\s+10\s+minutes\s+they\s+had\s+a\s+CS\s+difference\s+of\s+([-\d]+),\s+(\d+)\s+kills\s+and\s+(\d+)\s+assists',
        text
    )
    
    positive_cs_players = [(player, result, int(cs_diff)) for player, champion, result, cs_diff, kills, assists in matches if int(cs_diff) > 0]
    
    if not positive_cs_players:
        return 0
    
    wins = sum(1 for player, result, cs_diff in positive_cs_players if result == 'won')
    return wins / len(positive_cs_players)

def q2(text):
    """Return the total number of kills at 10 minutes across all rows."""
    matches = re.findall(
        r'(\w+)\s+played\s+(\w+)\s+in\s+the\s+jungle\s+and\s+(won|lost);\s+at\s+10\s+minutes\s+they\s+had\s+a\s+CS\s+difference\s+of\s+([-\d]+),\s+(\d+)\s+kills\s+and\s+(\d+)\s+assists',
        text
    )
    
    total_kills = sum(int(kills) for player, champion, result, cs_diff, kills, assists in matches)
    return total_kills

def q3(text):
    """Return the name of the champion with the most assists at 10 minutes in total."""
    matches = re.findall(
        r'(\w+)\s+played\s+(\w+)\s+in\s+the\s+jungle\s+and\s+(won|lost);\s+at\s+10\s+minutes\s+they\s+had\s+a\s+CS\s+difference\s+of\s+([-\d]+),\s+(\d+)\s+kills\s+and\s+(\d+)\s+assists',
        text
    )
    
    champion_assists = {}
    for player, champion, result, cs_diff, kills, assists in matches:
        assists_int = int(assists)
        if champion not in champion_assists:
            champion_assists[champion] = 0
        champion_assists[champion] += assists_int
    
    return max(champion_assists, key=champion_assists.get)

def q1(text):
    """Return the win rate of players with a positive CS difference at 10 minutes."""
    lines = text.strip().split('\n')
    
    # Skip header row
    data_lines = lines[2:]
    
    positive_cs_count = 0
    positive_cs_wins = 0
    
    for line in data_lines:
        if not line.strip():
            continue
        
        parts = [p.strip() for p in line.split('|') if p.strip()]
        if len(parts) < 6:
            continue
        
        player, champion, result, cs_diff_10, kills_at_10, assists_at_10 = parts[:6]
        
        cs_diff = int(cs_diff_10)
        
        if cs_diff > 0:
            positive_cs_count += 1
            if result == 'win':
                positive_cs_wins += 1
    
    if positive_cs_count == 0:
        return 0
    
    return positive_cs_wins / positive_cs_count

def q3(text):
    """Return the name of the champion with the most assists at 10 minutes in total."""
    lines = text.strip().split('\n')
    
    # Skip header row
    data_lines = lines[2:]
    
    champion_assists = {}
    
    for line in data_lines:
        if not line.strip():
            continue
        
        parts = [p.strip() for p in line.split('|') if p.strip()]
        if len(parts) < 6:
            continue
        
        player, champion, result, cs_diff_10, kills_at_10, assists_at_10 = parts[:6]
        
        assists = int(assists_at_10)
        
        if champion not in champion_assists:
            champion_assists[champion] = 0
        
        champion_assists[champion] += assists
    
    if not champion_assists:
        return None
    
    return max(champion_assists, key=champion_assists.get)

def q1(text):
    """Return the average damage per minute of players who lost."""
    lines = text.strip().split('\n')
    # Skip header row
    data_lines = lines[1:]
    
    lost_damages = []
    for line in data_lines:
        if not line.strip():
            continue
        parts = line.split('|')
        # Remove empty strings from split (happens due to leading/trailing |)
        parts = [p.strip() for p in parts if p.strip()]
        
        player, team, champion, result, damage_per_min, damage_taken_per_min = parts
        
        if result.lower() == 'loss':
            lost_damages.append(float(damage_per_min))
    
    return sum(lost_damages) / len(lost_damages) if lost_damages else 0

        damage_taken = float(damage_taken_per_min)
        if damage_taken > max_damage_taken:

    return max_player

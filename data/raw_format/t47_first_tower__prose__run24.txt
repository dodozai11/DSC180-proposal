import re

def q1(text):
    """Return the win rate of teams that took the first tower."""
    first_tower_teams = []
    
    # Parse the text to find all game records
    pattern = r"In game (\w+), (.+?) (won|lost); they (took|did not take) the first tower and finished with (\d+) void grubs and (\d+) heralds\."
    matches = re.findall(pattern, text)
    
    for match in matches:
        game_id, team_name, result, tower_status, void_grubs, heralds = match
        took_first_tower = tower_status == "took"
        won = result == "won"
        
        if took_first_tower:
            first_tower_teams.append(won)
    
    if len(first_tower_teams) == 0:
        return 0
    
    win_rate = sum(first_tower_teams) / len(first_tower_teams)
    return win_rate


def q2(text):
    """Return the average number of void grubs taken by winning teams."""
    winning_teams_void_grubs = []
    
    # Parse the text to find all game records
    pattern = r"In game (\w+), (.+?) (won|lost); they (took|did not take) the first tower and finished with (\d+) void grubs and (\d+) heralds\."
    matches = re.findall(pattern, text)
    
    for match in matches:
        game_id, team_name, result, tower_status, void_grubs, heralds = match
        won = result == "won"
        void_grubs_count = int(void_grubs)
        
        if won:
            winning_teams_void_grubs.append(void_grubs_count)
    
    if len(winning_teams_void_grubs) == 0:
        return 0
    
    average = sum(winning_teams_void_grubs) / len(winning_teams_void_grubs)
    return average


def q3(text):
    """Return how many teams took at least one herald and still lost."""
    count = 0
    
    # Parse the text to find all game records
    pattern = r"In game (\w+), (.+?) (won|lost); they (took|did not take) the first tower and finished with (\d+) void grubs and (\d+) heralds\."
    matches = re.findall(pattern, text)
    
    for match in matches:
        game_id, team_name, result, tower_status, void_grubs, heralds = match
        lost = result == "lost"
        heralds_count = int(heralds)
        
        if lost and heralds_count >= 1:
            count += 1
    
    return count

def q1(text):
    """Returns the average gold difference at 15 minutes of the winning teams."""
    lines = text.strip().split('\n')
    # Skip header line
    data_lines = lines[1:]
    
    winning_gold_diffs = []
    for line in data_lines:
        parts = line.split('|')[1:-1]  # Remove empty strings from split
        parts = [p.strip() for p in parts]
        
        game, team, side, result, gold_diff_15 = parts
        gold_diff_15 = int(gold_diff_15)
        
        if result == 'win':
            winning_gold_diffs.append(gold_diff_15)
    
    return sum(winning_gold_diffs) / len(winning_gold_diffs)


def q2(text):
    """Returns how many teams won despite a negative gold difference at 15 minutes."""
    lines = text.strip().split('\n')
    # Skip header line
    data_lines = lines[1:]
    
    count = 0
    for line in data_lines:
        parts = line.split('|')[1:-1]  # Remove empty strings from split
        parts = [p.strip() for p in parts]
        
        game, team, side, result, gold_diff_15 = parts
        gold_diff_15 = int(gold_diff_15)
        
        if result == 'win' and gold_diff_15 < 0:
            count += 1
    
    return count


def q3(text):
    """Returns the win rate of teams that were ahead in gold at 15 minutes."""
    lines = text.strip().split('\n')
    # Skip header line
    data_lines = lines[1:]
    
    ahead_in_gold = 0
    ahead_and_won = 0
    
    for line in data_lines:
        parts = line.split('|')[1:-1]  # Remove empty strings from split
        parts = [p.strip() for p in parts]
        
        game, team, side, result, gold_diff_15 = parts
        gold_diff_15 = int(gold_diff_15)
        
        if gold_diff_15 > 0:
            ahead_in_gold += 1
            if result == 'win':
                ahead_and_won += 1
    
    if ahead_in_gold == 0:
        return 0
    
    return ahead_and_won / ahead_in_gold

def q1(text):
    """Return the average number of kills at 15 minutes of red-side teams."""
    lines = text.strip().split('\n')
    red_kills = []
    
    for line in lines[1:]:  # Skip header
        parts = [p.strip() for p in line.split('|')]
        if len(parts) > 4 and parts[2] == 'red':
            red_kills.append(int(parts[4]))
    
    return sum(red_kills) / len(red_kills) if red_kills else 0


def q2(text):
    """Return the name of the team with the largest experience difference at 15 minutes."""
    lines = text.strip().split('\n')
    max_xp_diff = None
    max_team = None
    
    for line in lines[1:]:  # Skip header
        parts = [p.strip() for p in line.split('|')]
        if len(parts) > 5:
            team = parts[1]
            xp_diff = int(parts[5])
            
            if max_xp_diff is None or abs(xp_diff) > abs(max_xp_diff):
                max_xp_diff = xp_diff
                max_team = team
    
    return max_team


def q3(text):
    """Return how many teams won despite a negative experience difference at 15 minutes."""
    lines = text.strip().split('\n')
    count = 0
    
    for line in lines[1:]:  # Skip header
        parts = [p.strip() for p in line.split('|')]
        if len(parts) > 5:
            result = parts[3]
            xp_diff = int(parts[5])
            
            if result == 'win' and xp_diff < 0:
                count += 1
    
    return count

import json
import re

def q1(text):
    """Return the win rate of teams that took the first tower."""
    data = json.loads(text)
    
    first_tower_teams = [d for d in data if d['first_tower'] == 'yes']
    
    if not first_tower_teams:
        return 0
    
    wins = sum(1 for d in first_tower_teams if d['result'] == 'win')
    return wins / len(first_tower_teams)


def q2(text):
    """Return the average number of void grubs taken by winning teams."""
    data = json.loads(text)
    
    winning_teams = [d for d in data if d['result'] == 'win']
    
    if not winning_teams:
        return 0
    
    total_void_grubs = sum(d['void_grubs'] for d in winning_teams)
    return total_void_grubs / len(winning_teams)


def q3(text):
    """Return how many teams took at least one herald and still lost."""
    data = json.loads(text)
    
    count = sum(1 for d in data if d['heralds'] >= 1 and d['result'] == 'loss')
    return count

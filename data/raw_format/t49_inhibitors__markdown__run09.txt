def q1(text):
    """Return the average number of inhibitors destroyed by winning teams."""
    lines = text.strip().split('\n')
    # Skip header row
    data_lines = lines[1:]
    
    total_inhibitors = 0
    win_count = 0
    
    for line in data_lines:
        parts = line.split('|')
        # Remove empty strings from split and strip whitespace
        parts = [p.strip() for p in parts if p.strip()]
        
        result = parts[2]
        inhibitors = int(parts[4])
        
        if result == 'win':
            total_inhibitors += inhibitors
            win_count += 1
    
    return total_inhibitors / win_count if win_count > 0 else 0

def q3(text):
    """Return the game label of the longest game."""
    lines = text.strip().split('\n')
    # Skip header row
    data_lines = lines[1:]
    
    max_duration = -1
    longest_game = None
    
    for line in data_lines:
        parts = line.split('|')
        # Remove empty strings from split and strip whitespace
        parts = [p.strip() for p in parts if p.strip()]
        
        game = parts[0]
        game_minutes = float(parts[3])
        
        if game_minutes > max_duration:
            max_duration = game_minutes
            longest_game = game
    
    return longest_game

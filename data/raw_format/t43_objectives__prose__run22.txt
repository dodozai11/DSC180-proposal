import re

def parse_data(text):
    """Parse the match data from text format."""
    matches = []
    
    # Split by "In game" to get individual game entries
    game_blocks = text.split("In game ")
    
    for block in game_blocks[1:]:  # Skip the first empty split
        lines = block.split("; ")
        
        # Parse game ID and first team info
        game_info = lines[0]
        game_id = game_info.split(",")[0].strip()
        
        # Parse each team's data
        for i in range(0, len(lines), 2):
            if i + 1 < len(lines):
                team_line = lines[i]
                stats_line = lines[i + 1]
                
                # Extract team name and result
                if i == 0:
                    # First team in the game
                    parts = team_line.split(" won") if " won" in team_line else team_line.split(" lost")
                    team_name = parts[0].strip()
                    result = "won" if " won" in team_line else "lost"
                else:
                    # Second team in the game
                    parts = stats_line.split(" won") if " won" in stats_line else stats_line.split(" lost")
                    team_name = parts[0].strip()
                    result = "won" if " won" in stats_line else "lost"
                    stats_line = parts[1] if len(parts) > 1 else ""
                
                # Extract first dragon status
                first_dragon = "took the first dragon" in team_line or "took the first dragon" in stats_line
                
                # Extract dragons, barons, towers
                dragons = int(re.search(r"(\d+) dragons", stats_line).group(1))
                barons = int(re.search(r"(\d+) barons", stats_line).group(1))
                towers = int(re.search(r"(\d+) towers", stats_line).group(1))
                
                matches.append({
                    'game_id': game_id,
                    'team': team_name,
                    'result': result,
                    'first_dragon': first_dragon,
                    'dragons': dragons,
                    'barons': barons,
                    'towers': towers
                })
    
    return matches


def parse_data_v2(text):
    """Better parsing approach."""
    matches = []
    
    # Find all game entries
    pattern = r"In game (\w+), ([^;]+) (won|lost); they (did not take|took) the first dragon and finished with (\d+) dragons, (\d+) barons and (\d+) towers"
    
    for match in re.finditer(pattern, text):
        game_id = match.group(1)
        team_name = match.group(2)
        result = match.group(3)
        first_dragon = match.group(4) == "took"
        dragons = int(match.group(5))
        barons = int(match.group(6))
        towers = int(match.group(7))
        
        matches.append({
            'game_id': game_id,
            'team': team_name,
            'result': result,
            'first_dragon': first_dragon,
            'dragons': dragons,
            'barons': barons,
            'towers': towers
        })
    
    return matches


def q1(text):
    """Return the win rate of teams that took the first dragon."""
    matches = parse_data_v2(text)
    
    first_dragon_teams = [m for m in matches if m['first_dragon']]
    
    if not first_dragon_teams:
        return 0.0
    
    wins = sum(1 for m in first_dragon_teams if m['result'] == 'won')
    
    return wins / len(first_dragon_teams)


def q2(text):
    """Return the average number of barons taken by winning teams."""
    matches = parse_data_v2(text)
    
    winning_teams = [m for m in matches if m['result'] == 'won']
    
    if not winning_teams:
        return 0.0
    
    total_barons = sum(m['barons'] for m in winning_teams)
    
    return total_barons / len(winning_teams)


def q3(text):
    """Return the total number of towers destroyed across all rows."""
    matches = parse_data_v2(text)
    
    total_towers = sum(m['towers'] for m in matches)
    
    return total_towers

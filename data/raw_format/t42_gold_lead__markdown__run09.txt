def q1(text):
    """Returns the average gold difference at 15 minutes of the winning teams."""
    lines = text.strip().split('\n')
    # Skip header row
    winning_gold_diffs = []
    
    for line in lines[1:]:
        parts = line.split('|')[1:-1]  # Remove empty strings from split
        parts = [p.strip() for p in parts]
        
        game, team, side, result, gold_diff_15 = parts
        
        if result == 'win':
            winning_gold_diffs.append(int(gold_diff_15))
    
    return sum(winning_gold_diffs) / len(winning_gold_diffs)


def q2(text):
    """Returns how many teams won despite a negative gold difference at 15 minutes."""
    lines = text.strip().split('\n')
    # Skip header row
    count = 0
    
    for line in lines[1:]:
        parts = line.split('|')[1:-1]  # Remove empty strings from split
        parts = [p.strip() for p in parts]
        
        game, team, side, result, gold_diff_15 = parts
        
        if result == 'win' and int(gold_diff_15) < 0:
            count += 1
    
    return count


def q3(text):
    """Returns the win rate of teams that were ahead in gold at 15 minutes."""
    lines = text.strip().split('\n')
    # Skip header row
    ahead_in_gold_count = 0
    ahead_and_won_count = 0
    
    for line in lines[1:]:
        parts = line.split('|')[1:-1]  # Remove empty strings from split
        parts = [p.strip() for p in parts]
        
        game, team, side, result, gold_diff_15 = parts
        
        gold_diff = int(gold_diff_15)
        
        if gold_diff > 0:
            ahead_in_gold_count += 1
            if result == 'win':
                ahead_and_won_count += 1
    
    if ahead_in_gold_count == 0:
        return 0
    
    return ahead_and_won_count / ahead_in_gold_count

def q1(text):
    """Returns the average number of kills at 15 minutes of red-side teams."""
    lines = text.strip().split('\n')
    # Skip header row
    data_lines = lines[1:]
    
    red_kills = []
    for line in data_lines:
        parts = [p.strip() for p in line.split('|')]
        # Filter out empty strings from split
        parts = [p for p in parts if p]
        
        side = parts[2]
        kills_at_15 = int(parts[4])
        
        if side == 'red':
            red_kills.append(kills_at_15)
    
    if len(red_kills) == 0:
        return 0
    
    return sum(red_kills) / len(red_kills)


def q2(text):
    """Returns the name of the team with the largest experience difference at 15 minutes."""
    lines = text.strip().split('\n')
    # Skip header row
    data_lines = lines[1:]
    
    max_xp_diff = float('-inf')
    max_team = None
    
    for line in data_lines:
        parts = [p.strip() for p in line.split('|')]
        # Filter out empty strings from split
        parts = [p for p in parts if p]
        
        team = parts[1]
        xp_diff_15 = int(parts[5])
        
        if xp_diff_15 > max_xp_diff:
            max_xp_diff = xp_diff_15
            max_team = team
    
    return max_team


def q3(text):
    """Returns how many teams won despite a negative experience difference at 15 minutes."""
    lines = text.strip().split('\n')
    # Skip header row
    data_lines = lines[1:]
    
    count = 0
    for line in data_lines:
        parts = [p.strip() for p in line.split('|')]
        # Filter out empty strings from split
        parts = [p for p in parts if p]
        
        result = parts[3]
        xp_diff_15 = int(parts[5])
        
        if result == 'win' and xp_diff_15 < 0:
            count += 1
    
    return count

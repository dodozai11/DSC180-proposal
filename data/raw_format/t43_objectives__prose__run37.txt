```python
import re

def parse_data(text):
    """Parse the match data from text format."""
    matches = []
    
    # Split by "In game" to get each game's data
    games = re.split(r'In game ', text)
    
    for game_text in games[1:]:  # Skip the first empty split
        lines = game_text.split('; ')
        
        # First line contains game ID and first team info
        first_line = lines[0]
        match_obj = re.match(r'(\w+),\s+(.+?)\s+(won|lost)', first_line)
        if match_obj:
            game_id = match_obj.group(1)
            team1 = match_obj.group(2)
            result1 = match_obj.group(3)
            
            # Parse team1 stats from remaining parts of first line
            rest_of_first = first_line[match_obj.end():]
            
            # Extract first dragon info
            first_dragon_match = re.search(r'(did not take|took) the first dragon', rest_of_first)
            team1_first_dragon = first_dragon_match.group(1) == 'took' if first_dragon_match else False
            
            # Extract dragons, barons, towers from the stats line
            stats_match = re.search(r'finished with (\d+) dragons?, (\d+) barons? and (\d+) towers?', rest_of_first)
            if stats_match:
                team1_dragons = int(stats_match.group(1))
                team1_barons = int(stats_match.group(2))
                team1_towers = int(stats_match.group(3))
            
            # Second team info is in the next line
            second_line = lines[1]
            match_obj2 = re.match(r'(.+?)\s+(won|lost)', second_line)
            if match_obj2:
                team2 = match_obj2.group(1)
                result2 = match_obj2.group(2)
                
                # Extract first dragon info for team2
                first_dragon_match2 = re.search(r'(did not take|took) the first dragon', second_line)
                team2_first_dragon = first_dragon_match2.group(1) == 'took' if first_dragon_match2 else False
                
                # Extract dragons, barons, towers from team2 stats
                stats_match2 = re.search(r'finished with (\d+) dragons?, (\d+) barons? and (\d+) towers?', second_line)
                if stats_match2:
                    team2_dragons = int(stats_match2.group(1))
                    team2_barons = int(stats_match2.group(2))
                    team2_towers = int(stats_match2.group(3))
                
                matches.append({
                    'game_id': game_id,
                    'team1': team1,
                    'team1_won': result1 == 'won',
                    'team1_first_dragon': team1_first_dragon,
                    'team1_dragons': team1_dragons,
                    'team1_barons': team1_barons,
                    'team1_towers': team1_towers,
                    'team2': team2,
                    'team2_won': result2 == 'won',
                    'team2_first_dragon': team2_first_dragon,
                    'team2_dragons': team2_dragons,
                    'team2_barons': team2_barons,
                    'team2_towers': team2_towers,
                })
    
    return matches

def q1(text):
    """Return the win rate of teams that took the first dragon."""
    matches = parse_data(text)
    
    first_dragon_wins = 0
    first_dragon_total = 0
    
    for match in matches:
        if match['team1_first_dragon']:
            first_dragon_total += 1
            if match['team1_won']:
                first_dragon_wins += 1
        
        if match['team2_first_dragon']:
            first_dragon_total += 1
            if match['team2_won']:
                first_dragon_wins += 1
    
    if first_dragon_total == 0:
        return 0
    
    return first_dragon_wins / first_dragon_total

def q2(text):
    """Return the average number of barons taken by winning teams."""
    matches = parse_data(text)
    
    winning_barons = []
    
    for match in matches:
        if match['team1_won']:
            winning_barons.append(match['team1_barons'])
        if match['team2_won']:
            winning_barons.append(match['team2_barons'])
    
    if len(winning_barons) == 0:
        return 0
    
    return sum(winning_barons) / len(winning_barons)

def q3(text):
    """Return the total number of towers destroyed across all rows."""
    matches = parse_data(text)
    
    total_towers = 0
    
    for match in matches:
        total_towers += match['team1_towers']
        total_towers += match['team2_towers']
    
    return total_towers
```

```python
import json
import re
from collections import Counter

def q1(text):
    """Return the name of the player with the highest KDA."""
    data = json.loads(text)
    
    max_kda = -1
    best_player = None
    
    for player_data in data:
        kills = player_data['kills']
        assists = player_data['assists']
        deaths = player_data['deaths']
        
        kda = (kills + assists) / max(deaths, 1)
        
        if kda > max_kda:
            max_kda = kda
            best_player = player_data['player']
    
    return best_player


def q2(text):
    """Return the average number of kills in games that were won."""
    data = json.loads(text)
    
    winning_kills = []
    
    for player_data in data:
        if player_data['result'] == 'win':
            winning_kills.append(player_data['kills'])
    
    if not winning_kills:
        return 0
    
    return sum(winning_kills) / len(winning_kills)


def q3(text):
    """Return the name of the most-picked champion."""
    data = json.loads(text)
    
    champions = [player_data['champion'] for player_data in data]
    champion_counts = Counter(champions)
    
    most_picked = champion_counts.most_common(1)[0][0]
    
    return most_picked
```

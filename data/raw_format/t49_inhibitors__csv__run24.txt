def q1(text):
    """Return the average number of inhibitors destroyed by winning teams."""
    lines = text.strip().split('\n')
    header = lines[0].split(',')
    
    inhibitors_index = header.index('inhibitors')
    result_index = header.index('result')
    
    winning_inhibitors = []
    for line in lines[1:]:
        values = line.split(',')
        if values[result_index] == 'win':
            winning_inhibitors.append(int(values[inhibitors_index]))
    
    return sum(winning_inhibitors) / len(winning_inhibitors) if winning_inhibitors else 0


def q2(text):
    """Return the average damage to towers of losing teams."""
    lines = text.strip().split('\n')
    header = lines[0].split(',')
    
    tower_damage_index = header.index('tower_damage')
    result_index = header.index('result')
    
    losing_damages = []
    for line in lines[1:]:
        values = line.split(',')
        if values[result_index] == 'loss':
            losing_damages.append(int(values[tower_damage_index]))
    
    return sum(losing_damages) / len(losing_damages) if losing_damages else 0


def q3(text):
    """Return the game label of the longest game."""
    lines = text.strip().split('\n')
    header = lines[0].split(',')
    
    game_index = header.index('game')
    game_minutes_index = header.index('game_minutes')
    
    max_duration = -1
    longest_game = None
    
    for line in lines[1:]:
        values = line.split(',')
        game_label = values[game_index]
        duration = float(values[game_minutes_index])
        
        if duration > max_duration:
            max_duration = duration
            longest_game = game_label
    
    return longest_game

import re
import anthropic


def q1(text: str) -> float:
    """Returns the average damage per minute of players who lost."""
    client = anthropic.Anthropic()
    
    message = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": f"""Parse the following League of Legends esports match data and calculate the average damage per minute of players who lost.

Data:
{text}

Return ONLY a single number representing the average damage per minute of players who lost, with no other text or explanation."""
            }
        ]
    )
    
    result = message.content[0].text.strip()
    return float(result)

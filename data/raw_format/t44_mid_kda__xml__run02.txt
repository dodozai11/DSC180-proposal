import xml.etree.ElementTree as ET
from collections import Counter

def q1(text):
    """Return the name of the player with the highest KDA."""
    root = ET.fromstring(f"<root>{text}</root>")
    
    max_kda = -1
    best_player = None
    
    for row in root.findall("row"):
        player = row.find("player").text
        kills = int(row.find("kills").text)
        deaths = int(row.find("deaths").text)
        assists = int(row.find("assists").text)
        
        kda = (kills + assists) / max(deaths, 1)
        
        if kda > max_kda:
            max_kda = kda
            best_player = player
    
    return best_player


def q2(text):
    """Return the average number of kills in games that were won."""
    root = ET.fromstring(f"<root>{text}</root>")
    
    win_kills = []
    
    for row in root.findall("row"):
        result = row.find("result").text
        kills = int(row.find("kills").text)
        
        if result == "win":
            win_kills.append(kills)
    
    if len(win_kills) == 0:
        return 0
    
    return sum(win_kills) / len(win_kills)


def q3(text):
    """Return the name of the most-picked champion."""
    root = ET.fromstring(f"<root>{text}</root>")
    
    champions = []
    
    for row in root.findall("row"):
        champion = row.find("champion").text
        champions.append(champion)
    
    counter = Counter(champions)
    most_common = counter.most_common(1)[0][0]
    
    return most_common

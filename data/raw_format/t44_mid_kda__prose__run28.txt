import re
from collections import Counter

def parse_data(text):
    """Parse the match data from text format."""
    players = []
    
    # Pattern to match each player's data
    pattern = r"(\w+) of ([\w\s]+) played ([\w']+) in \w+ and (\w+) with (\d+) kills, (\d+) deaths and (\d+) assists"
    
    matches = re.findall(pattern, text)
    
    for match in matches:
        player_name = match[0]
        team = match[1]
        champion = match[2]
        result = match[3]
        kills = int(match[4])
        deaths = int(match[5])
        assists = int(match[6])
        
        players.append({
            'name': player_name,
            'team': team,
            'champion': champion,
            'result': result,
            'kills': kills,
            'deaths': deaths,
            'assists': assists
        })
    
    return players

def q1(text):
    """Return the name of the player with the highest KDA."""
    players = parse_data(text)
    
    max_kda = -1
    best_player = None
    
    for player in players:
        kda = (player['kills'] + player['assists']) / max(player['deaths'], 1)
        if kda > max_kda:
            max_kda = kda
            best_player = player['name']
    
    return best_player

def q2(text):
    """Return the average number of kills in games that were won."""
    players = parse_data(text)
    
    won_games = [p for p in players if p['result'].lower() == 'won']
    
    if not won_games:
        return 0
    
    total_kills = sum(p['kills'] for p in won_games)
    return total_kills / len(won_games)

def q3(text):
    """Return the name of the most-picked champion."""
    players = parse_data(text)
    
    champions = [p['champion'] for p in players]
    champion_counts = Counter(champions)
    
    most_picked = champion_counts.most_common(1)[0][0]
    return most_picked

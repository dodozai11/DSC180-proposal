def q1(text):
    lines = text.strip().split('\n')
    header = lines[0].split('|')
    header = [h.strip() for h in header if h.strip()]
    
    max_cs = -1
    max_player = None
    
    for line in lines[2:]:
        if not line.strip():
            continue
        cells = line.split('|')
        cells = [c.strip() for c in cells if c.strip()]
        
        player = cells[0]
        cs_per_min = float(cells[4])
        
        if cs_per_min > max_cs:
            max_cs = cs_per_min
            max_player = player
    
    return max_player


def q2(text):
    lines = text.strip().split('\n')
    
    total_gold_share = 0
    win_count = 0
    
    for line in lines[2:]:
        if not line.strip():
            continue
        cells = line.split('|')
        cells = [c.strip() for c in cells if c.strip()]
        
        result = cells[3]
        gold_share = float(cells[5])
        
        if result == 'win':
            total_gold_share += gold_share
            win_count += 1
    
    return total_gold_share / win_count if win_count > 0 else 0


def q3(text):
    lines = text.strip().split('\n')
    
    count = 0
    
    for line in lines[2:]:
        if not line.strip():
            continue
        cells = line.split('|')
        cells = [c.strip() for c in cells if c.strip()]
        
        cs_per_min = float(cells[4])
        
        if cs_per_min > 9:
            count += 1
    
    return count

import re

def q1(text):
    """Return the average number of inhibitors destroyed by winning teams."""
    lines = text.strip().split('\n')
    header = lines[0]
    
    winning_inhibitors = []
    
    for line in lines[1:]:
        parts = [p.strip() for p in line.split('|') if p.strip()]
        if len(parts) >= 5:
            game, team, result, game_minutes, inhibitors = parts[0], parts[1], parts[2], parts[3], parts[4]
            if result.lower() == 'win':
                winning_inhibitors.append(int(inhibitors))
    
    if winning_inhibitors:
        return sum(winning_inhibitors) / len(winning_inhibitors)
    return 0

def q2(text):
    """Return the average damage to towers of losing teams."""
    lines = text.strip().split('\n')
    header = lines[0]
    
    losing_tower_damage = []
    
    for line in lines[1:]:
        parts = [p.strip() for p in line.split('|') if p.strip()]
        if len(parts) >= 6:
            game, team, result, game_minutes, inhibitors, tower_damage = parts[0], parts[1], parts[2], parts[3], parts[4], parts[5]
            if result.lower() == 'loss':
                losing_tower_damage.append(int(tower_damage))
    
    if losing_tower_damage:
        return sum(losing_tower_damage) / len(losing_tower_damage)
    return 0

def q3(text):
    """Return the game label of the longest game."""
    lines = text.strip().split('\n')
    header = lines[0]
    
    longest_game = None
    longest_duration = -1
    
    for line in lines[1:]:
        parts = [p.strip() for p in line.split('|') if p.strip()]
        if len(parts) >= 4:
            game, team, result, game_minutes = parts[0], parts[1], parts[2], parts[3]
            duration = float(game_minutes)
            if duration > longest_duration:
                longest_duration = duration
                longest_game = game
    
    return longest_game

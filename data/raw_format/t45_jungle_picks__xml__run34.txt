```python
import re
from collections import defaultdict

def q1(text):
    """Returns the number of different champions played."""
    champions = set()
    for match in re.finditer(r'<champion>([^<]+)</champion>', text):
        champions.add(match.group(1))
    return len(champions)


def q2(text):
    """Returns the name of the champion with the best win rate among champions played at least twice."""
    champion_stats = defaultdict(lambda: {'wins': 0, 'total': 0})
    
    # Find all player rows
    for match in re.finditer(r'<champion>([^<]+)</champion>.*?<result>([^<]+)</result>', text):
        champion = match.group(1)
        result = match.group(2)
        
        champion_stats[champion]['total'] += 1
        if result == 'win':
            champion_stats[champion]['wins'] += 1
    
    # Filter champions played at least twice
    qualified_champions = {champ: stats for champ, stats in champion_stats.items() if stats['total'] >= 2}
    
    if not qualified_champions:
        return None
    
    # Find champion with best win rate
    best_champion = max(qualified_champions.items(), 
                       key=lambda x: x[1]['wins'] / x[1]['total'])
    
    return best_champion[0]


def q3(text):
    """Returns the average number of deaths in games that were lost."""
    losses = []
    
    # Find all rows with loss result and their deaths
    for match in re.finditer(r'<result>loss</result>.*?<deaths>(\d+)</deaths>', text):
        deaths = int(match.group(1))
        losses.append(deaths)
    
    if not losses:
        return 0
    
    return sum(losses) / len(losses)
```

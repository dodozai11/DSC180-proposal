def q1(text):
    lines = text.strip().split('\n')
    header = lines[0].split('|')[1:-1]
    header = [h.strip() for h in header]
    
    positive_cs_diff = []
    for line in lines[2:]:
        if line.strip() == '':
            continue
        parts = [p.strip() for p in line.split('|')[1:-1]]
        if len(parts) == len(header):
            row = dict(zip(header, parts))
            cs_diff = int(row['cs_diff_10'])
            if cs_diff > 0:
                result = row['result']
                positive_cs_diff.append(result)
    
    if len(positive_cs_diff) == 0:
        return 0
    
    wins = sum(1 for r in positive_cs_diff if r == 'win')
    return wins / len(positive_cs_diff)

def q3(text):
    lines = text.strip().split('\n')
    header = lines[0].split('|')[1:-1]
    header = [h.strip() for h in header]
    
    champion_assists = {}
    for line in lines[2:]:
        if line.strip() == '':
            continue
        parts = [p.strip() for p in line.split('|')[1:-1]]
        if len(parts) == len(header):
            row = dict(zip(header, parts))
            champion = row['champion']
            assists = int(row['assists_at_10'])
            if champion not in champion_assists:
                champion_assists[champion] = 0
            champion_assists[champion] += assists
    
    if not champion_assists:
        return None
    
    return max(champion_assists, key=champion_assists.get)

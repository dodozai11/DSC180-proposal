```python
import re

def parse_data(text):
    """Parse the text data into a list of game records."""
    games = []
    # Pattern to match each game record
    pattern = r"In game (\w+), (.+?) played on the (\w+) side and (won|lost), with a gold difference of ([-\d]+) at 15 minutes\."
    matches = re.findall(pattern, text)
    
    for match in matches:
        game_id, team, side, result, gold_diff = match
        games.append({
            'game_id': game_id,
            'team': team,
            'side': side,
            'result': result,
            'gold_diff': int(gold_diff)
        })
    
    return games

def q1(text):
    """Return the average gold difference at 15 minutes of the winning teams."""
    games = parse_data(text)
    winning_gold_diffs = [game['gold_diff'] for game in games if game['result'] == 'won']
    
    if not winning_gold_diffs:
        return 0
    
    return sum(winning_gold_diffs) / len(winning_gold_diffs)

def q2(text):
    """Return how many teams won despite a negative gold difference at 15 minutes."""
    games = parse_data(text)
    won_with_negative = sum(1 for game in games if game['result'] == 'won' and game['gold_diff'] < 0)
    
    return won_with_negative

def q3(text):
    """Return the win rate of teams that were ahead in gold at 15 minutes."""
    games = parse_data(text)
    
    # Teams ahead in gold (positive gold difference)
    ahead_in_gold = [game for game in games if game['gold_diff'] > 0]
    
    if not ahead_in_gold:
        return 0
    
    wins = sum(1 for game in ahead_in_gold if game['result'] == 'won')
    win_rate = wins / len(ahead_in_gold)
    
    return win_rate
```

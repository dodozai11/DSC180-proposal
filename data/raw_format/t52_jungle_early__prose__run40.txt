```python
import re
from collections import defaultdict

def parse_data(text):
    """Parse the match data and return a list of player records."""
    pattern = r"(\w+(?:\s+\w+)*?)\s+played\s+(\w+)\s+in\s+the\s+jungle\s+and\s+(won|lost);\s+at\s+10\s+minutes\s+they\s+had\s+a\s+CS\s+difference\s+of\s+(-?\d+),\s+(\d+)\s+kills\s+and\s+(\d+)\s+assists\."
    matches = re.findall(pattern, text)
    
    records = []
    for match in matches:
        player_name, champion, result, cs_diff, kills, assists = match
        records.append({
            'player': player_name.strip(),
            'champion': champion,
            'result': result,
            'cs_diff': int(cs_diff),
            'kills': int(kills),
            'assists': int(assists)
        })
    
    return records


def q1(text):
    """Return the win rate of players with a positive CS difference at 10 minutes."""
    records = parse_data(text)
    
    positive_cs_players = [r for r in records if r['cs_diff'] > 0]
    
    if len(positive_cs_players) == 0:
        return 0
    
    wins = sum(1 for r in positive_cs_players if r['result'] == 'won')
    win_rate = wins / len(positive_cs_players)
    
    return win_rate


def q2(text):
    """Return the total number of kills at 10 minutes across all rows."""
    records = parse_data(text)
    total_kills = sum(r['kills'] for r in records)
    return total_kills


def q3(text):
    """Return the name of the champion with the most assists at 10 minutes in total."""
    records = parse_data(text)
    
    champion_assists = defaultdict(int)
    for r in records:
        champion_assists[r['champion']] += r['assists']
    
    champion_with_most_assists = max(champion_assists, key=champion_assists.get)
    return champion_with_most_assists
```

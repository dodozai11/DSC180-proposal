def q1(text):
    """Returns the average gold difference at 15 minutes of the winning teams."""
    lines = text.strip().split('\n')[1:]  # Skip header
    winning_gold_diffs = []
    
    for line in lines:
        parts = line.split(',')
        result = parts[3]
        gold_diff_15 = int(parts[4])
        
        if result == 'win':
            winning_gold_diffs.append(gold_diff_15)
    
    return sum(winning_gold_diffs) / len(winning_gold_diffs) if winning_gold_diffs else 0


def q2(text):
    """Returns how many teams won despite a negative gold difference at 15 minutes."""
    lines = text.strip().split('\n')[1:]  # Skip header
    count = 0
    
    for line in lines:
        parts = line.split(',')
        result = parts[3]
        gold_diff_15 = int(parts[4])
        
        if result == 'win' and gold_diff_15 < 0:
            count += 1
    
    return count


def q3(text):
    """Returns the win rate of teams that were ahead in gold at 15 minutes."""
    lines = text.strip().split('\n')[1:]  # Skip header
    ahead_in_gold = 0
    wins_while_ahead = 0
    
    for line in lines:
        parts = line.split(',')
        result = parts[3]
        gold_diff_15 = int(parts[4])
        
        if gold_diff_15 > 0:  # Team was ahead in gold
            ahead_in_gold += 1
            if result == 'win':
                wins_while_ahead += 1
    
    return wins_while_ahead / ahead_in_gold if ahead_in_gold > 0 else 0

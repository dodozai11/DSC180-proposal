import xml.etree.ElementTree as ET

def q1(text):
    """Return the average number of inhibitors destroyed by winning teams."""
    root = ET.fromstring(f"<root>{text}</root>")
    winning_inhibitors = []
    
    for row in root.findall('row'):
        if row.find('result').text == 'win':
            inhibitors = int(row.find('inhibitors').text)
            winning_inhibitors.append(inhibitors)
    
    return sum(winning_inhibitors) / len(winning_inhibitors) if winning_inhibitors else 0

def q2(text):
    """Return the average damage to towers of losing teams."""
    root = ET.fromstring(f"<root>{text}</root>")
    losing_damages = []
    
    for row in root.findall('row'):
        if row.find('result').text == 'loss':
            tower_damage = int(row.find('tower_damage').text)
            losing_damages.append(tower_damage)
    
    return sum(losing_damages) / len(losing_damages) if losing_damages else 0

def q3(text):
    """Return the game label of the longest game."""
    root = ET.fromstring(f"<root>{text}</root>")
    max_duration = -1
    longest_game = None
    
    for row in root.findall('row'):
        game_minutes = float(row.find('game_minutes').text)
        if game_minutes > max_duration:
            max_duration = game_minutes
            longest_game = row.find('game').text
    
    return longest_game

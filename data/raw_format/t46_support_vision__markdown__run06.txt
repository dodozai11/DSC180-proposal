def q1(text):
    lines = text.strip().split('\n')
    # Skip header row
    data = []
    for line in lines[1:]:
        parts = [p.strip() for p in line.split('|') if p.strip()]
        if len(parts) >= 6:
            data.append({
                'player': parts[0],
                'team': parts[1],
                'result': parts[2],
                'vision_score': int(parts[3]),
                'wards_placed': int(parts[4]),
                'control_wards': int(parts[5])
            })
    
    wins = [d['vision_score'] for d in data if d['result'] == 'win']
    losses = [d['vision_score'] for d in data if d['result'] == 'loss']
    
    avg_win = sum(wins) / len(wins) if wins else 0
    avg_loss = sum(losses) / len(losses) if losses else 0
    
    return avg_win - avg_loss

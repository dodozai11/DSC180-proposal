def q1(text):
    lines = text.strip().split('\n')[1:]  # Skip header
    
    first_dragon_wins = 0
    first_dragon_total = 0
    
    for line in lines:
        parts = line.split(',')
        first_dragon = parts[3]
        result = parts[2]
        
        if first_dragon == 'yes':
            first_dragon_total += 1
            if result == 'win':
                first_dragon_wins += 1
    
    if first_dragon_total == 0:
        return 0
    
    return first_dragon_wins / first_dragon_total


def q2(text):
    lines = text.strip().split('\n')[1:]  # Skip header
    
    winning_barons = []
    
    for line in lines:
        parts = line.split(',')
        result = parts[2]
        barons = int(parts[5])
        
        if result == 'win':
            winning_barons.append(barons)
    
    if len(winning_barons) == 0:
        return 0
    
    return sum(winning_barons) / len(winning_barons)


def q3(text):
    lines = text.strip().split('\n')[1:]  # Skip header
    
    total_towers = 0
    
    for line in lines:
        parts = line.split(',')
        towers = int(parts[6])
        total_towers += towers
    
    return total_towers

import re
from typing import List, Dict

def parse_data(text: str) -> List[Dict[str, str]]:
    """Parse XML-like text data into a list of dictionaries."""
    rows = []
    row_pattern = r'<row>(.*?)</row>'
    for row_match in re.finditer(row_pattern, text, re.DOTALL):
        row_content = row_match.group(1)
        row_data = {}
        field_pattern = r'<(\w+)>([^<]+)</\1>'
        for field_match in re.finditer(field_pattern, row_content):
            key = field_match.group(1)
            value = field_match.group(2)
            row_data[key] = value
        rows.append(row_data)
    return rows


def q1(text: str) -> float:
    """Return the average vision score of winners minus average vision score of losers."""
    rows = parse_data(text)
    
    winners_vision = []
    losers_vision = []
    
    for row in rows:
        vision_score = int(row['vision_score'])
        if row['result'] == 'win':
            winners_vision.append(vision_score)
        else:
            losers_vision.append(vision_score)
    
    avg_winners = sum(winners_vision) / len(winners_vision) if winners_vision else 0
    avg_losers = sum(losers_vision) / len(losers_vision) if losers_vision else 0
    
    return avg_winners - avg_losers


def q2(text: str) -> str:
    """Return the name of the player who placed the most wards."""
    rows = parse_data(text)
    
    max_wards = -1
    max_player = None
    
    for row in rows:
        wards_placed = int(row['wards_placed'])
        if wards_placed > max_wards:
            max_wards = wards_placed
            max_player = row['player']
    
    return max_player


def q3(text: str) -> int:
    """Return how many players bought at least 15 control wards."""
    rows = parse_data(text)
    
    count = 0
    for row in rows:
        control_wards = int(row['control_wards'])
        if control_wards >= 15:
            count += 1
    
    return count

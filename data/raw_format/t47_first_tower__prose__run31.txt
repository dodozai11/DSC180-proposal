```python
import re

def q1(text):
    """Return the win rate of teams that took the first tower."""
    first_tower_teams = []
    
    # Parse all match data
    matches = re.findall(
        r'In game (G\d+), (.+?) (won|lost); they (took|did not take) the first tower',
        text
    )
    
    for game, team, result, tower_status in matches:
        if tower_status == "took":
            won = result == "won"
            first_tower_teams.append(won)
    
    if not first_tower_teams:
        return 0
    
    wins = sum(first_tower_teams)
    return wins / len(first_tower_teams)


def q2(text):
    """Return the average number of void grubs taken by winning teams."""
    winning_void_grubs = []
    
    # Parse all match data with void grubs and heralds
    matches = re.findall(
        r'In game (G\d+), (.+?) (won|lost); they (?:took|did not take) the first tower and finished with (\d+) void grubs',
        text
    )
    
    for game, team, result, void_grubs in matches:
        if result == "won":
            winning_void_grubs.append(int(void_grubs))
    
    if not winning_void_grubs:
        return 0
    
    return sum(winning_void_grubs) / len(winning_void_grubs)


def q3(text):
    """Return how many teams took at least one herald and still lost."""
    lost_with_herald = 0
    
    # Parse all match data with heralds
    matches = re.findall(
        r'In game (G\d+), (.+?) (won|lost); they (?:took|did not take) the first tower and finished with \d+ void grubs and (\d+) heralds',
        text
    )
    
    for game, team, result, heralds in matches:
        if result == "lost" and int(heralds) >= 1:
            lost_with_herald += 1
    
    return lost_with_herald
```

```python
import re

def parse_data(text):
    """Parse the match data and return a list of player dictionaries."""
    players = []
    # Pattern to match: "Name of Team played Champion in lane and (won|lost), dealing X dpm and taking Y dpm"
    pattern = r'(.+?)\s+of\s+(.+?)\s+played\s+(.+?)\s+in\s+the\s+(\w+)\s+lane\s+and\s+(won|lost),\s+dealing\s+([\d.]+)\s+damage\s+per\s+minute\s+and\s+taking\s+([\d.]+)\s+damage\s+per\s+minute'
    
    matches = re.finditer(pattern, text)
    for match in matches:
        player = {
            'name': match.group(1),
            'team': match.group(2),
            'champion': match.group(3),
            'lane': match.group(4),
            'result': match.group(5),
            'dpm_dealt': float(match.group(6)),
            'dpm_taken': float(match.group(7))
        }
        players.append(player)
    
    return players

def q1(text):
    """Return the average damage per minute of players who lost."""
    players = parse_data(text)
    lost_players = [p for p in players if p['result'] == 'lost']
    
    if not lost_players:
        return 0
    
    avg_dpm = sum(p['dpm_dealt'] for p in lost_players) / len(lost_players)
    return avg_dpm

def q2(text):
    """Return the name of the player who took the most damage per minute."""
    players = parse_data(text)
    
    if not players:
        return None
    
    max_player = max(players, key=lambda p: p['dpm_taken'])
    return max_player['name']

def q3(text):
    """Return the number of different teams."""
    players = parse_data(text)
    teams = set(p['team'] for p in players)
    return len(teams)
```

def q1(text):
    """Return the win rate of teams that took the first dragon."""
    lines = text.strip().split('\n')
    # Skip header row
    data = []
    for line in lines[1:]:
        parts = [p.strip() for p in line.split('|') if p.strip()]
        if len(parts) >= 7:
            data.append(parts)
    
    first_dragon_wins = 0
    first_dragon_total = 0
    
    for row in data:
        game, team, result, first_dragon, dragons, barons, towers = row[0], row[1], row[2], row[3], row[4], row[5], row[6]
        if first_dragon == "yes":
            first_dragon_total += 1
            if result == "win":
                first_dragon_wins += 1
    
    if first_dragon_total == 0:
        return 0
    return first_dragon_wins / first_dragon_total


def q2(text):
    """Return the average number of barons taken by winning teams."""
    lines = text.strip().split('\n')

    data = []
    for line in lines[1:]:
        parts = [p.strip() for p in line.split('|') if p.strip()]

            data.append(parts)
    
    winning_barons = []
    
    for row in data:
        game, team, result, first_dragon, dragons, barons, towers = row[0], row[1], row[2], row[3], row[4], row[5], row[6]
        if result == "win":
            winning_barons.append(int(barons))
    
    if len(winning_barons) == 0:
        return 0
    return sum(winning_barons) / len(winning_barons)

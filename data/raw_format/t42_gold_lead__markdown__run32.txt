def parse_data(text):
    """Parse the data from the text format into a list of dictionaries."""
    lines = text.strip().split('\n')
    headers = [h.strip() for h in lines[0].split('|')[1:-1]]
    
    data = []
    for line in lines[2:]:
        if line.strip():
            values = [v.strip() for v in line.split('|')[1:-1]]
            row = {headers[i]: values[i] for i in range(len(headers))}
            row['gold_diff_15'] = int(row['gold_diff_15'])
            data.append(row)
    
    return data


def q1(text):
    """Return the average gold difference at 15 minutes of the winning teams."""
    data = parse_data(text)
    winning_teams = [row['gold_diff_15'] for row in data if row['result'] == 'win']
    return sum(winning_teams) / len(winning_teams) if winning_teams else 0


def q2(text):
    """Return how many teams won despite a negative gold difference at 15 minutes."""
    data = parse_data(text)
    count = sum(1 for row in data if row['result'] == 'win' and row['gold_diff_15'] < 0)
    return count


def q3(text):
    """Return the win rate of teams that were ahead in gold at 15 minutes."""
    data = parse_data(text)
    ahead_teams = [row for row in data if row['gold_diff_15'] > 0]
    
    if not ahead_teams:
        return 0
    
    wins = sum(1 for row in ahead_teams if row['result'] == 'win')
    return wins / len(ahead_teams)

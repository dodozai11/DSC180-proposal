def q1(text):
    lines = text.strip().split('\n')
    header = lines[0]
    rows = [line.split(',') for line in lines[1:]]
    
    positive_cs_diff_rows = []
    for row in rows:
        cs_diff_10 = int(row[3])
        if cs_diff_10 > 0:
            positive_cs_diff_rows.append(row)
    
    if len(positive_cs_diff_rows) == 0:
        return 0
    
    wins = sum(1 for row in positive_cs_diff_rows if row[2] == 'win')
    return wins / len(positive_cs_diff_rows)


def q2(text):
    lines = text.strip().split('\n')
    rows = [line.split(',') for line in lines[1:]]
    
    total_kills = sum(int(row[4]) for row in rows)
    return total_kills


def q3(text):
    lines = text.strip().split('\n')
    rows = [line.split(',') for line in lines[1:]]
    
    champion_assists = {}
    for row in rows:
        champion = row[1]
        assists = int(row[5])
        if champion not in champion_assists:
            champion_assists[champion] = 0
        champion_assists[champion] += assists
    
    return max(champion_assists, key=champion_assists.get)

def q1(text):
    """Return the win rate of teams that took the first tower."""
    lines = text.strip().split('\n')
    
    # Skip header
    data = []
    for line in lines[1:]:
        parts = line.split('|')
        parts = [p.strip() for p in parts if p.strip()]
        if len(parts) >= 5:
            data.append({
                'game': parts[0],
                'team': parts[1],
                'result': parts[2],
                'first_tower': parts[3],
                'void_grubs': int(parts[4]),
                'heralds': int(parts[5])
            })
    
    # Filter teams that took first tower
    first_tower_teams = [row for row in data if row['first_tower'] == 'yes']
    
    if not first_tower_teams:
        return 0
    
    # Count wins among teams with first tower
    wins = sum(1 for row in first_tower_teams if row['result'] == 'win')
    
    return wins / len(first_tower_teams)


def q2(text):
    """Return the average number of void grubs taken by winning teams."""
    lines = text.strip().split('\n')
    
    # Skip header
    data = []
    for line in lines[1:]:
        parts = line.split('|')
        parts = [p.strip() for p in parts if p.strip()]
        if len(parts) >= 5:
            data.append({
                'game': parts[0],
                'team': parts[1],
                'result': parts[2],
                'first_tower': parts[3],
                'void_grubs': int(parts[4]),
                'heralds': int(parts[5])
            })
    
    # Filter winning teams
    winning_teams = [row for row in data if row['result'] == 'win']
    
    if not winning_teams:
        return 0
    
    # Calculate average void grubs
    total_grubs = sum(row['void_grubs'] for row in winning_teams)
    
    return total_grubs / len(winning_teams)

import re

def q1(text):
    """Return the win rate of teams that took the first tower."""
    first_tower_teams = []
    
    # Parse all game data
    pattern = r'In game (\w+), (.+?) (won|lost); they (did not take|took) the first tower and finished with (\d+) void grubs and (\d+) heralds\.'
    matches = re.findall(pattern, text)
    
    for match in matches:
        game_id, team, result, tower_status, void_grubs, heralds = match
        took_first_tower = tower_status == "took"
        won = result == "won"
        
        if took_first_tower:
            first_tower_teams.append(won)
    
    if len(first_tower_teams) == 0:
        return 0
    
    wins = sum(first_tower_teams)
    return wins / len(first_tower_teams)


def q2(text):
    """Return the average number of void grubs taken by winning teams."""
    winning_void_grubs = []
    
    # Parse all game data
    pattern = r'In game (\w+), (.+?) (won|lost); they (did not take|took) the first tower and finished with (\d+) void grubs and (\d+) heralds\.'
    matches = re.findall(pattern, text)
    
    for match in matches:
        game_id, team, result, tower_status, void_grubs, heralds = match
        won = result == "won"
        void_grubs_count = int(void_grubs)
        
        if won:
            winning_void_grubs.append(void_grubs_count)
    
    if len(winning_void_grubs) == 0:
        return 0
    
    return sum(winning_void_grubs) / len(winning_void_grubs)


def q3(text):
    """Return how many teams took at least one herald and still lost."""
    teams_with_herald_that_lost = 0
    
    # Parse all game data
    pattern = r'In game (\w+), (.+?) (won|lost); they (did not take|took) the first tower and finished with (\d+) void grubs and (\d+) heralds\.'
    matches = re.findall(pattern, text)
    
    for match in matches:
        game_id, team, result, tower_status, void_grubs, heralds = match
        lost = result == "lost"
        heralds_count = int(heralds)
        
        if lost and heralds_count >= 1:
            teams_with_herald_that_lost += 1
    
    return teams_with_herald_that_lost

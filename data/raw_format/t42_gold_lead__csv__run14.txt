def q1(text):
    """Return the average gold difference at 15 minutes of the winning teams."""
    lines = text.strip().split('\n')
    gold_diffs = []
    
    for line in lines[1:]:  # Skip header
        parts = line.split(',')
        result = parts[3]
        gold_diff = int(parts[4])
        
        if result == 'win':
            gold_diffs.append(gold_diff)
    
    return sum(gold_diffs) / len(gold_diffs) if gold_diffs else 0


def q2(text):
    """Return how many teams won despite a negative gold difference at 15 minutes."""
    lines = text.strip().split('\n')
    count = 0
    
    for line in lines[1:]:  # Skip header
        parts = line.split(',')
        result = parts[3]
        gold_diff = int(parts[4])
        
        if result == 'win' and gold_diff < 0:
            count += 1
    
    return count


def q3(text):
    """Return the win rate of teams that were ahead in gold at 15 minutes."""
    lines = text.strip().split('\n')
    ahead_count = 0
    ahead_wins = 0
    
    for line in lines[1:]:  # Skip header
        parts = line.split(',')
        result = parts[3]
        gold_diff = int(parts[4])
        
        if gold_diff > 0:
            ahead_count += 1
            if result == 'win':
                ahead_wins += 1
    
    return ahead_wins / ahead_count if ahead_count > 0 else 0

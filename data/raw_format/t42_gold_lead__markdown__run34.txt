def q1(text):
    """Return the average gold difference at 15 minutes of the winning teams."""
    lines = text.strip().split('\n')
    # Skip header row
    data_lines = lines[1:]
    
    winning_gold_diffs = []
    for line in data_lines:
        parts = [p.strip() for p in line.split('|') if p.strip()]
        if len(parts) >= 5:
            result = parts[3]
            gold_diff = int(parts[4])
            if result == 'win':
                winning_gold_diffs.append(gold_diff)
    
    if len(winning_gold_diffs) == 0:
        return 0
    return sum(winning_gold_diffs) / len(winning_gold_diffs)


def q2(text):
    """Return how many teams won despite a negative gold difference at 15 minutes."""
    lines = text.strip().split('\n')
    # Skip header row
    data_lines = lines[1:]
    
    count = 0
    for line in data_lines:
        parts = [p.strip() for p in line.split('|') if p.strip()]
        if len(parts) >= 5:
            result = parts[3]
            gold_diff = int(parts[4])
            if result == 'win' and gold_diff < 0:
                count += 1
    
    return count


def q3(text):
    """Return the win rate of teams that were ahead in gold at 15 minutes."""
    lines = text.strip().split('\n')
    # Skip header row
    data_lines = lines[1:]
    
    ahead_in_gold_count = 0
    ahead_and_won_count = 0
    
    for line in data_lines:
        parts = [p.strip() for p in line.split('|') if p.strip()]
        if len(parts) >= 5:
            result = parts[3]
            gold_diff = int(parts[4])
            if gold_diff > 0:
                ahead_in_gold_count += 1
                if result == 'win':
                    ahead_and_won_count += 1
    
    if ahead_in_gold_count == 0:
        return 0
    return ahead_and_won_count / ahead_in_gold_count

```python
import json
import re

def q1(text):
    """Return the average number of kills at 15 minutes of red-side teams."""
    data = json.loads(text)
    red_side_kills = [row["kills_at_15"] for row in data if row["side"] == "red"]
    return sum(red_side_kills) / len(red_side_kills) if red_side_kills else 0

def q2(text):
    """Return the name of the team with the largest experience difference at 15 minutes."""
    data = json.loads(text)
    max_team = max(data, key=lambda row: abs(row["xp_diff_15"]))
    return max_team["team"]

def q3(text):
    """Return how many teams won despite a negative experience difference at 15 minutes."""
    data = json.loads(text)
    count = sum(1 for row in data if row["result"] == "win" and row["xp_diff_15"] < 0)
    return count
```

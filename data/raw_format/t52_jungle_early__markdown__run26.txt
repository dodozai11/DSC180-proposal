def q1(text):
    lines = text.strip().split('\n')
    header = lines[0].split('|')[1:-1]
    header = [h.strip() for h in header]
    
    player_idx = header.index('player')
    cs_diff_idx = header.index('cs_diff_10')
    result_idx = header.index('result')
    
    wins = 0
    total = 0
    
    for line in lines[2:]:
        if line.strip() == '':
            continue
        cells = line.split('|')[1:-1]
        cells = [c.strip() for c in cells]
        
        cs_diff = int(cells[cs_diff_idx])
        result = cells[result_idx]
        
        if cs_diff > 0:
            total += 1
            if result == 'win':
                wins += 1
    
    if total == 0:
        return 0
    return wins / total


def q2(text):
    lines = text.strip().split('\n')
    header = lines[0].split('|')[1:-1]
    header = [h.strip() for h in header]
    
    kills_idx = header.index('kills_at_10')
    
    total_kills = 0
    
    for line in lines[2:]:
        if line.strip() == '':
            continue
        cells = line.split('|')[1:-1]
        cells = [c.strip() for c in cells]
        
        kills = int(cells[kills_idx])
        total_kills += kills
    
    return total_kills


def q3(text):
    lines = text.strip().split('\n')
    header = lines[0].split('|')[1:-1]
    header = [h.strip() for h in header]
    
    champion_idx = header.index('champion')
    assists_idx = header.index('assists_at_10')
    
    champion_assists = {}
    
    for line in lines[2:]:
        if line.strip() == '':
            continue
        cells = line.split('|')[1:-1]
        cells = [c.strip() for c in cells]
        
        champion = cells[champion_idx]
        assists = int(cells[assists_idx])
        
        if champion not in champion_assists:
            champion_assists[champion] = 0
        champion_assists[champion] += assists
    
    return max(champion_assists, key=champion_assists.get)

def q1(text):
    lines = text.strip().split('\n')
    header = lines[0].split('|')[1:-1]
    header = [h.strip() for h in header]
    
    vision_idx = header.index('vision_score')
    result_idx = header.index('result')
    
    wins_vision = []
    losses_vision = []
    
    for line in lines[2:]:
        if line.strip():
            cells = line.split('|')[1:-1]
            cells = [c.strip() for c in cells]
            vision = float(cells[vision_idx])
            result = cells[result_idx]
            
            if result == 'win':
                wins_vision.append(vision)
            elif result == 'loss':
                losses_vision.append(vision)
    
    avg_wins = sum(wins_vision) / len(wins_vision) if wins_vision else 0
    avg_losses = sum(losses_vision) / len(losses_vision) if losses_vision else 0
    
    return avg_wins - avg_losses


def q2(text):
    lines = text.strip().split('\n')
    header = lines[0].split('|')[1:-1]
    header = [h.strip() for h in header]
    
    player_idx = header.index('player')
    wards_idx = header.index('wards_placed')
    
    max_wards = -1
    max_player = None
    
    for line in lines[2:]:
        if line.strip():
            cells = line.split('|')[1:-1]
            cells = [c.strip() for c in cells]
            player = cells[player_idx]
            wards = int(cells[wards_idx])
            
            if wards > max_wards:
                max_wards = wards
                max_player = player
    
    return max_player


def q3(text):
    lines = text.strip().split('\n')
    header = lines[0].split('|')[1:-1]
    header = [h.strip() for h in header]
    
    control_wards_idx = header.index('control_wards')
    
    count = 0
    
    for line in lines[2:]:
        if line.strip():
            cells = line.split('|')[1:-1]
            cells = [c.strip() for c in cells]
            control_wards = int(cells[control_wards_idx])
            
            if control_wards >= 15:
                count += 1
    
    return count

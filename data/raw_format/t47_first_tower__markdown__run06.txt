def q1(text):
    """Return the win rate of teams that took the first tower."""
    lines = text.strip().split('\n')
    
    # Skip header line
    data_lines = lines[1:]
    
    first_tower_wins = 0
    first_tower_total = 0
    
    for line in data_lines:
        parts = line.split('|')
        # Remove empty strings from split (due to leading/trailing |)
        parts = [p.strip() for p in parts if p.strip()]
        
        if len(parts) >= 5:
            first_tower = parts[2]  # third column (0-indexed: 2)
            result = parts[3]       # fourth column (0-indexed: 3)
            
            if first_tower == 'yes':
                first_tower_total += 1
                if result == 'win':
                    first_tower_wins += 1
    
    if first_tower_total == 0:
        return 0
    
    return first_tower_wins / first_tower_total


def q2(text):
    """Return the average number of void grubs taken by winning teams."""
    lines = text.strip().split('\n')
    
    # Skip header line
    data_lines = lines[1:]
    
    winning_void_grubs = []
    
    for line in data_lines:
        parts = line.split('|')
        # Remove empty strings from split (due to leading/trailing |)
        parts = [p.strip() for p in parts if p.strip()]
        
        if len(parts) >= 5:
            result = parts[3]           # fourth column (0-indexed: 3)
            void_grubs = int(parts[4])  # fifth column (0-indexed: 4)
            
            if result == 'win':
                winning_void_grubs.append(void_grubs)
    
    if len(winning_void_grubs) == 0:
        return 0
    
    return sum(winning_void_grubs) / len(winning_void_grubs)


def q3(text):
    """Return how many teams took at least one herald and still lost."""
    lines = text.strip().split('\n')
    
    # Skip header line
    data_lines = lines[1:]
    
    count = 0
    
    for line in data_lines:
        parts = line.split('|')
        # Remove empty strings from split (due to leading/trailing |)
        parts = [p.strip() for p in parts if p.strip()]
        
        if len(parts) >= 6:
            result = parts[3]      # fourth column (0-indexed: 3)
            heralds = int(parts[5])  # sixth column (0-indexed: 5)
            
            if result == 'loss' and heralds >= 1:
                count += 1
    
    return count

```python
import re

def q1(text):
    """Return the average vision score of winners minus average vision score of losers."""
    players = parse_players(text)
    
    winners = [p for p in players if p['result'] == 'won']
    losers = [p for p in players if p['result'] == 'lost']
    
    avg_winners = sum(p['vision_score'] for p in winners) / len(winners)
    avg_losers = sum(p['vision_score'] for p in losers) / len(losers)
    
    return avg_winners - avg_losers


def q2(text):
    """Return the name of the player who placed the most wards."""
    players = parse_players(text)
    
    max_player = max(players, key=lambda p: p['wards_placed'])
    return max_player['name']


def q3(text):
    """Return the count of players who bought at least 15 control wards."""
    players = parse_players(text)
    
    count = sum(1 for p in players if p['control_wards_bought'] >= 15)
    return count


def parse_players(text):
    """Parse player data from text format."""
    players = []
    
    # Pattern to match: "Name of Team result with a vision score of X, Y wards placed and Z control wards bought."
    pattern = r"(\w+) of (.+?) (won|lost) with a vision score of (\d+), (\d+) wards placed and (\d+) control wards bought\."
    
    matches = re.findall(pattern, text)
    
    for match in matches:
        name, team, result, vision_score, wards_placed, control_wards_bought = match
        players.append({
            'name': name,
            'team': team,
            'result': result,
            'vision_score': int(vision_score),
            'wards_placed': int(wards_placed),
            'control_wards_bought': int(control_wards_bought)
        })
    
    return players
```

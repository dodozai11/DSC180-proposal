```python
import re

def parse_data(text):
    """Parse the text data into structured format."""
    pattern = r"In game (\w+), (.+?) played on the (\w+) side and (\w+), with a gold difference of ([-\d]+) at 15 minutes\."
    matches = re.findall(pattern, text)
    
    data = []
    for match in matches:
        game_id, team, side, result, gold_diff = match
        data.append({
            'game_id': game_id,
            'team': team,
            'side': side,
            'result': result,
            'gold_diff': int(gold_diff)
        })
    
    return data

def q1(text):
    """Return the average gold difference at 15 minutes of the winning teams."""
    data = parse_data(text)
    
    winning_teams = [row['gold_diff'] for row in data if row['result'] == 'won']
    
    if not winning_teams:
        return 0
    
    return sum(winning_teams) / len(winning_teams)

def q2(text):
    """Return how many teams won despite a negative gold difference at 15 minutes."""
    data = parse_data(text)
    
    count = sum(1 for row in data if row['result'] == 'won' and row['gold_diff'] < 0)
    
    return count

def q3(text):
    """Return the win rate of teams that were ahead in gold at 15 minutes."""
    data = parse_data(text)
    
    ahead_in_gold = [row for row in data if row['gold_diff'] > 0]
    
    if not ahead_in_gold:
        return 0
    
    wins = sum(1 for row in ahead_in_gold if row['result'] == 'won')
    
    return wins / len(ahead_in_gold)
```

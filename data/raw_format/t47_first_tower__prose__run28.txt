import re

def q1(text):
    """Returns the win rate of teams that took the first tower."""
    took_first_tower = []
    
    # Parse all game entries
    pattern = r'In game (\w+), (.+?) (won|lost); they (took|did not take) the first tower and finished with (\d+) void grubs and (\d+) heralds\.'
    matches = re.findall(pattern, text)
    
    for match in matches:
        game_id, team, result, tower_status, void_grubs, heralds = match
        took_tower = tower_status == "took"
        won = result == "won"
        
        if took_tower:
            took_first_tower.append(won)
    
    if len(took_first_tower) == 0:
        return 0
    
    win_rate = sum(took_first_tower) / len(took_first_tower)
    return win_rate


def q2(text):
    """Returns the average number of void grubs taken by winning teams."""
    winning_void_grubs = []
    
    # Parse all game entries
    pattern = r'In game (\w+), (.+?) (won|lost); they (took|did not take) the first tower and finished with (\d+) void grubs and (\d+) heralds\.'
    matches = re.findall(pattern, text)
    
    for match in matches:
        game_id, team, result, tower_status, void_grubs, heralds = match
        won = result == "won"
        void_grubs_count = int(void_grubs)
        
        if won:
            winning_void_grubs.append(void_grubs_count)
    
    if len(winning_void_grubs) == 0:
        return 0
    
    average = sum(winning_void_grubs) / len(winning_void_grubs)
    return average


def q3(text):
    """Returns how many teams took at least one herald and still lost."""
    count = 0
    
    # Parse all game entries
    pattern = r'In game (\w+), (.+?) (won|lost); they (took|did not take) the first tower and finished with (\d+) void grubs and (\d+) heralds\.'
    matches = re.findall(pattern, text)
    
    for match in matches:
        game_id, team, result, tower_status, void_grubs, heralds = match
        lost = result == "lost"
        heralds_count = int(heralds)
        
        if lost and heralds_count >= 1:
            count += 1
    
    return count

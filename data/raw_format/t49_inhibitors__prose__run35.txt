import re

def q1(text):
    """Return the average number of inhibitors destroyed by winning teams."""
    winning_inhibitors = []
    
    # Find all game entries
    pattern = r'In game (\w+), (\w+(?:\s+\w+)*) (won|lost) after ([\d.]+) minutes, destroying (\d+) inhibitors'
    matches = re.findall(pattern, text)
    
    for match in matches:
        game_id, team, result, duration, inhibitors = match
        if result == 'won':
            winning_inhibitors.append(int(inhibitors))
    
    if not winning_inhibitors:
        return 0
    return sum(winning_inhibitors) / len(winning_inhibitors)


def q2(text):
    """Return the average damage to towers of losing teams."""
    losing_damage = []
    
    # Find all game entries
    pattern = r'In game (\w+), (\w+(?:\s+\w+)*) (won|lost) after ([\d.]+) minutes, destroying (\d+) inhibitors and dealing (\d+) damage to towers'
    matches = re.findall(pattern, text)
    
    for match in matches:
        game_id, team, result, duration, inhibitors, damage = match
        if result == 'lost':
            losing_damage.append(int(damage))
    
    if not losing_damage:
        return 0
    return sum(losing_damage) / len(losing_damage)


def q3(text):
    """Return the game label of the longest game."""
    longest_game = None
    longest_duration = -1
    
    # Find all game entries
    pattern = r'In game (\w+), (\w+(?:\s+\w+)*) (won|lost) after ([\d.]+) minutes'
    matches = re.findall(pattern, text)
    
    seen_games = set()
    for match in matches:
        game_id, team, result, duration = match
        if game_id not in seen_games:
            seen_games.add(game_id)
            duration_float = float(duration)
            if duration_float > longest_duration:
                longest_duration = duration_float
                longest_game = game_id
    
    return longest_game

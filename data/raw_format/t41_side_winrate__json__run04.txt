```python
import json
import re

def q1(text):
    """Return the win rate of blue-side teams"""
    data = json.loads(text)
    
    blue_games = [row for row in data if row["side"] == "blue"]
    blue_wins = [row for row in blue_games if row["result"] == "win"]
    
    if len(blue_games) == 0:
        return 0
    
    return len(blue_wins) / len(blue_games)


def q2(text):
    """Return the average length in minutes of games won by red-side teams"""
    data = json.loads(text)
    
    red_wins = [row for row in data if row["side"] == "red" and row["result"] == "win"]
    
    if len(red_wins) == 0:
        return 0
    
    total_minutes = sum(row["game_minutes"] for row in red_wins)
    return total_minutes / len(red_wins)


def q3(text):
    """Return the name of the team with the most wins"""
    data = json.loads(text)
    
    wins_by_team = {}
    
    for row in data:
        if row["result"] == "win":
            team = row["team"]
            wins_by_team[team] = wins_by_team.get(team, 0) + 1
    
    if not wins_by_team:
        return None
    
    return max(wins_by_team, key=wins_by_team.get)
```

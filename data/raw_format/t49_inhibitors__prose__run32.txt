```python
import re

def q1(text):
    """Return the average number of inhibitors destroyed by winning teams."""
    winning_inhibitors = []
    
    # Find all matches for winning teams
    pattern = r'(\w+[\w\s]*?)\s+won\s+after\s+([\d.]+)\s+minutes,\s+destroying\s+(\d+)\s+inhibitors'
    matches = re.findall(pattern, text)
    
    for match in matches:
        inhibitors = int(match[2])
        winning_inhibitors.append(inhibitors)
    
    if winning_inhibitors:
        return sum(winning_inhibitors) / len(winning_inhibitors)
    return 0

def q2(text):
    """Return the average damage to towers of losing teams."""
    losing_damages = []
    
    # Find all matches for losing teams
    pattern = r'(\w+[\w\s]*?)\s+lost\s+after\s+([\d.]+)\s+minutes,\s+destroying\s+\d+\s+inhibitors\s+and\s+dealing\s+([\d]+)\s+damage'
    matches = re.findall(pattern, text)
    
    for match in matches:
        damage = int(match[2])
        losing_damages.append(damage)
    
    if losing_damages:
        return sum(losing_damages) / len(losing_damages)
    return 0

def q3(text):
    """Return the game label of the longest game."""
    games = {}
    
    # Find all game entries with duration
    pattern = r'In game (G\d+),\s+(\w+[\w\s]*?)\s+(won|lost)\s+after\s+([\d.]+)\s+minutes'
    matches = re.findall(pattern, text)
    
    for match in matches:
        game_label = match[0]
        duration = float(match[3])
        
        # Store the duration (both teams have same duration, so just store once)
        if game_label not in games:
            games[game_label] = duration
    
    if games:
        longest_game = max(games.items(), key=lambda x: x[1])
        return longest_game[0]
    return None
```

```python
import re
from collections import defaultdict

def parse_matches(text):
    """Parse the match data from text format."""
    matches = []
    # Pattern: game GXXX, Team Name played on the (blue|red) side and (won|lost); the game lasted X.X minutes.
    pattern = r'In game (G\d+), (.+?) played on the (blue|red) side and (won|lost); the game lasted ([\d.]+) minutes\.'
    
    for match in re.finditer(pattern, text):
        game_id = match.group(1)
        team_name = match.group(2)
        side = match.group(3)
        result = match.group(4)
        duration = float(match.group(5))
        
        matches.append({
            'game_id': game_id,
            'team': team_name,
            'side': side,
            'result': result,
            'duration': duration
        })
    
    return matches

def q1(text):
    """Return the win rate of blue-side teams."""
    matches = parse_matches(text)
    
    blue_wins = 0
    blue_total = 0
    
    for match in matches:
        if match['side'] == 'blue':
            blue_total += 1
            if match['result'] == 'won':
                blue_wins += 1
    
    if blue_total == 0:
        return 0
    
    return blue_wins / blue_total

def q2(text):
    """Return the average length in minutes of games won by the red-side team."""
    matches = parse_matches(text)
    
    red_wins = []
    
    for match in matches:
        if match['side'] == 'red' and match['result'] == 'won':
            red_wins.append(match['duration'])
    
    if len(red_wins) == 0:
        return 0
    
    return sum(red_wins) / len(red_wins)

def q3(text):
    """Return the name of the team with the most wins."""
    matches = parse_matches(text)
    
    team_wins = defaultdict(int)
    
    for match in matches:
        if match['result'] == 'won':
            team_wins[match['team']] += 1
    
    if not team_wins:
        return None
    
    return max(team_wins, key=team_wins.get)
```

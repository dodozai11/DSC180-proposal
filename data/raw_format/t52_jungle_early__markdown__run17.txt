def q2(text):
    lines = text.strip().split('\n')
    
    total_kills = 0
    
    for line in lines[2:]:

def q3(text):
    lines = text.strip().split('\n')
    
    champion_assists = {}
    
    for line in lines[2:]:
        if line.strip() == '':
            continue
        cells = line.split('|')[1:-1]
        cells = [c.strip() for c in cells]
        
        champion = cells[1]
        assists_at_10 = int(cells[5])
        
        if champion not in champion_assists:
            champion_assists[champion] = 0
        champion_assists[champion] += assists_at_10
    
    return max(champion_assists, key=champion_assists.get)

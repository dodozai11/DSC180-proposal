```python
import re
from xml.etree import ElementTree as ET

def q1(text):
    """Return the win rate of blue-side teams"""
    root = ET.fromstring(f"<root>{text}</root>")
    blue_games = [row for row in root.findall('row') if row.find('side').text == 'blue']
    blue_wins = [row for row in blue_games if row.find('result').text == 'win']
    
    if len(blue_games) == 0:
        return 0
    return len(blue_wins) / len(blue_games)


def q2(text):
    """Return the average length in minutes of games won by red-side team"""
    root = ET.fromstring(f"<root>{text}</root>")
    red_wins = [row for row in root.findall('row') 
                if row.find('side').text == 'red' and row.find('result').text == 'win']
    
    if len(red_wins) == 0:
        return 0
    
    total_minutes = sum(float(row.find('game_minutes').text) for row in red_wins)
    return total_minutes / len(red_wins)


def q3(text):
    """Return the name of the team with the most wins"""
    root = ET.fromstring(f"<root>{text}</root>")
    wins_by_team = {}
    
    for row in root.findall('row'):
        team = row.find('team').text
        result = row.find('result').text
        
        if result == 'win':
            wins_by_team[team] = wins_by_team.get(team, 0) + 1
    
    if not wins_by_team:
        return None
    
    return max(wins_by_team, key=wins_by_team.get)
```

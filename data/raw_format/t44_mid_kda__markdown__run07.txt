def q1(text):
    lines = text.strip().split('\n')
    header = lines[0].split('|')[1:-1]
    header = [h.strip() for h in header]
    
    best_player = None
    best_kda = -1
    
    for line in lines[2:]:
        if not line.strip():
            continue
        cells = line.split('|')[1:-1]
        cells = [c.strip() for c in cells]
        
        player = cells[0]
        kills = int(cells[3])
        deaths = int(cells[4])
        assists = int(cells[5])
        
        kda = (kills + assists) / max(deaths, 1)
        
        if kda > best_kda:
            best_kda = kda
            best_player = player
    
    return best_player


def q2(text):
    lines = text.strip().split('\n')
    
    total_kills = 0
    win_count = 0
    
    for line in lines[2:]:
        if not line.strip():
            continue
        cells = line.split('|')[1:-1]
        cells = [c.strip() for c in cells]
        
        kills = int(cells[3])
        result = cells[6]
        
        if result == 'win':
            total_kills += kills
            win_count += 1
    
    return total_kills / win_count if win_count > 0 else 0


def q3(text):
    lines = text.strip().split('\n')
    
    champion_counts = {}
    
    for line in lines[2:]:
        if not line.strip():
            continue
        cells = line.split('|')[1:-1]
        cells = [c.strip() for c in cells]
        
        champion = cells[2]
        champion_counts[champion] = champion_counts.get(champion, 0) + 1
    
    return max(champion_counts, key=champion_counts.get)

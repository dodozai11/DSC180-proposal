def q1(text):
    lines = text.strip().split('\n')
    header = lines[0].split('|')[1:-1]
    header = [h.strip() for h in header]
    
    max_cs = -1
    max_player = None
    
    for line in lines[2:]:
        if line.strip() == '':
            continue
        parts = [p.strip() for p in line.split('|')[1:-1]]
        if len(parts) < len(header):
            continue
        
        player = parts[0]
        cs_per_min = float(parts[4])
        
        if cs_per_min > max_cs:
            max_cs = cs_per_min
            max_player = player
    
    return max_player


def q2(text):
    lines = text.strip().split('\n')
    
    total_gold_share = 0
    win_count = 0
    
    for line in lines[2:]:
        if line.strip() == '':
            continue
        parts = [p.strip() for p in line.split('|')[1:-1]]
        if len(parts) < 5:
            continue
        
        result = parts[3]
        gold_share = float(parts[5])
        
        if result == 'win':
            total_gold_share += gold_share
            win_count += 1
    
    if win_count == 0:
        return 0
    return total_gold_share / win_count


def q3(text):
    lines = text.strip().split('\n')
    
    count = 0
    
    for line in lines[2:]:
        if line.strip() == '':
            continue
        parts = [p.strip() for p in line.split('|')[1:-1]]
        if len(parts) < 5:
            continue
        
        cs_per_min = float(parts[4])
        
        if cs_per_min > 9:
            count += 1
    
    return count

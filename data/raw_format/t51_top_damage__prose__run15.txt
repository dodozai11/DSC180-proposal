import re

def q1(text):
    """Return the average damage per minute of players who lost."""
    pattern = r"(\w+(?:\s+\w+)*) of ([\w\s]+) played ([\w']+) in the top lane and (won|lost), dealing ([\d.]+) damage per minute and taking ([\d.]+) damage per minute"
    matches = re.findall(pattern, text)
    
    lost_damages = []
    for match in matches:
        player_name, team, champion, result, damage_dealt, damage_taken = match
        if result == "lost":
            lost_damages.append(float(damage_dealt))
    
    if lost_damages:
        return sum(lost_damages) / len(lost_damages)
    return 0

def q2(text):
    """Return the name of the player who took the most damage per minute."""
    pattern = r"(\w+(?:\s+\w+)*) of ([\w\s]+) played ([\w']+) in the top lane and (won|lost), dealing ([\d.]+) damage per minute and taking ([\d.]+) damage per minute"
    matches = re.findall(pattern, text)
    
    max_damage_taken = -1
    max_player = None
    
    for match in matches:
        player_name, team, champion, result, damage_dealt, damage_taken = match
        damage_taken_float = float(damage_taken)
        if damage_taken_float > max_damage_taken:
            max_damage_taken = damage_taken_float
            max_player = player_name
    
    return max_player

def q3(text):
    """Return the number of different teams."""
    pattern = r"(\w+(?:\s+\w+)*) of ([\w\s]+) played ([\w']+) in the top lane and (won|lost), dealing ([\d.]+) damage per minute and taking ([\d.]+) damage per minute"
    matches = re.findall(pattern, text)
    
    teams = set()
    for match in matches:
        player_name, team, champion, result, damage_dealt, damage_taken = match
        teams.add(team.strip())
    
    return len(teams)

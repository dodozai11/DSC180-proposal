```python
import re

def parse_data(text):
    """Parse the text data and return a list of tuples (game_id, team, side, result, gold_diff)"""
    pattern = r"In game (\w+), (\w+(?:\s+\w+)*) played on the (\w+) side and (\w+), with a gold difference of ([-\d]+) at 15 minutes"
    matches = re.findall(pattern, text)
    
    data = []
    for match in matches:
        game_id = match[0]
        team = match[1]
        side = match[2]
        result = match[3]
        gold_diff = int(match[4])
        data.append((game_id, team, side, result, gold_diff))
    
    return data

def q1(text):
    """Return the average gold difference at 15 minutes of the winning teams."""
    data = parse_data(text)
    winning_gold_diffs = []
    
    for game_id, team, side, result, gold_diff in data:
        if result == "won":
            winning_gold_diffs.append(gold_diff)
    
    if len(winning_gold_diffs) == 0:
        return 0
    
    return sum(winning_gold_diffs) / len(winning_gold_diffs)

def q2(text):
    """Return how many teams won despite a negative gold difference at 15 minutes."""
    data = parse_data(text)
    count = 0
    
    for game_id, team, side, result, gold_diff in data:
        if result == "won" and gold_diff < 0:
            count += 1
    
    return count

def q3(text):
    """Return the win rate of teams that were ahead in gold at 15 minutes."""
    data = parse_data(text)
    ahead_in_gold = []
    
    for game_id, team, side, result, gold_diff in data:
        if gold_diff > 0:
            ahead_in_gold.append((result, gold_diff))
    
    if len(ahead_in_gold) == 0:
        return 0
    
    wins = sum(1 for result, _ in ahead_in_gold if result == "won")
    return wins / len(ahead_in_gold)
```

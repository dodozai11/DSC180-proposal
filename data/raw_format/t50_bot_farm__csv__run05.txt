def q1(text):
    lines = text.strip().split('\n')
    max_cs_player = None
    max_cs = -1
    
    for line in lines[1:]:
        parts = line.split(',')
        player = parts[0]
        cs_per_min = float(parts[4])
        
        if cs_per_min > max_cs:
            max_cs = cs_per_min
            max_cs_player = player
    
    return max_cs_player


def q2(text):
    lines = text.strip().split('\n')
    total_gold_share = 0
    win_count = 0
    
    for line in lines[1:]:
        parts = line.split(',')
        result = parts[3]
        gold_share = float(parts[5])
        
        if result == 'win':
            total_gold_share += gold_share
            win_count += 1
    
    return total_gold_share / win_count if win_count > 0 else 0


def q3(text):
    lines = text.strip().split('\n')
    count = 0
    
    for line in lines[1:]:
        parts = line.split(',')
        cs_per_min = float(parts[4])
        
        if cs_per_min > 9:
            count += 1
    
    return count

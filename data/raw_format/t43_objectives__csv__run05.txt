def q1(text):
    """Calculate win rate of teams that took first dragon"""
    lines = text.strip().split('\n')[1:]  # Skip header
    
    first_dragon_wins = 0
    first_dragon_total = 0
    
    for line in lines:
        parts = line.split(',')
        first_dragon = parts[2]  # first_dragon column
        result = parts[2]  # result column
        
        # Re-parse correctly
        game, team, result, first_dragon, dragons, barons, towers = parts
        
        if first_dragon == 'yes':
            first_dragon_total += 1
            if result == 'win':
                first_dragon_wins += 1
    
    if first_dragon_total == 0:
        return 0
    
    return first_dragon_wins / first_dragon_total


def q2(text):
    """Calculate average number of barons taken by winning teams"""
    lines = text.strip().split('\n')[1:]  # Skip header
    
    winning_barons = []
    
    for line in lines:
        parts = line.split(',')
        game, team, result, first_dragon, dragons, barons, towers = parts
        
        if result == 'win':
            winning_barons.append(int(barons))
    
    if len(winning_barons) == 0:
        return 0
    
    return sum(winning_barons) / len(winning_barons)


def q3(text):
    """Calculate total number of towers destroyed across all rows"""

import json
import re

def q1(text):
    """Returns the average gold difference at 15 minutes of the winning teams."""
    data = json.loads(text)
    winning_teams = [row for row in data if row["result"] == "win"]
    if not winning_teams:
        return 0
    avg_gold_diff = sum(row["gold_diff_15"] for row in winning_teams) / len(winning_teams)
    return avg_gold_diff

def q2(text):
    """Returns how many teams won despite a negative gold difference at 15 minutes."""
    data = json.loads(text)
    count = 0
    for row in data:
        if row["result"] == "win" and row["gold_diff_15"] < 0:
            count += 1
    return count

def q3(text):
    """Returns the win rate of teams that were ahead in gold at 15 minutes."""
    data = json.loads(text)
    teams_ahead = [row for row in data if row["gold_diff_15"] > 0]
    if not teams_ahead:
        return 0
    wins = sum(1 for row in teams_ahead if row["result"] == "win")
    win_rate = wins / len(teams_ahead)
    return win_rate

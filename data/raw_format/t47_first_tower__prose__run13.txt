import re

def q1(text):
    """Return the win rate of teams that took the first tower."""
    first_tower_teams = []
    
    # Parse all game entries
    pattern = r'In game (G\d+), (.+?) (won|lost); they (took|did not take) the first tower'
    matches = re.findall(pattern, text)
    
    for game, team, result, tower_status in matches:
        took_tower = tower_status == "took"
        won = result == "won"
        first_tower_teams.append((took_tower, won))
    
    # Filter teams that took first tower
    teams_with_tower = [won for took_tower, won in first_tower_teams if took_tower]
    
    if not teams_with_tower:
        return 0
    
    win_rate = sum(teams_with_tower) / len(teams_with_tower)
    return win_rate


def q2(text):
    """Return the average number of void grubs taken by winning teams."""
    winning_teams_grubs = []
    
    # Parse all game entries with void grubs info
    pattern = r'In game (G\d+), (.+?) (won|lost);.+?finished with (\d+) void grubs'
    matches = re.findall(pattern, text)
    
    for game, team, result, grubs in matches:
        won = result == "won"
        grubs_count = int(grubs)
        if won:
            winning_teams_grubs.append(grubs_count)
    
    if not winning_teams_grubs:
        return 0
    
    avg_grubs = sum(winning_teams_grubs) / len(winning_teams_grubs)
    return avg_grubs


def q3(text):
    """Return how many teams took at least one herald and still lost."""
    count = 0
    
    # Parse all game entries with herald info
    pattern = r'In game (G\d+), (.+?) (won|lost);.+?finished with \d+ void grubs and (\d+) heralds'
    matches = re.findall(pattern, text)
    
    for game, team, result, heralds in matches:
        lost = result == "lost"
        heralds_count = int(heralds)
        if lost and heralds_count >= 1:
            count += 1
    
    return count

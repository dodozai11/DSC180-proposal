def q1(text):
    lines = text.strip().split('\n')
    header = lines[0].split(',')
    
    rows = [line.split(',') for line in lines[1:]]
    
    cs_diff_idx = header.index('cs_diff_10')
    result_idx = header.index('result')
    
    positive_cs_diff_rows = [row for row in rows if int(row[cs_diff_idx]) > 0]
    
    if not positive_cs_diff_rows:
        return 0
    
    wins = sum(1 for row in positive_cs_diff_rows if row[result_idx] == 'win')
    win_rate = wins / len(positive_cs_diff_rows)
    
    return win_rate


def q2(text):
    lines = text.strip().split('\n')
    header = lines[0].split(',')
    
    rows = [line.split(',') for line in lines[1:]]
    
    kills_idx = header.index('kills_at_10')
    
    total_kills = sum(int(row[kills_idx]) for row in rows)
    
    return total_kills


def q3(text):
    lines = text.strip().split('\n')
    header = lines[0].split(',')
    
    rows = [line.split(',') for line in lines[1:]]
    
    champion_idx = header.index('champion')
    assists_idx = header.index('assists_at_10')
    
    champion_assists = {}
    for row in rows:
        champion = row[champion_idx]
        assists = int(row[assists_idx])
        champion_assists[champion] = champion_assists.get(champion, 0) + assists
    
    champion_with_most_assists = max(champion_assists, key=champion_assists.get)
    
    return champion_with_most_assists

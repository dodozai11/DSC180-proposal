def q1(text):
    """Return the average number of inhibitors destroyed by winning teams."""
    lines = text.strip().split('\n')
    # Skip header line
    data_lines = lines[1:]
    
    winning_inhibitors = []
    for line in data_lines:
        parts = line.split('|')
        parts = [p.strip() for p in parts if p.strip()]
        
        if len(parts) >= 5:
            result = parts[2]
            inhibitors = int(parts[4])
            
            if result == 'win':
                winning_inhibitors.append(inhibitors)
    
    if winning_inhibitors:
        return sum(winning_inhibitors) / len(winning_inhibitors)
    return 0

def q3(text):
    """Return the game label of the longest game."""
    lines = text.strip().split('\n')
    # Skip header line
    data_lines = lines[1:]
    
    max_game = None
    max_duration = -1
    
    for line in data_lines:
        parts = line.split('|')
        parts = [p.strip() for p in parts if p.strip()]
        
        if len(parts) >= 4:
            game = parts[0]
            game_minutes = float(parts[3])
            
            if game_minutes > max_duration:
                max_duration = game_minutes
                max_game = game
    
    return max_game

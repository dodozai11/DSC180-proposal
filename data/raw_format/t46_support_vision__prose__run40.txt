```python
import re

def parse_data(text):
    """Parse the data and return a list of player dictionaries."""
    pattern = r"(\w+) of (.+?) (won|lost) with a vision score of (\d+), (\d+) wards placed and (\d+) control wards bought\."
    matches = re.findall(pattern, text)
    
    players = []
    for match in matches:
        player = {
            'name': match[0],
            'team': match[1],
            'result': match[2],
            'vision_score': int(match[3]),
            'wards_placed': int(match[4]),
            'control_wards_bought': int(match[5])
        }
        players.append(player)
    
    return players


def q1(text):
    """Return the average vision score of players who won minus the average vision score of players who lost."""
    players = parse_data(text)
    
    winners = [p for p in players if p['result'] == 'won']
    losers = [p for p in players if p['result'] == 'lost']
    
    avg_vision_winners = sum(p['vision_score'] for p in winners) / len(winners) if winners else 0
    avg_vision_losers = sum(p['vision_score'] for p in losers) / len(losers) if losers else 0
    
    return avg_vision_winners - avg_vision_losers


def q2(text):
    """Return the name of the player who placed the most wards."""
    players = parse_data(text)
    
    max_player = max(players, key=lambda p: p['wards_placed'])
    return max_player['name']


def q3(text):
    """Return how many players bought at least 15 control wards."""
    players = parse_data(text)
    
    count = sum(1 for p in players if p['control_wards_bought'] >= 15)
    return count
```

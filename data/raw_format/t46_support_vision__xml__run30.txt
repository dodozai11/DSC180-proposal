import xml.etree.ElementTree as ET

def q1(text):
    root = ET.fromstring(f"<root>{text}</root>")
    
    wins = []
    losses = []
    
    for row in root.findall('row'):
        vision_score = int(row.find('vision_score').text)
        result = row.find('result').text
        
        if result == 'win':
            wins.append(vision_score)
        else:
            losses.append(vision_score)
    
    avg_wins = sum(wins) / len(wins) if wins else 0
    avg_losses = sum(losses) / len(losses) if losses else 0
    
    return avg_wins - avg_losses

def q2(text):
    root = ET.fromstring(f"<root>{text}</root>")
    
    max_wards = -1
    max_player = None
    
    for row in root.findall('row'):
        player = row.find('player').text
        wards_placed = int(row.find('wards_placed').text)
        
        if wards_placed > max_wards:
            max_wards = wards_placed
            max_player = player
    
    return max_player

def q3(text):
    root = ET.fromstring(f"<root>{text}</root>")
    
    count = 0
    
    for row in root.findall('row'):
        control_wards = int(row.find('control_wards').text)
        
        if control_wards >= 15:
            count += 1
    
    return count

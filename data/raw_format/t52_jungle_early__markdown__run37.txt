def q1(text):
    """Return the win rate of players with a positive CS difference at 10 minutes."""
    lines = text.strip().split('\n')
    # Skip header row
    rows = lines[2:]
    
    positive_cs_diff_count = 0
    positive_cs_diff_wins = 0
    
    for row in rows:
        parts = [p.strip() for p in row.split('|') if p.strip()]
        if len(parts) < 5:
            continue
        
        player = parts[0]
        champion = parts[1]
        result = parts[2]
        cs_diff_10 = int(parts[3])
        
        if cs_diff_10 > 0:
            positive_cs_diff_count += 1
            if result == 'win':
                positive_cs_diff_wins += 1
    
    if positive_cs_diff_count == 0:
        return 0
    
    return positive_cs_diff_wins / positive_cs_diff_count


def q2(text):
    """Return the total number of kills at 10 minutes across all rows."""
    lines = text.strip().split('\n')
    # Skip header row
    rows = lines[2:]
    
    total_kills = 0
    
    for row in rows:
        parts = [p.strip() for p in row.split('|') if p.strip()]
        if len(parts) < 5:
            continue
        
        kills_at_10 = int(parts[4])
        total_kills += kills_at_10
    
    return total_kills


def q3(text):
    """Return the name of the champion with the most assists at 10 minutes in total."""
    lines = text.strip().split('\n')
    # Skip header row
    rows = lines[2:]
    
    champion_assists = {}
    
    for row in rows:
        parts = [p.strip() for p in row.split('|') if p.strip()]
        if len(parts) < 6:
            continue
        
        champion = parts[1]
        assists_at_10 = int(parts[5])
        
        if champion not in champion_assists:
            champion_assists[champion] = 0
        champion_assists[champion] += assists_at_10
    
    if not champion_assists:
        return None
    
    return max(champion_assists, key=champion_assists.get)

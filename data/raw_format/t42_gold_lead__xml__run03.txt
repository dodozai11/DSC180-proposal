import xml.etree.ElementTree as ET

def q1(text):
    """Returns the average gold difference at 15 minutes of the winning teams."""
    root = ET.fromstring(f"<root>{text}</root>")
    winning_gold_diffs = []
    
    for row in root.findall('row'):
        result = row.find('result').text
        gold_diff_15 = int(row.find('gold_diff_15').text)
        
        if result == 'win':
            winning_gold_diffs.append(gold_diff_15)
    
    return sum(winning_gold_diffs) / len(winning_gold_diffs) if winning_gold_diffs else 0

def q2(text):
    """Returns how many teams won despite a negative gold difference at 15 minutes."""
    root = ET.fromstring(f"<root>{text}</root>")
    count = 0
    
    for row in root.findall('row'):
        result = row.find('result').text
        gold_diff_15 = int(row.find('gold_diff_15').text)
        
        if result == 'win' and gold_diff_15 < 0:
            count += 1
    
    return count

def q3(text):
    """Returns the win rate of teams that were ahead in gold at 15 minutes."""
    root = ET.fromstring(f"<root>{text}</root>")
    ahead_in_gold = 0
    ahead_and_won = 0
    
    for row in root.findall('row'):
        result = row.find('result').text
        gold_diff_15 = int(row.find('gold_diff_15').text)
        
        if gold_diff_15 > 0:
            ahead_in_gold += 1
            if result == 'win':
                ahead_and_won += 1
    
    return ahead_and_won / ahead_in_gold if ahead_in_gold > 0 else 0

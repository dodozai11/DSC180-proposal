def q1(text):
    """Return the win rate of blue-side teams."""
    lines = text.strip().split('\n')
    blue_wins = 0
    blue_total = 0
    
    for line in lines[1:]:  # Skip header
        parts = line.split(',')
        side = parts[2]
        result = parts[3]
        
        if side == 'blue':
            blue_total += 1
            if result == 'win':
                blue_wins += 1
    
    return blue_wins / blue_total if blue_total > 0 else 0


def q2(text):
    """Return the average length in minutes of games won by the red-side team."""
    lines = text.strip().split('\n')
    red_win_times = []
    
    for line in lines[1:]:  # Skip header
        parts = line.split(',')
        side = parts[2]
        result = parts[3]
        game_minutes = float(parts[4])
        
        if side == 'red' and result == 'win':
            red_win_times.append(game_minutes)
    
    return sum(red_win_times) / len(red_win_times) if red_win_times else 0


def q3(text):
    """Return the name of the team with the most wins."""
    lines = text.strip().split('\n')
    team_wins = {}
    
    for line in lines[1:]:  # Skip header
        parts = line.split(',')
        team = parts[1]
        result = parts[3]
        
        if team not in team_wins:
            team_wins[team] = 0
        
        if result == 'win':
            team_wins[team] += 1
    
    return max(team_wins, key=team_wins.get) if team_wins else None

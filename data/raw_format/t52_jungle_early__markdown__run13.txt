def q1(text):
    lines = text.strip().split('\n')
    header = lines[0].split('|')[1:-1]
    header = [h.strip() for h in header]
    
    positive_cs_diff = []
    for line in lines[2:]:
        if line.strip():
            cols = line.split('|')[1:-1]
            cols = [c.strip() for c in cols]
            row = dict(zip(header, cols))
            
            cs_diff = int(row['cs_diff_10'])
            if cs_diff > 0:
                result = row['result']
                positive_cs_diff.append(1 if result == 'win' else 0)
    
    if len(positive_cs_diff) == 0:
        return 0
    return sum(positive_cs_diff) / len(positive_cs_diff)


def q2(text):
    lines = text.strip().split('\n')
    header = lines[0].split('|')[1:-1]
    header = [h.strip() for h in header]
    
    total_kills = 0
    for line in lines[2:]:
        if line.strip():
            cols = line.split('|')[1:-1]
            cols = [c.strip() for c in cols]
            row = dict(zip(header, cols))
            
            kills = int(row['kills_at_10'])
            total_kills += kills
    
    return total_kills


def q3(text):
    lines = text.strip().split('\n')
    header = lines[0].split('|')[1:-1]
    header = [h.strip() for h in header]
    
    champion_assists = {}
    for line in lines[2:]:
        if line.strip():
            cols = line.split('|')[1:-1]
            cols = [c.strip() for c in cols]
            row = dict(zip(header, cols))
            
            champion = row['champion']
            assists = int(row['assists_at_10'])
            
            if champion not in champion_assists:
                champion_assists[champion] = 0
            champion_assists[champion] += assists
    
    return max(champion_assists, key=champion_assists.get)

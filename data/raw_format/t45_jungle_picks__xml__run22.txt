import re
from collections import defaultdict

def q1(text):
    """Return the number of different champions played."""
    champions = set(re.findall(r'<champion>([^<]+)</champion>', text))
    return len(champions)

def q2(text):
    """Return the name of the champion with the best win rate among champions played at least twice."""
    champion_stats = defaultdict(lambda: {'wins': 0, 'total': 0})
    
    # Parse all rows
    rows = re.findall(r'<row>.*?</row>', text, re.DOTALL)
    for row in rows:
        champion = re.search(r'<champion>([^<]+)</champion>', row).group(1)
        result = re.search(r'<result>([^<]+)</result>', row).group(1)
        
        champion_stats[champion]['total'] += 1
        if result == 'win':
            champion_stats[champion]['wins'] += 1
    
    # Filter champions played at least twice
    qualified_champions = {champ: stats for champ, stats in champion_stats.items() if stats['total'] >= 2}
    
    if not qualified_champions:
        return None
    
    # Find champion with best win rate
    best_champion = max(qualified_champions.items(), key=lambda x: x[1]['wins'] / x[1]['total'])
    return best_champion[0]

def q3(text):
    """Return the average number of deaths in games that were lost."""
    losses = []
    
    # Parse all rows
    rows = re.findall(r'<row>.*?</row>', text, re.DOTALL)
    for row in rows:
        result = re.search(r'<result>([^<]+)</result>', row).group(1)
        deaths = int(re.search(r'<deaths>([^<]+)</deaths>', row).group(1))
        
        if result == 'loss':
            losses.append(deaths)
    
    if not losses:
        return 0
    
    return sum(losses) / len(losses)

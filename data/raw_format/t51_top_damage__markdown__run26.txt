def q1(text):
    """Return the average damage per minute of players who lost."""
    lines = text.strip().split('\n')
    # Skip header row
    rows = lines[1:]
    
    total_damage = 0
    count = 0
    
    for row in rows:
        parts = [p.strip() for p in row.split('|') if p.strip()]
        # Format: player, team, champion, result, damage_per_min, damage_taken_per_min
        if len(parts) >= 5:
            result = parts[3]
            damage_per_min = float(parts[4])
            
            if result.lower() == 'loss':
                total_damage += damage_per_min
                count += 1
    
    return total_damage / count if count > 0 else 0


def q2(text):
    """Return the name of the player who took the most damage per minute."""
    lines = text.strip().split('\n')
    # Skip header row
    rows = lines[1:]
    
    max_damage_taken = -1
    player_with_max = None
    
    for row in rows:
        parts = [p.strip() for p in row.split('|') if p.strip()]
        # Format: player, team, champion, result, damage_per_min, damage_taken_per_min
        if len(parts) >= 6:
            player = parts[0]
            damage_taken_per_min = float(parts[5])
            
            if damage_taken_per_min > max_damage_taken:
                max_damage_taken = damage_taken_per_min
                player_with_max = player
    
    return player_with_max


def q3(text):
    """Return the number of different teams."""
    lines = text.strip().split('\n')
    # Skip header row
    rows = lines[1:]
    
    teams = set()
    
    for row in rows:
        parts = [p.strip() for p in row.split('|') if p.strip()]
        # Format: player, team, champion, result, damage_per_min, damage_taken_per_min
        if len(parts) >= 2:
            team = parts[1]
            teams.add(team)
    
    return len(teams)

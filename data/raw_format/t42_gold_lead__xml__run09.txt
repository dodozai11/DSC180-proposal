import xml.etree.ElementTree as ET

def q1(text):
    """Return the average gold difference at 15 minutes of the winning teams."""
    root = ET.fromstring(f"<root>{text}</root>")
    winning_gold_diffs = []
    
    for row in root.findall('row'):
        if row.find('result').text == 'win':
            gold_diff = int(row.find('gold_diff_15').text)
            winning_gold_diffs.append(gold_diff)
    
    return sum(winning_gold_diffs) / len(winning_gold_diffs) if winning_gold_diffs else 0


def q2(text):
    """Return how many teams won despite a negative gold difference at 15 minutes."""
    root = ET.fromstring(f"<root>{text}</root>")
    count = 0
    
    for row in root.findall('row'):
        if row.find('result').text == 'win':
            gold_diff = int(row.find('gold_diff_15').text)
            if gold_diff < 0:
                count += 1
    
    return count


def q3(text):
    """Return the win rate of teams that were ahead in gold at 15 minutes."""
    root = ET.fromstring(f"<root>{text}</root>")
    ahead_in_gold = 0
    ahead_and_won = 0
    
    for row in root.findall('row'):
        gold_diff = int(row.find('gold_diff_15').text)
        if gold_diff > 0:
            ahead_in_gold += 1
            if row.find('result').text == 'win':
                ahead_and_won += 1
    
    return ahead_and_won / ahead_in_gold if ahead_in_gold > 0 else 0

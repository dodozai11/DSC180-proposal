def q1(text):
    """Return the win rate of blue-side teams."""
    lines = text.strip().split('\n')
    blue_games = 0
    blue_wins = 0
    
    for line in lines[1:]:  # Skip header
        parts = line.split(',')
        side = parts[2]
        result = parts[3]
        
        if side == 'blue':
            blue_games += 1
            if result == 'win':
                blue_wins += 1
    
    if blue_games == 0:
        return 0
    return blue_wins / blue_games


def q2(text):
    """Return the average length in minutes of games won by red-side teams."""
    lines = text.strip().split('\n')
    red_wins = []
    
    for line in lines[1:]:  # Skip header
        parts = line.split(',')
        side = parts[2]
        result = parts[3]
        game_minutes = float(parts[4])
        
        if side == 'red' and result == 'win':
            red_wins.append(game_minutes)
    
    if len(red_wins) == 0:
        return 0
    return sum(red_wins) / len(red_wins)


def q3(text):
    """Return the name of the team with the most wins."""
    lines = text.strip().split('\n')
    team_wins = {}
    
    for line in lines[1:]:  # Skip header
        parts = line.split(',')
        team = parts[1]
        result = parts[3]
        
        if team not in team_wins:
            team_wins[team] = 0
        
        if result == 'win':
            team_wins[team] += 1
    
    if not team_wins:
        return None
    return max(team_wins, key=team_wins.get)

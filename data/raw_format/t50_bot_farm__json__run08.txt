```python
import json
import re


def q1(text):
    """Return the name of the player with the highest CS per minute."""
    data = json.loads(text)
    player_with_max_cs = max(data, key=lambda x: x["cs_per_min"])
    return player_with_max_cs["player"]


def q2(text):
    """Return the average gold share of players who won."""
    data = json.loads(text)
    winning_players = [player for player in data if player["result"] == "win"]
    average_gold_share = sum(player["gold_share"] for player in winning_players) / len(winning_players)
    return average_gold_share


def q3(text):
    """Return how many players had more than 9 CS per minute."""
    data = json.loads(text)
    count = sum(1 for player in data if player["cs_per_min"] > 9)
    return count
```

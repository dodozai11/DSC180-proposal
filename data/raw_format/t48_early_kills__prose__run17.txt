```python
import re

def q1(text):
    """Returns the average number of kills at 15 minutes of red-side teams."""
    red_kills = []
    
    # Find all matches for red side teams
    pattern = r'In game \w+, .+ played on the red side and \w+, with (\d+) kills'
    matches = re.findall(pattern, text)
    
    for match in matches:
        red_kills.append(int(match))
    
    if red_kills:
        return sum(red_kills) / len(red_kills)
    return 0

def q2(text):
    """Returns the name of the team with the largest experience difference at 15 minutes."""
    # Find all team records with their experience differences
    pattern = r'In game \w+, (.+?) played on the (blue|red) side and \w+, with \d+ kills and an experience difference of (-?\d+) at 15 minutes'
    matches = re.findall(pattern, text)
    
    max_exp_diff = float('-inf')
    max_team = None
    
    for team, side, exp_diff in matches:
        exp_diff = int(exp_diff)
        if exp_diff > max_exp_diff:
            max_exp_diff = exp_diff
            max_team = team
    
    return max_team

def q3(text):
    """Returns how many teams won despite a negative experience difference at 15 minutes."""
    # Find all team records
    pattern = r'In game \w+, (.+?) played on the (blue|red) side and (\w+), with \d+ kills and an experience difference of (-?\d+) at 15 minutes'
    matches = re.findall(pattern, text)
    
    count = 0
    for team, side, result, exp_diff in matches:
        exp_diff = int(exp_diff)
        if result.lower() == 'won' and exp_diff < 0:
            count += 1
    
    return count
```

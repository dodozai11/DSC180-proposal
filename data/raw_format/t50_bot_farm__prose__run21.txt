```python
import re

def q1(text):
    """Return the name of the player with the highest CS per minute."""
    pattern = r'(\w+(?:\s+\w+)*) of .+ played .+ in the bot lane and (?:won|lost), with ([\d.]+) CS per minute'
    matches = re.findall(pattern, text)
    
    max_player = None
    max_cs = -1
    for player, cs in matches:
        cs_value = float(cs)
        if cs_value > max_cs:
            max_cs = cs_value
            max_player = player
    
    return max_player

def q2(text):
    """Return the average gold share of players who won."""
    pattern = r'(\w+(?:\s+\w+)*) of .+ played .+ in the bot lane and (won|lost), with [\d.]+ CS per minute and a gold share of ([\d.]+)'
    matches = re.findall(pattern, text)
    
    total_gold = 0
    count = 0
    for player, result, gold_share in matches:
        if result == 'won':
            total_gold += float(gold_share)
            count += 1
    
    if count == 0:
        return 0
    return total_gold / count

def q3(text):
    """Return how many players had more than 9 CS per minute."""
    pattern = r'(\w+(?:\s+\w+)*) of .+ played .+ in the bot lane and (?:won|lost), with ([\d.]+) CS per minute'
    matches = re.findall(pattern, text)
    
    count = 0
    for player, cs in matches:
        if float(cs) > 9:
            count += 1
    
    return count
```

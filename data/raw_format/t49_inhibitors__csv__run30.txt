def q1(text):
    """Return the average number of inhibitors destroyed by winning teams."""
    lines = text.strip().split('\n')
    header = lines[0].split(',')
    
    inhibitors_index = header.index('inhibitors')
    result_index = header.index('result')
    
    total_inhibitors = 0
    count = 0
    
    for line in lines[1:]:
        parts = line.split(',')
        if parts[result_index] == 'win':
            total_inhibitors += int(parts[inhibitors_index])
            count += 1
    
    return total_inhibitors / count if count > 0 else 0


def q2(text):
    """Return the average damage to towers of losing teams."""
    lines = text.strip().split('\n')
    header = lines[0].split(',')
    
    tower_damage_index = header.index('tower_damage')
    result_index = header.index('result')
    
    total_damage = 0
    count = 0
    
    for line in lines[1:]:
        parts = line.split(',')
        if parts[result_index] == 'loss':
            total_damage += int(parts[tower_damage_index])
            count += 1
    
    return total_damage / count if count > 0 else 0


def q3(text):
    """Return the game label of the longest game."""
    lines = text.strip().split('\n')
    header = lines[0].split(',')
    
    game_index = header.index('game')
    game_minutes_index = header.index('game_minutes')
    
    max_duration = -1
    longest_game = None
    
    for line in lines[1:]:
        parts = line.split(',')
        duration = float(parts[game_minutes_index])
        if duration > max_duration:
            max_duration = duration
            longest_game = parts[game_index]
    
    return longest_game

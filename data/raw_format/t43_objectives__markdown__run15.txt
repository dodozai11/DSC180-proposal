def q1(text):
    """Return the win rate of teams that took the first dragon."""
    lines = text.strip().split('\n')
    
    # Skip header row
    first_dragon_wins = 0
    first_dragon_total = 0
    
    for line in lines[1:]:
        parts = line.split('|')
        # Remove empty strings from split (due to leading/trailing |)
        parts = [p.strip() for p in parts if p.strip()]
        
        if len(parts) >= 3:
            first_dragon = parts[2]
            result = parts[1]
            
            if first_dragon == 'yes':
                first_dragon_total += 1
                if result == 'win':
                    first_dragon_wins += 1
    
    if first_dragon_total == 0:
        return 0
    
    return first_dragon_wins / first_dragon_total


def q2(text):
    """Return the average number of barons taken by winning teams."""
    lines = text.strip().split('\n')
    
    # Skip header row
    barons_sum = 0
    win_count = 0
    
    for line in lines[1:]:
        parts = line.split('|')
        # Remove empty strings from split (due to leading/trailing |)
        parts = [p.strip() for p in parts if p.strip()]
        
        if len(parts) >= 6:
            result = parts[2]
            barons = int(parts[5])
            
            if result == 'win':
                barons_sum += barons
                win_count += 1
    
    if win_count == 0:
        return 0
    
    return barons_sum / win_count


def q3(text):
    """Return the total number of towers destroyed across all rows."""
    lines = text.strip().split('\n')
    
    # Skip header row
    towers_total = 0
    
    for line in lines[1:]:
        parts = line.split('|')
        # Remove empty strings from split (due to leading/trailing |)
        parts = [p.strip() for p in parts if p.strip()]
        
        if len(parts) >= 7:
            towers = int(parts[6])
            towers_total += towers
    
    return towers_total

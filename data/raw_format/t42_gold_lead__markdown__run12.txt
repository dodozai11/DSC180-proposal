```python
import csv
from io import StringIO

def q1(text):
    """Returns the average gold difference at 15 minutes of the winning teams."""
    lines = text.strip().split('\n')
    reader = csv.DictReader(StringIO('\n'.join(lines)))
    
    winning_gold_diffs = []
    for row in reader:
        if row['result'] == 'win':
            winning_gold_diffs.append(int(row['gold_diff_15']))
    
    return sum(winning_gold_diffs) / len(winning_gold_diffs) if winning_gold_diffs else 0


def q2(text):
    """Returns how many teams won despite a negative gold difference at 15 minutes."""
    lines = text.strip().split('\n')
    reader = csv.DictReader(StringIO('\n'.join(lines)))
    
    count = 0
    for row in reader:
        if row['result'] == 'win' and int(row['gold_diff_15']) < 0:
            count += 1
    
    return count


def q3(text):
    """Returns the win rate of teams that were ahead in gold at 15 minutes."""
    lines = text.strip().split('\n')
    reader = csv.DictReader(StringIO('\n'.join(lines)))
    
    ahead_in_gold = 0
    ahead_and_won = 0
    
    for row in reader:
        if int(row['gold_diff_15']) > 0:
            ahead_in_gold += 1
            if row['result'] == 'win':
                ahead_and_won += 1
    
    return ahead_and_won / ahead_in_gold if ahead_in_gold > 0 else 0
```

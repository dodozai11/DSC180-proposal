def parse_data(text):
    """Parse the table data from text format."""
    lines = text.strip().split('\n')
    # Skip header and separator lines
    data = []
    for line in lines[2:]:  # Skip first two lines (header and separator)
        if line.strip() and '|' in line:
            parts = [p.strip() for p in line.split('|')[1:-1]]  # Remove empty strings from split
            if len(parts) == 6:
                data.append({
                    'player': parts[0],
                    'team': parts[1],
                    'champion': parts[2],
                    'result': parts[3],
                    'cs_per_min': float(parts[4]),
                    'gold_share': float(parts[5])
                })
    return data


def q1(text):
    """Return the name of the player with the highest CS per minute."""
    data = parse_data(text)
    return max(data, key=lambda x: x['cs_per_min'])['player']


def q2(text):
    """Return the average gold share of players who won."""
    data = parse_data(text)
    winners = [player for player in data if player['result'] == 'win']
    if not winners:
        return 0
    avg_gold_share = sum(player['gold_share'] for player in winners) / len(winners)
    return avg_gold_share


def q3(text):
    """Return how many players had more than 9 CS per minute."""
    data = parse_data(text)
    count = sum(1 for player in data if player['cs_per_min'] > 9)
    return count

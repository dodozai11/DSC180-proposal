def q1(text):
    """Returns the average gold difference at 15 minutes of the winning teams."""
    lines = text.strip().split('\n')
    # Skip header row
    winning_gold_diffs = []
    
    for line in lines[1:]:
        parts = line.split('|')
        # Remove empty strings from splitting on |
        parts = [p.strip() for p in parts if p.strip()]
        
        result = parts[3]
        gold_diff_15 = int(parts[4])
        
        if result == 'win':
            winning_gold_diffs.append(gold_diff_15)
    
    return sum(winning_gold_diffs) / len(winning_gold_diffs)


def q2(text):
    """Returns how many teams won despite a negative gold difference at 15 minutes."""
    lines = text.strip().split('\n')
    # Skip header row
    count = 0
    
    for line in lines[1:]:
        parts = line.split('|')
        # Remove empty strings from splitting on |
        parts = [p.strip() for p in parts if p.strip()]
        
        result = parts[3]
        gold_diff_15 = int(parts[4])
        
        if result == 'win' and gold_diff_15 < 0:
            count += 1
    
    return count


def q3(text):
    """Returns the win rate of teams that were ahead in gold at 15 minutes."""
    lines = text.strip().split('\n')
    # Skip header row
    ahead_in_gold = 0
    ahead_and_won = 0
    
    for line in lines[1:]:
        parts = line.split('|')
        # Remove empty strings from splitting on |
        parts = [p.strip() for p in parts if p.strip()]
        
        result = parts[3]
        gold_diff_15 = int(parts[4])
        
        if gold_diff_15 > 0:
            ahead_in_gold += 1
            if result == 'win':
                ahead_and_won += 1
    
    if ahead_in_gold == 0:
        return 0
    
    return ahead_and_won / ahead_in_gold

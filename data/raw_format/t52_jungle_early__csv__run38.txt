def q1(text):
    lines = text.strip().split('\n')
    header = lines[0].split(',')
    
    # Find column indices
    cs_diff_idx = header.index('cs_diff_10')
    result_idx = header.index('result')
    
    positive_cs_wins = 0
    positive_cs_total = 0
    
    for line in lines[1:]:
        values = line.split(',')
        cs_diff = int(values[cs_diff_idx])
        result = values[result_idx]
        
        if cs_diff > 0:
            positive_cs_total += 1
            if result == 'win':
                positive_cs_wins += 1
    
    if positive_cs_total == 0:
        return 0
    
    return positive_cs_wins / positive_cs_total


def q2(text):
    lines = text.strip().split('\n')
    header = lines[0].split(',')
    
    # Find column index
    kills_idx = header.index('kills_at_10')
    
    total_kills = 0
    
    for line in lines[1:]:
        values = line.split(',')
        kills = int(values[kills_idx])
        total_kills += kills
    
    return total_kills


def q3(text):
    lines = text.strip().split('\n')
    header = lines[0].split(',')
    
    # Find column indices
    champion_idx = header.index('champion')
    assists_idx = header.index('assists_at_10')
    
    champion_assists = {}
    
    for line in lines[1:]:
        values = line.split(',')
        champion = values[champion_idx]
        assists = int(values[assists_idx])
        
        if champion not in champion_assists:
            champion_assists[champion] = 0
        champion_assists[champion] += assists
    
    return max(champion_assists, key=champion_assists.get)

import xml.etree.ElementTree as ET

def q1(text):
    """Return the win rate of teams that took the first tower."""
    root = ET.fromstring(f"<root>{text}</root>")
    
    first_tower_teams = []
    for row in root.findall('row'):
        if row.find('first_tower').text == 'yes':
            result = row.find('result').text
            first_tower_teams.append(result)
    
    if len(first_tower_teams) == 0:
        return 0
    
    wins = sum(1 for r in first_tower_teams if r == 'win')
    return wins / len(first_tower_teams)


def q2(text):
    """Return the average number of void grubs taken by winning teams."""
    root = ET.fromstring(f"<root>{text}</root>")
    
    winning_void_grubs = []
    for row in root.findall('row'):
        if row.find('result').text == 'win':
            void_grubs = int(row.find('void_grubs').text)
            winning_void_grubs.append(void_grubs)
    
    if len(winning_void_grubs) == 0:
        return 0
    
    return sum(winning_void_grubs) / len(winning_void_grubs)


def q3(text):
    """Return how many teams took at least one herald and still lost."""
    root = ET.fromstring(f"<root>{text}</root>")
    
    count = 0
    for row in root.findall('row'):
        heralds = int(row.find('heralds').text)
        result = row.find('result').text
        if heralds >= 1 and result == 'loss':
            count += 1
    
    return count

import re

def q1(text):
    """Returns the win rate of teams that took the first dragon."""
    first_dragon_games = []
    
    # Parse the text to extract game data
    pattern = r'In game (G\d+), (.+?) (won|lost); they (took|did not take) the first dragon and finished with (\d+) dragons, (\d+) barons and (\d+) towers\.'
    matches = re.findall(pattern, text)
    
    for match in matches:
        game_id, team_name, result, first_dragon, dragons, barons, towers = match
        took_first_dragon = first_dragon == "took"
        won = result == "won"
        
        if took_first_dragon:
            first_dragon_games.append(won)
    
    if len(first_dragon_games) == 0:
        return 0
    
    win_rate = sum(first_dragon_games) / len(first_dragon_games)
    return win_rate


def q2(text):
    """Returns the average number of barons taken by winning teams."""
    winning_barons = []
    
    # Parse the text to extract game data
    pattern = r'In game (G\d+), (.+?) (won|lost); they (took|did not take) the first dragon and finished with (\d+) dragons, (\d+) barons and (\d+) towers\.'
    matches = re.findall(pattern, text)
    
    for match in matches:
        game_id, team_name, result, first_dragon, dragons, barons, towers = match
        won = result == "won"
        baron_count = int(barons)
        
        if won:
            winning_barons.append(baron_count)
    
    if len(winning_barons) == 0:
        return 0
    
    average_barons = sum(winning_barons) / len(winning_barons)
    return average_barons

    for match in matches:
        game_id, team_name, result, first_dragon, dragons, barons, towers = match
        tower_count = int(towers)
        total_towers += tower_count
    
    return total_towers

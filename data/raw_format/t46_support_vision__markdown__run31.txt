def q1(text):
    lines = text.strip().split('\n')
    headers = lines[0].split('|')[1:-1]
    headers = [h.strip() for h in headers]
    
    vision_idx = headers.index('vision_score')
    result_idx = headers.index('result')
    
    wins_vision = []
    losses_vision = []
    
    for line in lines[2:]:
        if line.strip() == '':
            continue
        parts = line.split('|')[1:-1]
        parts = [p.strip() for p in parts]
        
        result = parts[result_idx]
        vision = float(parts[vision_idx])
        
        if result == 'win':
            wins_vision.append(vision)
        else:
            losses_vision.append(vision)
    
    avg_wins = sum(wins_vision) / len(wins_vision) if wins_vision else 0
    avg_losses = sum(losses_vision) / len(losses_vision) if losses_vision else 0
    
    return avg_wins - avg_losses


def q2(text):
    lines = text.strip().split('\n')
    headers = lines[0].split('|')[1:-1]
    headers = [h.strip() for h in headers]
    
    player_idx = headers.index('player')
    wards_idx = headers.index('wards_placed')
    
    max_wards = -1
    max_player = None
    
    for line in lines[2:]:
        if line.strip() == '':
            continue
        parts = line.split('|')[1:-1]
        parts = [p.strip() for p in parts]
        
        player = parts[player_idx]
        wards = int(parts[wards_idx])
        
        if wards > max_wards:
            max_wards = wards
            max_player = player
    
    return max_player


def q3(text):
    lines = text.strip().split('\n')
    headers = lines[0].split('|')[1:-1]
    headers = [h.strip() for h in headers]
    
    control_wards_idx = headers.index('control_wards')
    
    count = 0
    
    for line in lines[2:]:
        if line.strip() == '':
            continue
        parts = line.split('|')[1:-1]
        parts = [p.strip() for p in parts]
        
        control_wards = int(parts[control_wards_idx])
        
        if control_wards >= 15:
            count += 1
    
    return count

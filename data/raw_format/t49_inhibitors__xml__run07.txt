import xml.etree.ElementTree as ET

def q1(text):
    """Return the average number of inhibitors destroyed by winning teams."""
    root = ET.fromstring(f"<root>{text}</root>")
    winning_inhibitors = [int(row.find('inhibitors').text) 
                          for row in root.findall('row') 
                          if row.find('result').text == 'win']
    return sum(winning_inhibitors) / len(winning_inhibitors)

def q2(text):
    """Return the average damage to towers of losing teams."""
    root = ET.fromstring(f"<root>{text}</root>")
    losing_damage = [int(row.find('tower_damage').text) 
                     for row in root.findall('row') 
                     if row.find('result').text == 'loss']
    return sum(losing_damage) / len(losing_damage)

def q3(text):
    """Return the game label of the longest game."""
    root = ET.fromstring(f"<root>{text}</root>")
    longest_game = max(root.findall('row'), 
                       key=lambda row: float(row.find('game_minutes').text))
    return longest_game.find('game').text

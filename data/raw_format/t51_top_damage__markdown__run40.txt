def q1(text):
    """Return the average damage per minute of players who lost."""
    lines = text.strip().split('\n')
    # Skip header row
    data_lines = lines[1:]
    
    total_damage = 0
    count = 0
    
    for line in data_lines:
        parts = line.split('|')
        # Remove empty strings from split and strip whitespace
        parts = [p.strip() for p in parts if p.strip()]
        
        if len(parts) >= 4:
            result = parts[3]
            damage_per_min = float(parts[4])
            
            if result == 'loss':
                total_damage += damage_per_min
                count += 1
    
    if count == 0:
        return 0
    return total_damage / count


def q2(text):
    """Return the name of the player who took the most damage per minute."""
    lines = text.strip().split('\n')
    # Skip header row
    data_lines = lines[1:]
    
    max_damage = -1
    max_player = None
    
    for line in data_lines:
        parts = line.split('|')
        # Remove empty strings from split and strip whitespace
        parts = [p.strip() for p in parts if p.strip()]
        
        if len(parts) >= 5:
            player = parts[0]
            damage_taken_per_min = float(parts[5])
            
            if damage_taken_per_min > max_damage:
                max_damage = damage_taken_per_min
                max_player = player
    
    return max_player


def q3(text):
    """Return the number of different teams."""
    lines = text.strip().split('\n')
    # Skip header row
    data_lines = lines[1:]
    
    teams = set()
    
    for line in data_lines:
        parts = line.split('|')
        # Remove empty strings from split and strip whitespace
        parts = [p.strip() for p in parts if p.strip()]
        
        if len(parts) >= 2:
            team = parts[1]
            teams.add(team)
    
    return len(teams)

def q1(text):
    """Returns the number of different champions played."""
    champions = set()
    for line in text.strip().split('\n'):
        parts = line.split()
        # Find "played" and get the next word which is the champion
        for i, word in enumerate(parts):
            if word == "played" and i + 1 < len(parts):
                champion = parts[i + 1]
                champions.add(champion)
                break
    return len(champions)


def q2(text):
    """Returns the champion with the best win rate among champions played at least twice."""
    champion_stats = {}
    
    for line in text.strip().split('\n'):
        parts = line.split()
        
        # Find champion name
        champion = None
        for i, word in enumerate(parts):
            if word == "played" and i + 1 < len(parts):
                champion = parts[i + 1]
                break
        
        if not champion:
            continue
        
        # Find win/loss
        won = "won" in parts
        
        # Initialize champion stats if not seen
        if champion not in champion_stats:
            champion_stats[champion] = {"wins": 0, "total": 0}
        
        champion_stats[champion]["total"] += 1
        if won:
            champion_stats[champion]["wins"] += 1
    
    # Filter champions played at least twice
    candidates = {champ: stats for champ, stats in champion_stats.items() if stats["total"] >= 2}
    
    if not candidates:
        return None
    
    # Find champion with best win rate
    best_champion = max(candidates.items(), key=lambda x: x[1]["wins"] / x[1]["total"])
    return best_champion[0]


def q3(text):
    """Returns the average number of deaths in games that were lost."""
    deaths_in_losses = []
    
    for line in text.strip().split('\n'):
        parts = line.split()
        
        # Check if game was lost
        lost = "lost" in parts
        
        if not lost:
            continue
        
        # Find deaths - it's the number before "times" at the end
        for i, word in enumerate(parts):
            if word == "dying" and i + 1 < len(parts):
                deaths = int(parts[i + 1])
                deaths_in_losses.append(deaths)
                break
    
    if not deaths_in_losses:
        return 0
    
    return sum(deaths_in_losses) / len(deaths_in_losses)

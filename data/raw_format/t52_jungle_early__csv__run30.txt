def q1(text):
    lines = text.strip().split('\n')
    header = lines[0].split(',')
    
    cs_diff_10_idx = header.index('cs_diff_10')
    result_idx = header.index('result')
    
    positive_cs_diff_rows = []
    for line in lines[1:]:
        parts = line.split(',')
        cs_diff_10 = int(parts[cs_diff_10_idx])
        result = parts[result_idx]
        if cs_diff_10 > 0:
            positive_cs_diff_rows.append(result)
    
    if len(positive_cs_diff_rows) == 0:
        return 0
    
    wins = sum(1 for result in positive_cs_diff_rows if result == 'win')
    return wins / len(positive_cs_diff_rows)


def q2(text):
    lines = text.strip().split('\n')
    header = lines[0].split(',')
    
    kills_at_10_idx = header.index('kills_at_10')
    
    total_kills = 0
    for line in lines[1:]:
        parts = line.split(',')
        kills_at_10 = int(parts[kills_at_10_idx])
        total_kills += kills_at_10
    
    return total_kills


def q3(text):
    lines = text.strip().split('\n')
    header = lines[0].split(',')
    
    champion_idx = header.index('champion')
    assists_at_10_idx = header.index('assists_at_10')
    
    champion_assists = {}
    for line in lines[1:]:
        parts = line.split(',')
        champion = parts[champion_idx]
        assists_at_10 = int(parts[assists_at_10_idx])
        
        if champion not in champion_assists:
            champion_assists[champion] = 0
        champion_assists[champion] += assists_at_10
    
    return max(champion_assists, key=champion_assists.get)

```python
import re

def q1(text):
    """Return the average gold difference at 15 minutes of the winning teams."""
    winning_gold_diffs = []
    
    # Find all game entries
    pattern = r"In game (\w+), (.+?) played on the (blue|red) side and (won|lost), with a gold difference of ([-\d]+) at 15 minutes\."
    matches = re.findall(pattern, text)
    
    for match in matches:
        game_id, team, side, result, gold_diff = match
        gold_diff = int(gold_diff)
        
        if result == "won":
            winning_gold_diffs.append(gold_diff)
    
    if not winning_gold_diffs:
        return 0
    
    return sum(winning_gold_diffs) / len(winning_gold_diffs)


def q2(text):
    """Return how many teams won despite a negative gold difference at 15 minutes."""
    count = 0
    
    # Find all game entries
    pattern = r"In game (\w+), (.+?) played on the (blue|red) side and (won|lost), with a gold difference of ([-\d]+) at 15 minutes\."
    matches = re.findall(pattern, text)
    
    for match in matches:
        game_id, team, side, result, gold_diff = match
        gold_diff = int(gold_diff)
        
        if result == "won" and gold_diff < 0:
            count += 1
    
    return count


def q3(text):
    """Return the win rate of teams that were ahead in gold at 15 minutes."""
    ahead_wins = 0
    ahead_total = 0
    
    # Find all game entries
    pattern = r"In game (\w+), (.+?) played on the (blue|red) side and (won|lost), with a gold difference of ([-\d]+) at 15 minutes\."
    matches = re.findall(pattern, text)
    
    for match in matches:
        game_id, team, side, result, gold_diff = match
        gold_diff = int(gold_diff)
        
        # Teams ahead in gold have positive gold difference
        if gold_diff > 0:
            ahead_total += 1
            if result == "won":
                ahead_wins += 1
    
    if ahead_total == 0:
        return 0
    
    return ahead_wins / ahead_total
```

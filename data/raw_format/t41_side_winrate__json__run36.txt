import json
import re

def q1(text):
    """Returns the win rate of blue-side teams"""
    data = json.loads(text)
    blue_games = [game for game in data if game["side"] == "blue"]
    blue_wins = sum(1 for game in blue_games if game["result"] == "win")
    return blue_wins / len(blue_games) if blue_games else 0

def q2(text):
    """Returns the average length in minutes of games won by red-side teams"""
    data = json.loads(text)
    red_wins = [game for game in data if game["side"] == "red" and game["result"] == "win"]
    if not red_wins:
        return 0
    return sum(game["game_minutes"] for game in red_wins) / len(red_wins)

def q3(text):
    """Returns the name of the team with the most wins"""
    data = json.loads(text)
    team_wins = {}
    for game in data:
        if game["result"] == "win":
            team = game["team"]
            team_wins[team] = team_wins.get(team, 0) + 1
    return max(team_wins, key=team_wins.get) if team_wins else None

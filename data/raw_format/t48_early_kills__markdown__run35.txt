def q1(text):
    """Return the average number of kills at 15 minutes of red-side teams."""
    lines = text.strip().split('\n')
    # Skip header line
    red_kills = []
    for line in lines[1:]:
        parts = line.split('|')
        # Remove empty strings from split
        parts = [p.strip() for p in parts if p.strip()]
        # parts: [game, team, side, result, kills_at_15, xp_diff_15]
        side = parts[2]
        kills_at_15 = int(parts[4])
        if side == 'red':
            red_kills.append(kills_at_15)
    
    if not red_kills:
        return 0
    return sum(red_kills) / len(red_kills)


def q2(text):
    """Return the name of the team with the largest experience difference at 15 minutes."""
    lines = text.strip().split('\n')
    # Skip header line
    max_xp_diff = None
    max_team = None
    
    for line in lines[1:]:
        parts = line.split('|')
        # Remove empty strings from split
        parts = [p.strip() for p in parts if p.strip()]
        # parts: [game, team, side, result, kills_at_15, xp_diff_15]
        team = parts[1]
        xp_diff_15 = int(parts[5])
        
        if max_xp_diff is None or abs(xp_diff_15) > abs(max_xp_diff):
            max_xp_diff = xp_diff_15
            max_team = team
    
    return max_team


def q3(text):
    """Return how many teams won despite a negative experience difference at 15 minutes."""
    lines = text.strip().split('\n')
    # Skip header line
    count = 0
    
    for line in lines[1:]:
        parts = line.split('|')
        # Remove empty strings from split
        parts = [p.strip() for p in parts if p.strip()]
        # parts: [game, team, side, result, kills_at_15, xp_diff_15]
        result = parts[3]
        xp_diff_15 = int(parts[5])
        
        if result == 'win' and xp_diff_15 < 0:
            count += 1
    
    return count

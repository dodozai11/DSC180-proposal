import json
import re
from collections import Counter

def q1(text):
    """Return the win rate of blue-side teams."""
    data = json.loads(text)
    blue_games = [row for row in data if row['side'] == 'blue']
    if not blue_games:
        return 0
    blue_wins = sum(1 for row in blue_games if row['result'] == 'win')
    return blue_wins / len(blue_games)

def q2(text):
    """Return the average length in minutes of games won by red-side teams."""
    data = json.loads(text)
    red_wins = [row for row in data if row['side'] == 'red' and row['result'] == 'win']
    if not red_wins:
        return 0
    total_minutes = sum(row['game_minutes'] for row in red_wins)
    return total_minutes / len(red_wins)

def q3(text):
    """Return the name of the team with the most wins."""
    data = json.loads(text)
    wins = [row['team'] for row in data if row['result'] == 'win']
    if not wins:
        return None
    win_counts = Counter(wins)
    return win_counts.most_common(1)[0][0]

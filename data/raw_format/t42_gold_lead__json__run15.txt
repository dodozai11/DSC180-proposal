import json
import re

def q1(text):
    """Returns the average gold difference at 15 minutes of the winning teams."""
    data = json.loads(text)
    winning_gold_diffs = [row["gold_diff_15"] for row in data if row["result"] == "win"]
    return sum(winning_gold_diffs) / len(winning_gold_diffs)

def q2(text):
    """Returns how many teams won despite a negative gold difference at 15 minutes."""
    data = json.loads(text)
    wins_with_negative_gold = sum(1 for row in data if row["result"] == "win" and row["gold_diff_15"] < 0)
    return wins_with_negative_gold

def q3(text):
    """Returns the win rate of teams that were ahead in gold at 15 minutes."""
    data = json.loads(text)
    teams_ahead_in_gold = [row for row in data if row["gold_diff_15"] > 0]
    if len(teams_ahead_in_gold) == 0:
        return 0.0
    wins_among_ahead = sum(1 for row in teams_ahead_in_gold if row["result"] == "win")
    return wins_among_ahead / len(teams_ahead_in_gold)

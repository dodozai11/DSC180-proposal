import re
from collections import defaultdict

def parse_data(text):
    """Parse the table data from text format."""
    lines = text.strip().split('\n')
    # Skip header and separator lines
    data = []
    for line in lines[2:]:  # Skip first two lines (header and separator)
        if line.strip() and '|' in line:
            parts = [p.strip() for p in line.split('|')[1:-1]]  # Remove empty strings from split
            if len(parts) == 5:
                game, team, side, result, gold_diff = parts
                data.append({
                    'game': game,
                    'team': team,
                    'side': side,
                    'result': result,
                    'gold_diff_15': int(gold_diff)
                })
    return data

def q1(text):
    """Return the average gold difference at 15 minutes of the winning teams."""
    data = parse_data(text)
    winning_gold_diffs = [row['gold_diff_15'] for row in data if row['result'] == 'win']
    return sum(winning_gold_diffs) / len(winning_gold_diffs) if winning_gold_diffs else 0

def q2(text):
    """Return how many teams won despite a negative gold difference at 15 minutes."""
    data = parse_data(text)
    count = sum(1 for row in data if row['result'] == 'win' and row['gold_diff_15'] < 0)
    return count

def q3(text):
    """Return the win rate of teams that were ahead in gold at 15 minutes."""
    data = parse_data(text)
    ahead_in_gold = [row for row in data if row['gold_diff_15'] > 0]
    if not ahead_in_gold:
        return 0
    wins = sum(1 for row in ahead_in_gold if row['result'] == 'win')
    return wins / len(ahead_in_gold)

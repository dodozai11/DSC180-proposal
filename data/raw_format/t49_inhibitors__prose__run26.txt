import re

def q1(text):
    """Return the average number of inhibitors destroyed by winning teams."""
    winning_inhibitors = []
    
    # Find all matches where a team won
    pattern = r'(\w+(?:\s+\w+)*)\s+won\s+after\s+([\d.]+)\s+minutes,\s+destroying\s+(\d+)\s+inhibitors'
    matches = re.findall(pattern, text)
    
    for team, duration, inhibitors in matches:
        winning_inhibitors.append(int(inhibitors))
    
    return sum(winning_inhibitors) / len(winning_inhibitors) if winning_inhibitors else 0


def q2(text):
    """Return the average damage to towers of losing teams."""
    losing_damages = []
    
    # Find all matches where a team lost
    pattern = r'(\w+(?:\s+\w+)*)\s+lost\s+after\s+([\d.]+)\s+minutes,\s+destroying\s+(\d+)\s+inhibitors\s+and\s+dealing\s+(\d+)\s+damage\s+to\s+towers'
    matches = re.findall(pattern, text)
    
    for team, duration, inhibitors, damage in matches:
        losing_damages.append(int(damage))
    
    return sum(losing_damages) / len(losing_damages) if losing_damages else 0


def q3(text):
    """Return the game label of the longest game."""
    games = {}
    
    # Find all game entries (both wins and losses)
    pattern = r'In\s+game\s+(G\d+),\s+(\w+(?:\s+\w+)*)\s+(won|lost)\s+after\s+([\d.]+)\s+minutes'
    matches = re.findall(pattern, text)
    
    for game_label, team, result, duration in matches:
        duration = float(duration)
        if game_label not in games:
            games[game_label] = duration
    
    longest_game = max(games.items(), key=lambda x: x[1])
    return longest_game[0]

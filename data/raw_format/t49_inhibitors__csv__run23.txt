def q1(text):
    """Returns the average number of inhibitors destroyed by winning teams."""
    lines = text.strip().split('\n')
    header = lines[0].split(',')
    
    inhibitors_by_winners = []
    for line in lines[1:]:
        values = line.split(',')
        row = dict(zip(header, values))
        if row['result'] == 'win':
            inhibitors_by_winners.append(int(row['inhibitors']))
    
    return sum(inhibitors_by_winners) / len(inhibitors_by_winners)


def q2(text):
    """Returns the average damage to towers of losing teams."""
    lines = text.strip().split('\n')
    header = lines[0].split(',')
    
    tower_damage_by_losers = []
    for line in lines[1:]:
        values = line.split(',')
        row = dict(zip(header, values))
        if row['result'] == 'loss':
            tower_damage_by_losers.append(int(row['tower_damage']))
    
    return sum(tower_damage_by_losers) / len(tower_damage_by_losers)


def q3(text):
    """Returns the game label of the longest game."""
    lines = text.strip().split('\n')
    header = lines[0].split(',')
    
    max_duration = -1
    longest_game = None
    for line in lines[1:]:
        values = line.split(',')
        row = dict(zip(header, values))
        duration = float(row['game_minutes'])
        if duration > max_duration:
            max_duration = duration
            longest_game = row['game']
    
    return longest_game

def q1(text):
    """Return the win rate of teams that took the first dragon."""
    lines = text.strip().split('\n')
    
    # Skip header
    data = []
    for line in lines[1:]:
        parts = line.split(',')
        if len(parts) >= 6:
            data.append({
                'first_dragon': parts[2],
                'result': parts[3]
            })
    
    # Filter teams that took first dragon
    first_dragon_teams = [row for row in data if row['first_dragon'] == 'yes']
    
    if len(first_dragon_teams) == 0:
        return 0
    
    # Count wins among first dragon teams
    wins = sum(1 for row in first_dragon_teams if row['result'] == 'win')
    
    return wins / len(first_dragon_teams)


def q2(text):
    """Return the average number of barons taken by winning teams."""
    lines = text.strip().split('\n')
    
    # Skip header
    data = []
    for line in lines[1:]:
        parts = line.split(',')
        if len(parts) >= 6:
            data.append({
                'result': parts[2],
                'barons': int(parts[5])
            })
    
    # Filter winning teams
    winning_teams = [row for row in data if row['result'] == 'win']
    
    if len(winning_teams) == 0:
        return 0
    
    # Calculate average barons
    total_barons = sum(row['barons'] for row in winning_teams)
    
    return total_barons / len(winning_teams)


def q3(text):
    """Return the total number of towers destroyed across all rows."""
    lines = text.strip().split('\n')
    
    # Skip header
    total_towers = 0
    for line in lines[1:]:
        parts = line.split(',')
        if len(parts) >= 7:
            total_towers += int(parts[6])
    
    return total_towers

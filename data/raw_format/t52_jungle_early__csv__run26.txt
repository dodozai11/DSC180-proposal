def q1(text):
    """Return the win rate of players with a positive CS difference at 10 minutes."""
    lines = text.strip().split('\n')
    header = lines[0]
    rows = lines[1:]
    
    positive_cs_diff_rows = []
    for row in rows:
        parts = row.split(',')
        cs_diff_10 = int(parts[3])
        result = parts[2]
        if cs_diff_10 > 0:
            positive_cs_diff_rows.append(result)
    
    if len(positive_cs_diff_rows) == 0:
        return 0
    
    wins = sum(1 for result in positive_cs_diff_rows if result == 'win')
    return wins / len(positive_cs_diff_rows)


def q2(text):
    """Return the total number of kills at 10 minutes across all rows."""
    lines = text.strip().split('\n')
    header = lines[0]
    rows = lines[1:]
    
    total_kills = 0
    for row in rows:
        parts = row.split(',')
        kills_at_10 = int(parts[4])
        total_kills += kills_at_10
    
    return total_kills


def q3(text):
    """Return the name of the champion with the most assists at 10 minutes in total."""
    lines = text.strip().split('\n')
    header = lines[0]
    rows = lines[1:]
    
    champion_assists = {}
    for row in rows:
        parts = row.split(',')
        champion = parts[1]
        assists_at_10 = int(parts[5])
        
        if champion not in champion_assists:
            champion_assists[champion] = 0
        champion_assists[champion] += assists_at_10
    
    max_champion = max(champion_assists, key=champion_assists.get)
    return max_champion

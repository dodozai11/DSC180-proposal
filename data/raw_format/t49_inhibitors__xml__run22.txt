import xml.etree.ElementTree as ET
from statistics import mean

def q1(text):
    """Returns the average number of inhibitors destroyed by winning teams."""
    root = ET.fromstring(f"<root>{text}</root>")
    winning_inhibitors = [
        int(row.find("inhibitors").text)
        for row in root.findall("row")
        if row.find("result").text == "win"
    ]
    return mean(winning_inhibitors)

def q2(text):
    """Returns the average damage to towers of losing teams."""
    root = ET.fromstring(f"<root>{text}</root>")
    losing_damage = [
        int(row.find("tower_damage").text)
        for row in root.findall("row")
        if row.find("result").text == "loss"
    ]
    return mean(losing_damage)

def q3(text):
    """Returns the game label of the longest game."""
    root = ET.fromstring(f"<root>{text}</root>")
    longest_game = max(
        root.findall("row"),
        key=lambda row: float(row.find("game_minutes").text)
    )
    return longest_game.find("game").text

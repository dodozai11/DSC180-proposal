def q1(text):
    """Returns the average number of inhibitors destroyed by winning teams."""
    lines = text.strip().split('\n')
    # Skip header row
    data_lines = lines[1:]
    
    winning_inhibitors = []
    for line in data_lines:
        parts = line.split('|')
        parts = [p.strip() for p in parts]
        # Skip empty first and last elements from split
        game, team, result, game_minutes, inhibitors, tower_damage = parts[1:7]
        
        if result == 'win':
            winning_inhibitors.append(int(inhibitors))
    
    return sum(winning_inhibitors) / len(winning_inhibitors)


def q2(text):

    lines = text.strip().split('\n')

def q3(text):
    """Returns the game label of the longest game."""
    lines = text.strip().split('\n')
    # Skip header row
    data_lines = lines[1:]
    
    max_duration = -1
    longest_game = None
    
    for line in data_lines:
        parts = line.split('|')
        parts = [p.strip() for p in parts]
        # Skip empty first and last elements from split
        game, team, result, game_minutes, inhibitors, tower_damage = parts[1:7]
        
        duration = float(game_minutes)
        if duration > max_duration:
            max_duration = duration
            longest_game = game
    
    return longest_game

def q1(text):
    lines = text.strip().split('\n')
    header = lines[0].split('|')
    header = [h.strip() for h in header if h.strip()]
    
    max_cs = -1
    max_player = None
    
    for line in lines[2:]:
        if not line.strip():
            continue
        cols = line.split('|')
        cols = [c.strip() for c in cols if c.strip()]
        if len(cols) < len(header):
            continue
        
        player = cols[0]
        cs_per_min = float(cols[4])
        
        if cs_per_min > max_cs:
            max_cs = cs_per_min
            max_player = player
    
    return max_player


def q2(text):
    lines = text.strip().split('\n')
    header = lines[0].split('|')
    header = [h.strip() for h in header if h.strip()]
    
    total_gold_share = 0
    win_count = 0
    
    for line in lines[2:]:
        if not line.strip():
            continue
        cols = line.split('|')
        cols = [c.strip() for c in cols if c.strip()]
        if len(cols) < len(header):
            continue
        
        result = cols[3]
        gold_share = float(cols[5])
        
        if result == 'win':
            total_gold_share += gold_share
            win_count += 1
    
    return total_gold_share / win_count if win_count > 0 else 0

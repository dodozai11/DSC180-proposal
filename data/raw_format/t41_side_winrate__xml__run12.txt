import xml.etree.ElementTree as ET
from collections import Counter

def q1(text):
    """Returns the win rate of blue-side teams"""
    root = ET.fromstring(f"<root>{text}</root>")
    rows = root.findall("row")
    
    blue_games = 0
    blue_wins = 0
    
    for row in rows:
        side = row.find("side").text
        result = row.find("result").text
        
        if side == "blue":
            blue_games += 1
            if result == "win":
                blue_wins += 1
    
    if blue_games == 0:
        return 0
    
    return blue_wins / blue_games


def q2(text):
    """Returns the average length in minutes of games won by red-side teams"""
    root = ET.fromstring(f"<root>{text}</root>")
    rows = root.findall("row")
    
    red_win_times = []
    
    for row in rows:
        side = row.find("side").text
        result = row.find("result").text
        game_minutes = float(row.find("game_minutes").text)
        
        if side == "red" and result == "win":
            red_win_times.append(game_minutes)
    
    if len(red_win_times) == 0:
        return 0
    
    return sum(red_win_times) / len(red_win_times)


def q3(text):
    """Returns the name of the team with the most wins"""
    root = ET.fromstring(f"<root>{text}</root>")
    rows = root.findall("row")
    
    team_wins = Counter()
    
    for row in rows:
        team = row.find("team").text
        result = row.find("result").text
        
        if result == "win":
            team_wins[team] += 1
    
    if not team_wins:
        return None
    
    return team_wins.most_common(1)[0][0]

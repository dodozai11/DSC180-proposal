def q1(text):
    """Return the win rate of teams that took the first dragon."""
    lines = text.strip().split('\n')[1:]  # Skip header
    
    first_dragon_teams = []
    for line in lines:
        parts = line.split(',')
        first_dragon = parts[2]  # third column is first_dragon
        result = parts[3]  # second column is result
        
        if first_dragon == 'yes':
            first_dragon_teams.append(result)
    
    if not first_dragon_teams:
        return 0
    
    wins = sum(1 for result in first_dragon_teams if result == 'win')
    return wins / len(first_dragon_teams)


def q2(text):
    """Return the average number of barons taken by winning teams."""
    lines = text.strip().split('\n')[1:]  # Skip header
    
    winning_barons = []
    for line in lines:
        parts = line.split(',')
        result = parts[2]  # third column is result
        barons = int(parts[5])  # sixth column is barons
        
        if result == 'win':
            winning_barons.append(barons)
    
    if not winning_barons:
        return 0
    
    return sum(winning_barons) / len(winning_barons)


def q3(text):
    """Return the total number of towers destroyed across all rows."""
    lines = text.strip().split('\n')[1:]  # Skip header
    
    total_towers = 0
    for line in lines:
        parts = line.split(',')
        towers = int(parts[6])  # seventh column is towers
        total_towers += towers
    
    return total_towers

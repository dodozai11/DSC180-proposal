```python
import re
from collections import Counter

def parse_data(text):
    """Parse the match data from text format."""
    players = []
    # Pattern to match: "Name of Team played Champion in position and result with kills kills, deaths deaths and assists assists."
    pattern = r"(\w+) of (.+?) played (.+?) in (\w+) and (\w+) with (\d+) kills, (\d+) deaths and (\d+) assists"
    
    for match in re.finditer(pattern, text):
        player = {
            'name': match.group(1),
            'team': match.group(2),
            'champion': match.group(3),
            'position': match.group(4),
            'result': match.group(5),
            'kills': int(match.group(6)),
            'deaths': int(match.group(7)),
            'assists': int(match.group(8))
        }
        players.append(player)
    
    return players


def q1(text):
    """Return the name of the player with the highest KDA."""
    players = parse_data(text)
    
    best_player = None
    best_kda = -1
    
    for player in players:
        kda = (player['kills'] + player['assists']) / max(player['deaths'], 1)
        if kda > best_kda:
            best_kda = kda
            best_player = player['name']
    
    return best_player


def q2(text):
    """Return the average number of kills in games that were won."""
    players = parse_data(text)
    
    won_games = [p for p in players if p['result'] == 'won']
    
    if not won_games:
        return 0
    
    total_kills = sum(p['kills'] for p in won_games)
    return total_kills / len(won_games)


def q3(text):
    """Return the name of the most-picked champion."""
    players = parse_data(text)
    
    champions = [p['champion'] for p in players]
    champion_counts = Counter(champions)
    
    most_common_champion = champion_counts.most_common(1)[0][0]
    return most_common_champion
```

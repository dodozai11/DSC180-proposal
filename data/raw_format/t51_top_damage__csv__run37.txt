def q1(text):
    """Returns the average damage per minute of players who lost."""
    lines = text.strip().split('\n')
    header = lines[0]
    
    total_damage = 0
    count = 0
    
    for line in lines[1:]:
        parts = line.split(',')
        result = parts[3]
        damage_per_min = float(parts[4])
        
        if result == 'loss':
            total_damage += damage_per_min
            count += 1
    
    if count == 0:
        return 0
    
    return total_damage / count


def q2(text):
    """Returns the name of the player who took the most damage per minute."""
    lines = text.strip().split('\n')
    
    max_damage = -1
    max_player = None
    
    for line in lines[1:]:
        parts = line.split(',')
        player = parts[0]
        damage_taken_per_min = float(parts[5])
        
        if damage_taken_per_min > max_damage:
            max_damage = damage_taken_per_min
            max_player = player
    
    return max_player


def q3(text):
    """Returns the number of different teams."""
    lines = text.strip().split('\n')
    
    teams = set()
    
    for line in lines[1:]:
        parts = line.split(',')
        team = parts[1]
        teams.add(team)
    
    return len(teams)

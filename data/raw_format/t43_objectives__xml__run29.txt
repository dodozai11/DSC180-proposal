import xml.etree.ElementTree as ET

def q1(text):
    """Return the win rate of teams that took the first dragon."""
    root = ET.fromstring(f"<root>{text}</root>")
    rows = root.findall('row')
    
    first_dragon_teams = [row for row in rows if row.find('first_dragon').text == 'yes']
    
    if not first_dragon_teams:
        return 0
    
    wins = sum(1 for row in first_dragon_teams if row.find('result').text == 'win')
    return wins / len(first_dragon_teams)


def q2(text):
    """Return the average number of barons taken by winning teams."""
    root = ET.fromstring(f"<root>{text}</root>")
    rows = root.findall('row')
    
    winning_teams = [row for row in rows if row.find('result').text == 'win']
    
    if not winning_teams:
        return 0
    
    total_barons = sum(int(row.find('barons').text) for row in winning_teams)
    return total_barons / len(winning_teams)


def q3(text):
    """Return the total number of towers destroyed across all rows."""
    root = ET.fromstring(f"<root>{text}</root>")
    rows = root.findall('row')
    
    total_towers = sum(int(row.find('towers').text) for row in rows)
    return total_towers

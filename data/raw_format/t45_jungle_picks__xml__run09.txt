import re
from collections import defaultdict

def q1(text):
    """Return the number of different champions played."""
    champions = set()
    for match in re.finditer(r'<champion>([^<]+)</champion>', text):
        champions.add(match.group(1))
    return len(champions)

def q2(text):
    """Return the champion with the best win rate among champions played at least twice."""
    champion_stats = defaultdict(lambda: {'wins': 0, 'total': 0})
    
    # Parse all rows
    for match in re.finditer(r'<champion>([^<]+)</champion>.*?<result>([^<]+)</result>', text):
        champion = match.group(1)
        result = match.group(2)
        champion_stats[champion]['total'] += 1
        if result == 'win':
            champion_stats[champion]['wins'] += 1
    
    # Filter champions played at least twice
    qualified_champions = {
        champ: stats for champ, stats in champion_stats.items() 
        if stats['total'] >= 2
    }
    
    if not qualified_champions:
        return None
    
    # Find champion with best win rate
    best_champion = max(
        qualified_champions.items(),
        key=lambda x: x[1]['wins'] / x[1]['total']
    )
    
    return best_champion[0]

def q3(text):
    """Return the average number of deaths in games that were lost."""
    losses_deaths = []
    
    # Parse rows with result and deaths
    for match in re.finditer(r'<result>([^<]+)</result>.*?<deaths>(\d+)</deaths>', text):
        result = match.group(1)
        deaths = int(match.group(2))
        if result == 'loss':
            losses_deaths.append(deaths)
    
    if not losses_deaths:
        return 0
    
    return sum(losses_deaths) / len(losses_deaths)

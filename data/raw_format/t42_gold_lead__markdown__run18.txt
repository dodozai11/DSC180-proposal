```python
import csv
from io import StringIO

def q1(text):
    """Return the average gold difference at 15 minutes of the winning teams."""
    reader = csv.DictReader(StringIO(text), delimiter='|')
    
    winning_gold_diffs = []
    for row in reader:
        row = {k.strip(): v.strip() for k, v in row.items()}
        if row['result'] == 'win':
            winning_gold_diffs.append(int(row['gold_diff_15']))
    
    return sum(winning_gold_diffs) / len(winning_gold_diffs) if winning_gold_diffs else 0


def q2(text):
    """Return how many teams won despite a negative gold difference at 15 minutes."""
    reader = csv.DictReader(StringIO(text), delimiter='|')
    
    count = 0
    for row in reader:
        row = {k.strip(): v.strip() for k, v in row.items()}
        if row['result'] == 'win' and int(row['gold_diff_15']) < 0:
            count += 1
    
    return count


def q3(text):
    """Return the win rate of teams that were ahead in gold at 15 minutes."""
    reader = csv.DictReader(StringIO(text), delimiter='|')
    
    ahead_in_gold = 0
    ahead_and_won = 0
    
    for row in reader:
        row = {k.strip(): v.strip() for k, v in row.items()}
        gold_diff = int(row['gold_diff_15'])
        
        if gold_diff > 0:
            ahead_in_gold += 1
            if row['result'] == 'win':
                ahead_and_won += 1
    
    return ahead_and_won / ahead_in_gold if ahead_in_gold > 0 else 0
```

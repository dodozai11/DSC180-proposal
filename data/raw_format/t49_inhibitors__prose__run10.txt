```python
import re

def q1(text):
    """Returns the average number of inhibitors destroyed by winning teams."""
    winning_inhibitors = []
    
    matches = re.finditer(r'In game (G\d+), (\w+(?:\s+\w+)*) won after ([\d.]+) minutes, destroying (\d+) inhibitors', text)
    
    for match in matches:
        inhibitors = int(match.group(4))
        winning_inhibitors.append(inhibitors)
    
    if winning_inhibitors:
        return sum(winning_inhibitors) / len(winning_inhibitors)
    return 0

def q2(text):
    """Returns the average damage to towers of losing teams."""
    losing_damages = []
    
    matches = re.finditer(r'In game (G\d+), (\w+(?:\s+\w+)*) lost after ([\d.]+) minutes, destroying \d+ inhibitors and dealing (\d+) damage to towers', text)
    
    for match in matches:
        damage = int(match.group(4))
        losing_damages.append(damage)
    
    if losing_damages:
        return sum(losing_damages) / len(losing_damages)
    return 0

def q3(text):
    """Returns the game label of the longest game."""
    games = {}
    
    matches = re.finditer(r'In game (G\d+), \w+(?:\s+\w+)* (?:won|lost) after ([\d.]+) minutes', text)
    
    for match in matches:
        game_label = match.group(1)
        duration = float(match.group(2))
        
        if game_label not in games:
            games[game_label] = duration
    
    if games:
        longest_game = max(games.items(), key=lambda x: x[1])
        return longest_game[0]
    return None
```

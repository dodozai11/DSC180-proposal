import json
import re

def q1(text):
    """Returns the win rate of teams that took the first dragon."""
    data = json.loads(text)
    
    first_dragon_teams = [row for row in data if row['first_dragon'] == 'yes']
    
    if not first_dragon_teams:
        return 0
    
    wins = sum(1 for row in first_dragon_teams if row['result'] == 'win')
    return wins / len(first_dragon_teams)


def q2(text):
    """Returns the average number of barons taken by winning teams."""
    data = json.loads(text)
    
    winning_teams = [row for row in data if row['result'] == 'win']
    
    if not winning_teams:
        return 0
    
    total_barons = sum(row['barons'] for row in winning_teams)
    return total_barons / len(winning_teams)


def q3(text):
    """Returns the total number of towers destroyed across all rows."""
    data = json.loads(text)
    
    return sum(row['towers'] for row in data)

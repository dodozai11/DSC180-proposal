def q1(text):
    """Returns the win rate of teams that took the first tower."""
    lines = text.strip().split('\n')
    
    # Skip header
    data = []
    for line in lines[1:]:
        parts = line.split('|')
        parts = [p.strip() for p in parts if p.strip()]
        if len(parts) == 5:
            data.append({
                'game': parts[0],
                'team': parts[1],
                'result': parts[2],
                'first_tower': parts[3],
                'void_grubs': int(parts[4]),
                'heralds': int(parts[5])
            })
    
    first_tower_teams = [row for row in data if row['first_tower'] == 'yes']
    
    if not first_tower_teams:
        return 0
    
    wins = sum(1 for row in first_tower_teams if row['result'] == 'win')
    return wins / len(first_tower_teams)


def q2(text):

                'void_grubs': int(parts[4]),
                'heralds': int(parts[5])
            })
    
    winning_teams = [row for row in data if row['result'] == 'win']
    
    if not winning_teams:
        return 0
    
    total_grubs = sum(row['void_grubs'] for row in winning_teams)
    return total_grubs / len(winning_teams)

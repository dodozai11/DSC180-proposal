import json
from collections import defaultdict

def q1(text):
    """Return the number of different champions played."""
    data = json.loads(text)
    champions = set(row["champion"] for row in data)
    return len(champions)


def q2(text):
    """Return the champion with the best win rate among champions played at least twice."""
    data = json.loads(text)
    
    champion_stats = defaultdict(lambda: {"wins": 0, "total": 0})
    
    for row in data:
        champion = row["champion"]
        champion_stats[champion]["total"] += 1
        if row["result"] == "win":
            champion_stats[champion]["wins"] += 1
    
    # Filter champions played at least twice
    eligible_champions = {
        champ: stats 
        for champ, stats in champion_stats.items() 
        if stats["total"] >= 2
    }
    
    if not eligible_champions:
        return None
    
    # Find champion with best win rate
    best_champion = max(
        eligible_champions.items(),
        key=lambda x: x[1]["wins"] / x[1]["total"]
    )
    
    return best_champion[0]


def q3(text):
    """Return the average number of deaths in games that were lost."""
    data = json.loads(text)
    
    lost_games = [row for row in data if row["result"] == "loss"]
    
    if not lost_games:
        return 0
    
    total_deaths = sum(row["deaths"] for row in lost_games)
    average_deaths = total_deaths / len(lost_games)
    
    return average_deaths

def q1(text):
    """Return the win rate of blue-side teams"""
    lines = text.strip().split('\n')
    # Skip header row
    data_lines = lines[1:]
    
    blue_games = 0
    blue_wins = 0
    
    for line in data_lines:
        parts = [p.strip() for p in line.split('|')[1:-1]]  # Skip empty first and last elements
        if len(parts) >= 3:
            side = parts[1]
            result = parts[2]
            
            if side == 'blue':
                blue_games += 1
                if result == 'win':
                    blue_wins += 1
    
    if blue_games == 0:
        return 0
    return blue_wins / blue_games


def q2(text):
    """Return the average length in minutes of games won by red-side team"""
    lines = text.strip().split('\n')
    # Skip header row
    data_lines = lines[1:]
    
    red_wins = []
    
    for line in data_lines:
        parts = [p.strip() for p in line.split('|')[1:-1]]  # Skip empty first and last elements
        if len(parts) >= 4:
            side = parts[1]
            result = parts[2]
            game_minutes = float(parts[3])
            
            if side == 'red' and result == 'win':
                red_wins.append(game_minutes)
    
    if len(red_wins) == 0:
        return 0
    return sum(red_wins) / len(red_wins)


def q3(text):
    """Return the name of the team with the most wins"""
    lines = text.strip().split('\n')
    # Skip header row
    data_lines = lines[1:]
    
    team_wins = {}
    
    for line in data_lines:
        parts = [p.strip() for p in line.split('|')[1:-1]]  # Skip empty first and last elements
        if len(parts) >= 3:
            team = parts[0]
            result = parts[2]
            
            if team not in team_wins:
                team_wins[team] = 0
            
            if result == 'win':
                team_wins[team] += 1
    
    if not team_wins:
        return None
    
    return max(team_wins, key=team_wins.get)

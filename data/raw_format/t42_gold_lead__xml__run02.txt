```python
import re
from collections import defaultdict

def parse_data(text):
    """Parse XML-like text and return list of dictionaries"""
    rows = []
    pattern = r'<row><game>(\w+)</game><team>(.+?)</team><side>(\w+)</side><result>(\w+)</result><gold_diff_15>(-?\d+)</gold_diff_15></row>'
    matches = re.findall(pattern, text)
    
    for match in matches:
        rows.append({
            'game': match[0],
            'team': match[1],
            'side': match[2],
            'result': match[3],
            'gold_diff_15': int(match[4])
        })
    
    return rows

def q1(text):
    """Return the average gold difference at 15 minutes of the winning teams"""
    rows = parse_data(text)
    winning_gold_diffs = [row['gold_diff_15'] for row in rows if row['result'] == 'win']
    
    if len(winning_gold_diffs) == 0:
        return 0
    
    return sum(winning_gold_diffs) / len(winning_gold_diffs)

def q2(text):
    """Return how many teams won despite a negative gold difference at 15 minutes"""
    rows = parse_data(text)
    count = sum(1 for row in rows if row['result'] == 'win' and row['gold_diff_15'] < 0)
    return count

def q3(text):
    """Return the win rate of teams that were ahead in gold at 15 minutes"""
    rows = parse_data(text)
    
    ahead_teams = [row for row in rows if row['gold_diff_15'] > 0]
    
    if len(ahead_teams) == 0:
        return 0
    
    wins = sum(1 for row in ahead_teams if row['result'] == 'win')
    win_rate = wins / len(ahead_teams)
    
    return win_rate
```

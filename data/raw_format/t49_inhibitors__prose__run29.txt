import re

def q1(text):
    """Return the average number of inhibitors destroyed by winning teams."""
    winning_inhibitors = []
    
    # Find all matches with the pattern: team won after X minutes, destroying Y inhibitors
    matches = re.findall(r'(\w+(?:\s+\w+)*)\s+won\s+after\s+([\d.]+)\s+minutes,\s+destroying\s+(\d+)\s+inhibitors', text)
    
    for team, duration, inhibitors in matches:
        winning_inhibitors.append(int(inhibitors))
    
    if winning_inhibitors:
        return sum(winning_inhibitors) / len(winning_inhibitors)
    return 0

def q2(text):
    """Return the average damage to towers of losing teams."""
    losing_damage = []
    
    # Find all matches with the pattern: team lost after X minutes, destroying Y inhibitors and dealing Z damage to towers
    matches = re.findall(r'(\w+(?:\s+\w+)*)\s+lost\s+after\s+([\d.]+)\s+minutes,\s+destroying\s+(\d+)\s+inhibitors\s+and\s+dealing\s+(\d+)\s+damage\s+to\s+towers', text)
    
    for team, duration, inhibitors, damage in matches:
        losing_damage.append(int(damage))
    
    if losing_damage:
        return sum(losing_damage) / len(losing_damage)
    return 0

def q3(text):
    """Return the game label of the longest game."""
    games = {}
    
    # Find all matches with game label and duration
    matches = re.findall(r'In game (G\d+),\s+(\w+(?:\s+\w+)*)\s+(?:won|lost)\s+after\s+([\d.]+)\s+minutes', text)
    
    for game_label, team, duration in matches:
        duration_float = float(duration)
        if game_label not in games:
            games[game_label] = duration_float
    
    if games:
        longest_game = max(games.items(), key=lambda x: x[1])
        return longest_game[0]
    return None

import re
from collections import defaultdict

def q1(text):
    """Returns the number of different champions played."""
    champions = set()
    matches = re.findall(r'played (\w+(?:\'?\w+)?)', text)
    champions.update(matches)
    return len(champions)


def q2(text):
    """Returns the champion with the best win rate among champions played at least twice."""
    champion_stats = defaultdict(lambda: {'wins': 0, 'total': 0})
    
    # Parse each line
    lines = text.split('. ')
    for line in lines:
        line = line.strip()
        if not line:
            continue
        
        # Extract champion name
        champion_match = re.search(r'played (\w+(?:\'?\w+)?)', line)
        if not champion_match:
            continue
        champion = champion_match.group(1)
        
        # Extract win/loss
        won = 'won' in line
        
        champion_stats[champion]['total'] += 1
        if won:
            champion_stats[champion]['wins'] += 1
    
    # Find champion with best win rate among those played at least twice
    best_champion = None
    best_win_rate = -1
    
    for champion, stats in champion_stats.items():
        if stats['total'] >= 2:
            win_rate = stats['wins'] / stats['total']
            if win_rate > best_win_rate:
                best_win_rate = win_rate
                best_champion = champion
    
    return best_champion


def q3(text):
    """Returns the average number of deaths in games that were lost."""
    deaths_in_losses = []
    
    # Parse each line
    lines = text.split('. ')
    for line in lines:
        line = line.strip()
        if not line:
            continue
        
        # Check if game was lost
        if 'lost' not in line:
            continue
        
        # Extract number of deaths
        deaths_match = re.search(r'dying (\d+)', line)
        if deaths_match:
            deaths = int(deaths_match.group(1))
            deaths_in_losses.append(deaths)
    
    if not deaths_in_losses:
        return 0
    
    return sum(deaths_in_losses) / len(deaths_in_losses)

import json
import re

def q1(text):
    """Return the average vision score of winners minus average vision score of losers."""
    data = json.loads(text)
    
    winners = [player['vision_score'] for player in data if player['result'] == 'win']
    losers = [player['vision_score'] for player in data if player['result'] == 'loss']
    
    avg_winners = sum(winners) / len(winners) if winners else 0
    avg_losers = sum(losers) / len(losers) if losers else 0
    
    return avg_winners - avg_losers

def q2(text):
    """Return the name of the player who placed the most wards."""
    data = json.loads(text)
    
    player_with_most_wards = max(data, key=lambda p: p['wards_placed'])
    return player_with_most_wards['player']

def q3(text):
    """Return how many players bought at least 15 control wards."""
    data = json.loads(text)
    
    count = sum(1 for player in data if player['control_wards'] >= 15)
    return count

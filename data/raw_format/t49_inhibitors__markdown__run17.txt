def q1(text):
    """Return the average number of inhibitors destroyed by winning teams."""
    lines = text.strip().split('\n')
    header = lines[0]
    
    winning_inhibitors = []
    for line in lines[1:]:
        parts = line.split('|')[1:-1]  # Remove first and last empty parts
        parts = [p.strip() for p in parts]
        
        if len(parts) >= 3:
            result = parts[2]
            inhibitors = float(parts[4])
            
            if result == 'win':
                winning_inhibitors.append(inhibitors)
    
    if winning_inhibitors:
        return sum(winning_inhibitors) / len(winning_inhibitors)
    return 0


def q2(text):
    """Return the average damage to towers of losing teams."""
    lines = text.strip().split('\n')
    
    losing_damages = []
    for line in lines[1:]:
        parts = line.split('|')[1:-1]  # Remove first and last empty parts
        parts = [p.strip() for p in parts]
        
        if len(parts) >= 5:
            result = parts[2]
            tower_damage = float(parts[5])
            
            if result == 'loss':
                losing_damages.append(tower_damage)
    
    if losing_damages:
        return sum(losing_damages) / len(losing_damages)
    return 0


def q3(text):
    """Return the game label of the longest game."""
    lines = text.strip().split('\n')
    
    max_duration = -1
    longest_game = None
    
    for line in lines[1:]:
        parts = line.split('|')[1:-1]  # Remove first and last empty parts
        parts = [p.strip() for p in parts]
        
        if len(parts) >= 4:
            game = parts[0]
            game_minutes = float(parts[3])
            
            if game_minutes > max_duration:
                max_duration = game_minutes
                longest_game = game
    
    return longest_game

def q1(text):
    """Returns the win rate of players with a positive CS difference at 10 minutes."""
    lines = text.strip().split('\n')
    
    # Skip header row
    rows = lines[2:]
    
    positive_cs_wins = 0
    positive_cs_total = 0
    
    for row in rows:
        if not row.strip():
            continue
        
        parts = [p.strip() for p in row.split('|') if p.strip()]
        
        player = parts[0]
        champion = parts[1]
        result = parts[2]
        cs_diff_10 = int(parts[3])
        
        if cs_diff_10 > 0:
            positive_cs_total += 1
            if result == 'win':
                positive_cs_wins += 1
    
    if positive_cs_total == 0:
        return 0
    
    return positive_cs_wins / positive_cs_total


def q2(text):
    """Returns the total number of kills at 10 minutes across all rows."""
    lines = text.strip().split('\n')
    
    # Skip header row
    rows = lines[2:]
    
    total_kills = 0
    
    for row in rows:
        if not row.strip():
            continue
        
        parts = [p.strip() for p in row.split('|') if p.strip()]
        
        kills_at_10 = int(parts[4])
        total_kills += kills_at_10
    
    return total_kills


def q3(text):
    """Returns the name of the champion with the most assists at 10 minutes in total."""
    lines = text.strip().split('\n')
    
    # Skip header row
    rows = lines[2:]
    
    champion_assists = {}
    
    for row in rows:
        if not row.strip():
            continue
        
        parts = [p.strip() for p in row.split('|') if p.strip()]
        
        champion = parts[1]
        assists_at_10 = int(parts[5])
        
        if champion not in champion_assists:
            champion_assists[champion] = 0
        
        champion_assists[champion] += assists_at_10
    
    # Find champion with most assists
    max_champion = max(champion_assists, key=champion_assists.get)
    
    return max_champion

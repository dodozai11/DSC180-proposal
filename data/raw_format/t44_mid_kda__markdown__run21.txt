def q1(text):
    """Return the name of the player with the highest KDA."""
    lines = text.strip().split('\n')
    # Skip header row
    data = []
    for line in lines[1:]:
        parts = line.split('|')
        # Remove empty strings from split (due to leading/trailing |)
        parts = [p.strip() for p in parts if p.strip()]
        player = parts[0]
        kills = int(parts[3])
        deaths = int(parts[4])
        assists = int(parts[5])
        
        kda = (kills + assists) / max(deaths, 1)
        data.append((player, kda))
    
    # Return player with highest KDA
    return max(data, key=lambda x: x[1])[0]


def q2(text):
    """Return the average number of kills in games that were won."""
    lines = text.strip().split('\n')
    # Skip header row
    kills_in_wins = []
    for line in lines[1:]:
        parts = line.split('|')
        # Remove empty strings from split (due to leading/trailing |)
        parts = [p.strip() for p in parts if p.strip()]
        kills = int(parts[3])
        result = parts[6]
        
        if result == 'win':
            kills_in_wins.append(kills)
    
    # Return average
    if kills_in_wins:
        return sum(kills_in_wins) / len(kills_in_wins)
    return 0


def q3(text):
    """Return the name of the most-picked champion."""
    lines = text.strip().split('\n')
    # Skip header row
    champion_counts = {}
    for line in lines[1:]:
        parts = line.split('|')
        # Remove empty strings from split (due to leading/trailing |)
        parts = [p.strip() for p in parts if p.strip()]
        champion = parts[2]
        
        champion_counts[champion] = champion_counts.get(champion, 0) + 1
    
    # Return champion with highest count
    return max(champion_counts, key=champion_counts.get)

```python
import re

def q1(text):
    """Returns the average number of inhibitors destroyed by winning teams."""
    winning_inhibitors = []
    
    # Find all matches for winning teams
    pattern = r'In game (G\d+), (.+?) won after ([\d.]+) minutes, destroying (\d+) inhibitors'
    matches = re.findall(pattern, text)
    
    for match in matches:
        inhibitors = int(match[3])
        winning_inhibitors.append(inhibitors)
    
    if winning_inhibitors:
        return sum(winning_inhibitors) / len(winning_inhibitors)
    return 0

def q2(text):
    """Returns the average damage to towers of losing teams."""
    losing_damages = []
    
    # Find all matches for losing teams
    pattern = r'In game (G\d+), (.+?) lost after ([\d.]+) minutes, destroying \d+ inhibitors and dealing (\d+) damage to towers'
    matches = re.findall(pattern, text)
    
    for match in matches:
        damage = int(match[3])
        losing_damages.append(damage)
    
    if losing_damages:
        return sum(losing_damages) / len(losing_damages)
    return 0

def q3(text):
    """Returns the game label of the longest game."""
    longest_game = None
    longest_duration = -1
    
    # Find all games with their durations
    pattern = r'In game (G\d+), .+ (won|lost) after ([\d.]+) minutes'
    matches = re.findall(pattern, text)
    
    # Create a set of game IDs we've already processed to avoid duplicates
    processed_games = set()
    
    for match in matches:
        game_id = match[0]
        duration = float(match[2])
        
        # Only process each game once (since each game appears twice in the data)
        if game_id not in processed_games:
            processed_games.add(game_id)
            if duration > longest_duration:
                longest_duration = duration
                longest_game = game_id
    
    return longest_game
```

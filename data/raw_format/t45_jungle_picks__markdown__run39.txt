def q1(text):
    """Returns the number of different champions played."""
    lines = text.strip().split('\n')
    champions = set()
    
    # Skip header row
    for line in lines[1:]:
        if line.strip():
            parts = line.split('|')
            champion = parts[2].strip()
            champions.add(champion)
    
    return len(champions)


def q2(text):
    """Returns the champion with the best win rate among champions played at least twice."""
    lines = text.strip().split('\n')
    champion_stats = {}
    
    # Skip header row
    for line in lines[1:]:
        if line.strip():
            parts = line.split('|')
            champion = parts[2].strip()
            result = parts[3].strip()
            
            if champion not in champion_stats:
                champion_stats[champion] = {'wins': 0, 'total': 0}
            
            champion_stats[champion]['total'] += 1
            if result == 'win':
                champion_stats[champion]['wins'] += 1
    
    # Filter champions played at least twice
    candidates = {
        champ: stats['wins'] / stats['total']
        for champ, stats in champion_stats.items()
        if stats['total'] >= 2
    }
    
    if not candidates:
        return None
    
    return max(candidates, key=candidates.get)


def q3(text):
    """Returns the average number of deaths in games that were lost."""
    lines = text.strip().split('\n')
    losses = []
    
    # Skip header row
    for line in lines[1:]:
        if line.strip():
            parts = line.split('|')
            result = parts[3].strip()
            deaths = int(parts[4].strip())
            
            if result == 'loss':
                losses.append(deaths)
    
    if not losses:
        return 0
    
    return sum(losses) / len(losses)

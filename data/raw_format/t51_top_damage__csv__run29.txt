def q1(text):
    """Return the average damage per minute of players who lost."""
    lines = text.strip().split('\n')
    header = lines[0].split(',')
    
    damage_per_min_idx = header.index('damage_per_min')
    result_idx = header.index('result')
    
    total_damage = 0
    count = 0
    
    for line in lines[1:]:
        fields = line.split(',')
        if fields[result_idx] == 'loss':
            total_damage += float(fields[damage_per_min_idx])
            count += 1
    
    return total_damage / count if count > 0 else 0


def q2(text):
    """Return the name of the player who took the most damage per minute."""
    lines = text.strip().split('\n')
    header = lines[0].split(',')
    
    player_idx = header.index('player')
    damage_taken_per_min_idx = header.index('damage_taken_per_min')
    
    max_damage = -1
    max_player = None
    
    for line in lines[1:]:
        fields = line.split(',')
        damage_taken = float(fields[damage_taken_per_min_idx])
        if damage_taken > max_damage:
            max_damage = damage_taken
            max_player = fields[player_idx]
    
    return max_player


def q3(text):
    """Return the number of different teams."""
    lines = text.strip().split('\n')
    header = lines[0].split(',')
    
    team_idx = header.index('team')
    
    teams = set()
    for line in lines[1:]:
        fields = line.split(',')
        teams.add(fields[team_idx])
    
    return len(teams)

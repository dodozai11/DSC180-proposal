import re

def q1(text):
    """Returns the average number of inhibitors destroyed by winning teams."""
    winning_inhibitors = []
    
    # Find all matches for winning teams
    pattern = r'In game (\w+), (\w+(?:\s+\w+)*) won after ([\d.]+) minutes, destroying (\d+) inhibitors'
    matches = re.findall(pattern, text)
    
    for game, team, duration, inhibitors in matches:
        winning_inhibitors.append(int(inhibitors))
    
    if winning_inhibitors:
        return sum(winning_inhibitors) / len(winning_inhibitors)
    return 0


def q2(text):
    """Returns the average damage to towers of losing teams."""
    losing_damages = []
    
    # Find all matches for losing teams
    pattern = r'In game (\w+), (\w+(?:\s+\w+)*) lost after ([\d.]+) minutes, destroying (\d+) inhibitors and dealing (\d+) damage to towers'
    matches = re.findall(pattern, text)
    
    for game, team, duration, inhibitors, damage in matches:
        losing_damages.append(int(damage))
    
    if losing_damages:
        return sum(losing_damages) / len(losing_damages)
    return 0


def q3(text):
    """Returns the game label of the longest game."""
    longest_game = None
    longest_duration = -1
    
    # Find all game entries (both winning and losing)
    pattern = r'In game (\w+), (\w+(?:\s+\w+)*) (won|lost) after ([\d.]+) minutes'
    matches = re.findall(pattern, text)
    
    for game, team, result, duration in matches:
        duration_float = float(duration)
        if duration_float > longest_duration:
            longest_duration = duration_float
            longest_game = game
    
    return longest_game

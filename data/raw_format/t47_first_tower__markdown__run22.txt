def q1(text):
    """Return the win rate of teams that took the first tower."""
    lines = text.strip().split('\n')
    # Skip header row
    data = []
    for line in lines[1:]:
        parts = line.split('|')
        if len(parts) >= 6:
            game = parts[1].strip()
            team = parts[2].strip()
            result = parts[3].strip()
            first_tower = parts[4].strip()
            void_grubs = parts[5].strip()
            heralds = parts[6].strip()
            data.append({
                'game': game,
                'team': team,
                'result': result,
                'first_tower': first_tower,
                'void_grubs': void_grubs,
                'heralds': heralds
            })
    
    first_tower_teams = [row for row in data if row['first_tower'] == 'yes']
    if len(first_tower_teams) == 0:
        return 0
    
    wins = sum(1 for row in first_tower_teams if row['result'] == 'win')
    return wins / len(first_tower_teams)


def q2(text):
    """Return the average number of void grubs taken by winning teams."""
    lines = text.strip().split('\n')
    # Skip header row
    data = []
    for line in lines[1:]:
        parts = line.split('|')
        if len(parts) >= 6:
            game = parts[1].strip()
            team = parts[2].strip()
            result = parts[3].strip()
            first_tower = parts[4].strip()
            void_grubs = parts[5].strip()
            heralds = parts[6].strip()
            data.append({
                'game': game,
                'team': team,
                'result': result,
                'first_tower': first_tower,
                'void_grubs': int(void_grubs),
                'heralds': int(heralds)
            })
    
    winning_teams = [row for row in data if row['result'] == 'win']
    if len(winning_teams) == 0:
        return 0
    
    total_grubs = sum(row['void_grubs'] for row in winning_teams)
    return total_grubs / len(winning_teams)

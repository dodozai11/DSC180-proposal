import re

def q1(text):
    """Return the win rate of teams that took the first tower."""
    first_tower_teams = []
    
    # Parse all game data
    pattern = r"In game (G\d+), (.+?) (won|lost); they (took|did not take) the first tower"
    matches = re.findall(pattern, text)
    
    for match in matches:
        game_id, team_name, result, tower_status = match
        took_first_tower = tower_status == "took"
        won = result == "won"
        
        if took_first_tower:
            first_tower_teams.append(won)
    
    if len(first_tower_teams) == 0:
        return 0
    
    win_rate = sum(first_tower_teams) / len(first_tower_teams)
    return win_rate


def q2(text):
    """Return the average number of void grubs taken by winning teams."""
    winning_teams_grubs = []
    
    # Parse all game data with void grubs
    pattern = r"In game (G\d+), (.+?) (won|lost);.+?finished with (\d+) void grubs"
    matches = re.findall(pattern, text)
    
    for match in matches:
        game_id, team_name, result, void_grubs = match
        won = result == "won"
        void_grubs_count = int(void_grubs)
        
        if won:
            winning_teams_grubs.append(void_grubs_count)
    
    if len(winning_teams_grubs) == 0:
        return 0
    
    average_grubs = sum(winning_teams_grubs) / len(winning_teams_grubs)
    return average_grubs


def q3(text):
    """Return how many teams took at least one herald and still lost."""
    teams_with_herald_lost = 0
    
    # Parse all game data with heralds
    pattern = r"In game (G\d+), (.+?) (won|lost);.+?finished with \d+ void grubs and (\d+) heralds"
    matches = re.findall(pattern, text)
    
    for match in matches:
        game_id, team_name, result, heralds = match
        lost = result == "lost"
        heralds_count = int(heralds)
        
        if lost and heralds_count >= 1:
            teams_with_herald_lost += 1
    
    return teams_with_herald_lost

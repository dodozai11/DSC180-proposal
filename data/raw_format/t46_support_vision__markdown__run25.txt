def q1(text):
    """Return the average vision score of winners minus average vision score of losers."""
    lines = text.strip().split('\n')
    # Skip header line
    data = lines[1:]
    
    winners_vision = []
    losers_vision = []
    
    for line in data:
        parts = [p.strip() for p in line.split('|')[1:-1]]  # Remove empty strings from split
        if len(parts) >= 3:
            result = parts[2]
            vision_score = float(parts[3])
            
            if result == 'win':
                winners_vision.append(vision_score)
            elif result == 'loss':
                losers_vision.append(vision_score)
    
    avg_winners = sum(winners_vision) / len(winners_vision) if winners_vision else 0
    avg_losers = sum(losers_vision) / len(losers_vision) if losers_vision else 0
    
    return avg_winners - avg_losers


def q2(text):
    """Return the name of the player who placed the most wards."""
    lines = text.strip().split('\n')
    # Skip header line
    data = lines[1:]
    
    max_wards = -1
    max_player = None
    
    for line in data:
        parts = [p.strip() for p in line.split('|')[1:-1]]  # Remove empty strings from split
        if len(parts) >= 5:
            player = parts[0]
            wards_placed = int(parts[4])
            
            if wards_placed > max_wards:
                max_wards = wards_placed
                max_player = player
    
    return max_player


def q3(text):
    """Return how many players bought at least 15 control wards."""
    lines = text.strip().split('\n')
    # Skip header line
    data = lines[1:]
    
    count = 0
    
    for line in data:
        parts = [p.strip() for p in line.split('|')[1:-1]]  # Remove empty strings from split
        if len(parts) >= 6:
            control_wards = int(parts[5])
            
            if control_wards >= 15:
                count += 1
    
    return count

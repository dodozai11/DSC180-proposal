```python
import re

def q1(text):
    """Returns the average number of inhibitors destroyed by winning teams."""
    winning_inhibitors = []
    
    # Find all matches for winning teams
    pattern = r'In game (G\d+), (\w+(?:\s+\w+)*) won after ([\d.]+) minutes, destroying (\d+) inhibitors'
    matches = re.findall(pattern, text)
    
    for match in matches:
        inhibitors = int(match[3])
        winning_inhibitors.append(inhibitors)
    
    if not winning_inhibitors:
        return 0
    
    return sum(winning_inhibitors) / len(winning_inhibitors)


def q2(text):
    """Returns the average damage to towers of losing teams."""
    losing_damage = []
    
    # Find all matches for losing teams
    pattern = r'In game (G\d+), (\w+(?:\s+\w+)*) lost after ([\d.]+) minutes, destroying \d+ inhibitors and dealing (\d+) damage to towers'
    matches = re.findall(pattern, text)
    
    for match in matches:
        damage = int(match[3])
        losing_damage.append(damage)
    
    if not losing_damage:
        return 0
    
    return sum(losing_damage) / len(losing_damage)


def q3(text):
    """Returns the game label of the longest game."""
    max_duration = -1
    longest_game = None
    
    # Find all games with their durations
    pattern = r'In game (G\d+), \w+(?:\s+\w+)* (?:won|lost) after ([\d.]+) minutes'
    matches = re.findall(pattern, text)
    
    for match in matches:
        game_label = match[0]
        duration = float(match[1])
        
        if duration > max_duration:
            max_duration = duration
            longest_game = game_label
    
    return longest_game
```

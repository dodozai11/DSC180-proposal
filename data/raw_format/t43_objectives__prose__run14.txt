import re

def q1(text):
    """Return the win rate of teams that took the first dragon."""
    first_dragon_teams = []
    
    # Parse all games
    game_pattern = r'In game (\w+), (.+?) (won|lost); they (took|did not take) the first dragon and finished with (\d+) dragons, (\d+) barons and (\d+) towers\.'
    matches = re.findall(game_pattern, text)
    
    for match in matches:
        game_id, team_name, result, first_dragon, dragons, barons, towers = match
        took_first_dragon = first_dragon == "took"
        won = result == "won"
        
        if took_first_dragon:
            first_dragon_teams.append(won)
    
    if len(first_dragon_teams) == 0:
        return 0
    
    wins = sum(first_dragon_teams)
    win_rate = wins / len(first_dragon_teams)
    return win_rate


def q2(text):
    """Return the average number of barons taken by winning teams."""
    winning_barons = []
    
    # Parse all games
    game_pattern = r'In game (\w+), (.+?) (won|lost); they (took|did not take) the first dragon and finished with (\d+) dragons, (\d+) barons and (\d+) towers\.'
    matches = re.findall(game_pattern, text)
    
    for match in matches:
        game_id, team_name, result, first_dragon, dragons, barons, towers = match
        won = result == "won"
        barons_count = int(barons)
        
        if won:
            winning_barons.append(barons_count)
    
    if len(winning_barons) == 0:
        return 0
    
    average_barons = sum(winning_barons) / len(winning_barons)
    return average_barons

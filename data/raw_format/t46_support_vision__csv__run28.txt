def q1(text):
    lines = text.strip().split('\n')
    header = lines[0].split(',')
    
    vision_idx = header.index('vision_score')
    result_idx = header.index('result')
    
    wins_vision = []
    loss_vision = []
    
    for line in lines[1:]:
        parts = line.split(',')
        result = parts[result_idx]
        vision = int(parts[vision_idx])
        
        if result == 'win':
            wins_vision.append(vision)
        else:
            loss_vision.append(vision)
    
    avg_win = sum(wins_vision) / len(wins_vision) if wins_vision else 0
    avg_loss = sum(loss_vision) / len(loss_vision) if loss_vision else 0
    
    return avg_win - avg_loss


def q2(text):
    lines = text.strip().split('\n')
    header = lines[0].split(',')
    
    player_idx = header.index('player')
    wards_idx = header.index('wards_placed')
    
    max_wards = -1
    max_player = None
    
    for line in lines[1:]:
        parts = line.split(',')
        player = parts[player_idx]
        wards = int(parts[wards_idx])
        
        if wards > max_wards:
            max_wards = wards
            max_player = player
    
    return max_player


def q3(text):
    lines = text.strip().split('\n')
    header = lines[0].split(',')
    
    control_wards_idx = header.index('control_wards')
    
    count = 0
    for line in lines[1:]:
        parts = line.split(',')
        control_wards = int(parts[control_wards_idx])
        
        if control_wards >= 15:
            count += 1
    
    return count

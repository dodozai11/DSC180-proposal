def q1(text):
    """Return the win rate of players with a positive CS difference at 10 minutes."""
    lines = text.strip().split('\n')
    # Skip header row
    rows = lines[2:]
    
    positive_cs_players = []
    for row in rows:
        parts = [p.strip() for p in row.split('|') if p.strip()]
        if len(parts) >= 5:
            player, champion, result, cs_diff_10, kills_at_10 = parts[0], parts[1], parts[2], parts[3], parts[4]
            try:
                cs_diff = int(cs_diff_10)
                if cs_diff > 0:
                    positive_cs_players.append(result)
            except ValueError:
                pass
    
    if not positive_cs_players:
        return 0
    
    wins = sum(1 for result in positive_cs_players if result.lower() == 'win')
    return wins / len(positive_cs_players)


def q2(text):
    """Return the total number of kills at 10 minutes across all rows."""
    lines = text.strip().split('\n')
    # Skip header row
    rows = lines[2:]
    
    total_kills = 0
    for row in rows:
        parts = [p.strip() for p in row.split('|') if p.strip()]
        if len(parts) >= 5:
            kills_at_10 = parts[4]
            try:
                total_kills += int(kills_at_10)
            except ValueError:
                pass
    
    return total_kills


def q3(text):
    """Return the name of the champion with the most assists at 10 minutes in total."""
    lines = text.strip().split('\n')
    # Skip header row
    rows = lines[2:]
    
    champion_assists = {}
    for row in rows:
        parts = [p.strip() for p in row.split('|') if p.strip()]
        if len(parts) >= 6:
            champion = parts[1]
            assists_at_10 = parts[5]
            try:
                assists = int(assists_at_10)
                if champion not in champion_assists:
                    champion_assists[champion] = 0
                champion_assists[champion] += assists
            except ValueError:
                pass
    
    if not champion_assists:
        return None
    
    return max(champion_assists, key=champion_assists.get)

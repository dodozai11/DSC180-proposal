def q1(text):
    lines = text.strip().split('\n')
    header = lines[0]
    rows = lines[1:]
    
    positive_cs_diff_rows = []
    for row in rows:
        parts = row.split(',')
        cs_diff_10 = int(parts[3])
        if cs_diff_10 > 0:
            positive_cs_diff_rows.append(parts)
    
    if len(positive_cs_diff_rows) == 0:
        return 0
    
    wins = sum(1 for row in positive_cs_diff_rows if row[2] == 'win')
    return wins / len(positive_cs_diff_rows)


def q2(text):
    lines = text.strip().split('\n')
    rows = lines[1:]
    
    total_kills = 0
    for row in rows:
        parts = row.split(',')
        kills_at_10 = int(parts[4])
        total_kills += kills_at_10
    
    return total_kills


def q3(text):
    lines = text.strip().split('\n')
    rows = lines[1:]
    
    champion_assists = {}
    for row in rows:
        parts = row.split(',')
        champion = parts[1]
        assists_at_10 = int(parts[5])
        
        if champion not in champion_assists:
            champion_assists[champion] = 0
        champion_assists[champion] += assists_at_10
    
    return max(champion_assists, key=champion_assists.get)

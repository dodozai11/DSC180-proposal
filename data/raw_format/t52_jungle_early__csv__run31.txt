def q1(text):
    lines = text.strip().split('\n')
    header = lines[0].split(',')
    
    cs_diff_10_idx = header.index('cs_diff_10')
    result_idx = header.index('result')
    
    positive_cs_diff = []
    for line in lines[1:]:
        row = line.split(',')
        cs_diff = int(row[cs_diff_10_idx])
        result = row[result_idx]
        
        if cs_diff > 0:
            positive_cs_diff.append(result)
    
    if len(positive_cs_diff) == 0:
        return 0
    
    wins = sum(1 for result in positive_cs_diff if result == 'win')
    return wins / len(positive_cs_diff)


def q2(text):
    lines = text.strip().split('\n')
    header = lines[0].split(',')
    
    kills_at_10_idx = header.index('kills_at_10')
    
    total_kills = 0
    for line in lines[1:]:
        row = line.split(',')
        kills = int(row[kills_at_10_idx])
        total_kills += kills
    
    return total_kills


def q3(text):
    lines = text.strip().split('\n')
    header = lines[0].split(',')
    
    champion_idx = header.index('champion')
    assists_at_10_idx = header.index('assists_at_10')
    
    champion_assists = {}
    for line in lines[1:]:
        row = line.split(',')
        champion = row[champion_idx]
        assists = int(row[assists_at_10_idx])
        
        if champion not in champion_assists:
            champion_assists[champion] = 0
        champion_assists[champion] += assists
    
    return max(champion_assists, key=champion_assists.get)

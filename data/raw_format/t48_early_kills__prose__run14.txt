```python
import re

def q1(text):
    """Returns the average number of kills at 15 minutes of red-side teams."""
    red_kills = []
    
    # Find all matches for red-side teams
    pattern = r'In game \w+, (.+?) played on the red side and \w+, with (\d+) kills'
    matches = re.findall(pattern, text)
    
    for match in matches:
        kills = int(match[1])
        red_kills.append(kills)
    
    if len(red_kills) == 0:
        return 0
    
    return sum(red_kills) / len(red_kills)


def q2(text):
    """Returns the name of the team with the largest experience difference at 15 minutes."""
    teams_exp = []
    
    # Find all team records with experience differences
    pattern = r'In game \w+, (.+?) played on the \w+ side and \w+, with \d+ kills and an experience difference of (-?\d+) at 15 minutes'
    matches = re.findall(pattern, text)
    
    for match in matches:
        team_name = match[0]
        exp_diff = int(match[1])
        teams_exp.append((team_name, exp_diff))
    
    if len(teams_exp) == 0:
        return None
    
    # Find the team with the largest (most positive) experience difference
    max_team = max(teams_exp, key=lambda x: x[1])
    return max_team[0]


def q3(text):
    """Returns how many teams won despite a negative experience difference at 15 minutes."""
    count = 0
    
    # Find all winning teams with their experience differences
    pattern = r'In game \w+, (.+?) played on the \w+ side and won, with \d+ kills and an experience difference of (-?\d+) at 15 minutes'
    matches = re.findall(pattern, text)
    
    for match in matches:
        exp_diff = int(match[1])
        if exp_diff < 0:
            count += 1
    
    return count
```

```python
import re

def q1(text):
    """Returns the average vision score of winners minus average vision score of losers."""
    players = parse_players(text)
    
    winners = [p for p in players if p['result'] == 'won']
    losers = [p for p in players if p['result'] == 'lost']
    
    avg_winners = sum(p['vision_score'] for p in winners) / len(winners) if winners else 0
    avg_losers = sum(p['vision_score'] for p in losers) / len(losers) if losers else 0
    
    return avg_winners - avg_losers


def q2(text):
    """Returns the name of the player who placed the most wards."""
    players = parse_players(text)
    
    player_with_most_wards = max(players, key=lambda p: p['wards_placed'])
    return player_with_most_wards['name']


def q3(text):
    """Returns how many players bought at least 15 control wards."""
    players = parse_players(text)
    
    count = sum(1 for p in players if p['control_wards_bought'] >= 15)
    return count


def parse_players(text):
    """Parses the text and returns a list of player dictionaries."""
    players = []
    
    # Pattern to match each player entry
    # Format: "Name of Team (won|lost) with a vision score of X, Y wards placed and Z control wards bought."
    pattern = r'(\w+) of (.+?) (won|lost) with a vision score of (\d+), (\d+) wards placed and (\d+) control wards bought'
    
    matches = re.finditer(pattern, text)
    
    for match in matches:
        player = {
            'name': match.group(1),
            'team': match.group(2),
            'result': match.group(3),
            'vision_score': int(match.group(4)),
            'wards_placed': int(match.group(5)),
            'control_wards_bought': int(match.group(6))
        }
        players.append(player)
    
    return players
```

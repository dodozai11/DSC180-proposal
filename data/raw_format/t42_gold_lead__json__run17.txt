```python
import json
import re


def parse_data(text):
    """Parse the JSON data from the text string."""
    # Extract JSON array from the text
    json_match = re.search(r'\[.*\]', text, re.DOTALL)
    if json_match:
        data = json.loads(json_match.group())
        return data
    return []


def q1(text):
    """Return the average gold difference at 15 minutes of the winning teams."""
    data = parse_data(text)
    winning_teams = [row['gold_diff_15'] for row in data if row['result'] == 'win']
    
    if not winning_teams:
        return 0
    
    return sum(winning_teams) / len(winning_teams)


def q2(text):
    """Return how many teams won despite a negative gold difference at 15 minutes."""
    data = parse_data(text)
    count = sum(1 for row in data if row['result'] == 'win' and row['gold_diff_15'] < 0)
    return count


def q3(text):
    """Return the win rate of teams that were ahead in gold at 15 minutes."""
    data = parse_data(text)
    
    # Filter teams that were ahead in gold at 15 minutes (positive gold_diff_15)
    ahead_teams = [row for row in data if row['gold_diff_15'] > 0]
    
    if not ahead_teams:
        return 0
    
    # Count how many of these teams won
    wins = sum(1 for row in ahead_teams if row['result'] == 'win')
    
    # Calculate win rate
    win_rate = wins / len(ahead_teams)
    return win_rate
```

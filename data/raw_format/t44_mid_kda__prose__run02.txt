```python
import re
from collections import Counter

def parse_data(text):
    """Parse the League of Legends match data from text."""
    pattern = r'(\w+) of (.+?) played (\w+\'?\w*) in mid and (won|lost) with (\d+) kills, (\d+) deaths and (\d+) assists'
    matches = re.findall(pattern, text)
    
    players = []
    for match in matches:
        player_name, team, champion, result, kills, deaths, assists = match
        players.append({
            'name': player_name,
            'team': team,
            'champion': champion,
            'won': result == 'won',
            'kills': int(kills),
            'deaths': int(deaths),
            'assists': int(assists)
        })
    return players

def q1(text):
    """Return the name of the player with the highest KDA."""
    players = parse_data(text)
    
    best_player = None
    best_kda = -1
    
    for player in players:
        kda = (player['kills'] + player['assists']) / max(player['deaths'], 1)
        if kda > best_kda:
            best_kda = kda
            best_player = player['name']
    
    return best_player

def q2(text):
    """Return the average number of kills in games that were won."""
    players = parse_data(text)
    
    won_games = [p for p in players if p['won']]
    if not won_games:
        return 0
    
    total_kills = sum(p['kills'] for p in won_games)
    return total_kills / len(won_games)

def q3(text):
    """Return the name of the most-picked champion."""
    players = parse_data(text)
    
    champions = [p['champion'] for p in players]
    champion_counts = Counter(champions)
    
    return champion_counts.most_common(1)[0][0]
```
